// src/lib/ai/llm/pipeline.ts
//
// Full multi-model pipeline — 5 stages for generation, 3 stages for editing.
//
//  Stage 1 — kimi-k2-instruct       → JSON project spec (planner)
//  Stage 2 — your code              → split spec into per-file tasks
//  Stage 3 — deepseek-v3.2          → parallel codegen (one call per file)
//  Stage 4 — phi-3.5-mini-instruct  → syntax fix on broken JSX only
//  Stage 5 — your code              → assemble + ensureBoilerplate
//
//  Edit path (only affected files):
//  kimi → identifies which files change
//  deepseek → rewrites them in parallel
//  phi → validates the edited files
//

import OpenAI from "openai";
import { AIFileNode } from "../types/ai-project.types";
import {
  generateFilesInParallel,
  rewriteFilesWithDeepseek,
  type ProjectSpec,
} from "./deepseek-generator";
import { validateFilesWithPhi } from "./phi-fixer";

const client = new OpenAI({
  apiKey: process.env.NVIDIA_API_KEY!,
  baseURL: "https://integrate.api.nvidia.com/v1",
});

// ─────────────────────────────────────────────
// STAGE 1 — PLANNER  (kimi-k2-instruct)
// Small output, pure JSON. kimi excels at structured planning.
// max_tokens: 2000  temp: 0.2
// ─────────────────────────────────────────────

export async function planProject(
  userPrompt: string
): Promise<ProjectSpec | null> {
  console.log("🧠 [STAGE 1] kimi-k2-instruct — planning");

  try {
    const completion = await client.chat.completions.create({
      model: "moonshotai/kimi-k2-instruct",
      temperature: 0.2,
      max_tokens: 2000,
      messages: [
        {
          role: "system",
          content: `You are a React architect. Return ONLY raw JSON — no markdown, no backticks, no explanation. Start with { and end with }.

{
  "projectName": "PascalCase no spaces",
  "description": "one sentence",
  "globalContext": "2-3 sentences: shared design system, color palette, font, spacing used across all files",
  "files": [
    {
      "path": "src/components/Header.jsx",
      "description": "sticky nav with logo, links, CTA button — glassmorphism style",
      "dependencies": []
    }
  ]
}

Rules:
- Always include: src/App.jsx, src/App.css, src/main.jsx, index.html, package.json, vite.config.js
- One component per file; break every visual section into its own component
- No react-router-dom; no libraries besides react and react-dom
- For a landing page: 8–15 component files. For a large site: up to 30 files
- Each description must be specific enough to code from on its own
- List ALL files you want generated`,
        },
        {
          role: "user",
          content: userPrompt,
        },
      ],
    });

    let text = completion.choices[0]?.message?.content || "";
    text = text.replace(/```json/gi, "").replace(/```/g, "").trim();

    const start = text.indexOf("{");
    const end = text.lastIndexOf("}");

    if (start === -1 || end === -1) {
      console.error("❌ [STAGE 1] No JSON object found in planner response");
      return null;
    }

    try {
      const spec = JSON.parse(text.slice(start, end + 1)) as ProjectSpec;
      console.log(
        `✅ [STAGE 1] Plan: ${spec.files.length} files — ${spec.files.map((f) => f.path).join(", ")}`
      );
      return spec;
    } catch {
      // Attempt trailing-comma repair
      const fixed = text
        .slice(start, end + 1)
        .replace(/,(\s*[}\]])/g, "$1");
      const spec = JSON.parse(fixed) as ProjectSpec;
      console.log(`✅ [STAGE 1] Plan recovered: ${spec.files.length} files`);
      return spec;
    }
  } catch (err) {
    console.error("❌ [STAGE 1] Planner failed:", err);
    return null;
  }
}

// ─────────────────────────────────────────────
// STAGE 2 — COMPONENT SPLITTER  (your code)
// Splits the JSON spec into per-file task descriptors.
// No API call. Validates the spec and normalises paths.
// ─────────────────────────────────────────────

function splitPlanIntoTasks(spec: ProjectSpec): ProjectSpec {
  console.log(`📦 [STAGE 2] Splitting ${spec.files.length} file tasks`);

  const normalised: ProjectSpec = {
    ...spec,
    files: spec.files.map((f) => ({
      path: f.path.replace(/^\/+/, ""), // strip leading slashes
      description: f.description || `${f.path} component`,
      dependencies: Array.isArray(f.dependencies) ? f.dependencies : [],
    })),
  };

  console.log(
    `✅ [STAGE 2] Tasks ready: ${normalised.files.map((f) => f.path).join(", ")}`
  );
  return normalised;
}

// ─────────────────────────────────────────────
// STAGE 5 — ASSEMBLER  (your code)
// Merges generated files, writes package.json, vite.config,
// index.html, auto-builds App.jsx if missing, and — crucially —
// scans every JSX file for local CSS imports and stubs any that
// don't exist yet (prevents Vite "does the file exist?" crashes).
// ─────────────────────────────────────────────

/**
 * Parse all `import "./Foo.css"` and `import "../bar.css"` statements
 * from a JSX file and resolve them to project-relative paths.
 *
 * e.g. file at "src/components/Header.jsx" importing "./Header.css"
 *      → "src/components/Header.css"
 */
function extractLocalCssImports(file: AIFileNode): string[] {
  // Match: import "./Something.css"  or  import '../Something.css'
  const importRegex = /import\s+['"](\.[^'"]+\.css)['"]/g;
  const dir = file.path.split("/").slice(0, -1).join("/");
  const resolved: string[] = [];

  let match: RegExpExecArray | null;
  while ((match = importRegex.exec(file.content)) !== null) {
    const rel = match[1]; // e.g. "./Header.css"

    // Resolve relative to the file's directory
    const segments = `${dir}/${rel}`.split("/");
    const normalised: string[] = [];
    for (const seg of segments) {
      if (seg === "..") normalised.pop();
      else if (seg !== ".") normalised.push(seg);
    }
    resolved.push(normalised.join("/"));
  }
  return resolved;
}

/**
 * Generate a minimal stub CSS for a component based on its name.
 * Produces a scoped class reset + a couple of sensible defaults so
 * the component renders without layout breakage.
 */
function stubCss(cssPath: string): string {
  // Derive a BEM-style root class from the filename, e.g. "header"
  const name = cssPath
    .split("/")
    .pop()!
    .replace(".css", "")
    .toLowerCase();

  return `/* ${cssPath} — auto-generated stub */
.${name} {
  box-sizing: border-box;
}

.${name} * {
  box-sizing: border-box;
}
`;
}

function ensureBoilerplate(
  files: AIFileNode[],
  projectName: string
): AIFileNode[] {
  const map = new Map(files.map((f) => [f.path, f]));

  const ensure = (path: string, content: string) => {
    if (!map.has(path)) {
      map.set(path, { path, content, type: "file" });
      console.log(`➕ [STAGE 5] Ensured: ${path}`);
    }
  };

  ensure("package.json", makePackageJson(projectName));
  ensure("index.html", makeIndexHtml(projectName));
  ensure("src/main.jsx", makeMainJsx());
  ensure("vite.config.js", makeViteConfig());

  // Auto-build App.jsx from available components if missing
  if (!map.has("src/App.jsx")) {
    const components = Array.from(map.keys())
      .filter(
        (p) => p.startsWith("src/components/") && p.endsWith(".jsx")
      )
      .map((p) => ({
        name: p.split("/").pop()!.replace(".jsx", ""),
        importPath: "./" + p.replace("src/", "").replace(".jsx", ""),
      }));

    const imports = components
      .map((c) => `import ${c.name} from '${c.importPath}';`)
      .join("\n");
    const renders = components
      .map((c) => `      <${c.name} />`)
      .join("\n");

    map.set("src/App.jsx", {
      path: "src/App.jsx",
      content: `import React from 'react';\nimport './App.css';\n${imports}\n\nfunction App() {\n  return (\n    <div className="app">\n${renders}\n    </div>\n  );\n}\n\nexport default App;`,
      type: "file",
    });
    console.log("➕ [STAGE 5] Auto-built src/App.jsx");
  }

  if (!map.has("src/App.css")) {
    map.set("src/App.css", {
      path: "src/App.css",
      content: `* { margin: 0; padding: 0; box-sizing: border-box; }\nbody { font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif; }\n.app { min-height: 100vh; }`,
      type: "file",
    });
  }

  // ── CSS stub pass ─────────────────────────────────────────────────────
  // deepseek freely writes  import "./Header.css"  inside components but
  // the planner never lists those CSS files in the spec, so they're never
  // generated → Vite crashes with "does the file exist?".
  //
  // Solution: scan every JSX file, resolve every local .css import, and
  // create a stub for any path that isn't already in the map.
  // Zero extra API calls — pure string processing.
  const jsxFiles = Array.from(map.values()).filter((f) =>
    f.path.endsWith(".jsx")
  );

  let stubbedCount = 0;
  for (const jsx of jsxFiles) {
    const cssPaths = extractLocalCssImports(jsx);
    for (const cssPath of cssPaths) {
      if (!map.has(cssPath)) {
        map.set(cssPath, {
          path: cssPath,
          content: stubCss(cssPath),
          type: "file",
        });
        stubbedCount++;
        console.log(`🎨 [STAGE 5] Stubbed missing CSS: ${cssPath}`);
      }
    }
  }

  if (stubbedCount > 0) {
    console.log(`🎨 [STAGE 5] Created ${stubbedCount} CSS stub(s) for missing imports`);
  }
  // ─────────────────────────────────────────────────────────────────────

  const result = Array.from(map.values());
  console.log(`✅ [STAGE 5] Final project: ${result.length} files`);
  return result;
}

// ─────────────────────────────────────────────
// MAIN EXPORT — full 5-stage generation pipeline
// ─────────────────────────────────────────────

export async function generateProjectWithPipeline(
  userPrompt: string
): Promise<AIFileNode[]> {
  console.log("🚀 ═══════════════════════════════");
  console.log("🚀 PIPELINE START — GENERATION");
  console.log("🚀 ═══════════════════════════════");

  // ── Stage 1: Plan ──────────────────────────
  const rawSpec = await planProject(userPrompt);
  if (!rawSpec) {
    console.error("❌ Planning failed — using minimal fallback");
    return minimalFallback(userPrompt);
  }

  // ── Stage 2: Split ─────────────────────────
  const spec = splitPlanIntoTasks(rawSpec);

  // ── Stage 3: Parallel codegen (deepseek) ───
  const generated = await generateFilesInParallel(spec);

  // ── Stage 4: Validate / fix (phi) ──────────
  const validated = await validateFilesWithPhi(generated);

  // ── Stage 5: Assemble ──────────────────────
  const result = ensureBoilerplate(validated, spec.projectName);

  console.log("✅ ═══════════════════════════════");
  console.log(`✅ PIPELINE DONE — ${result.length} files`);
  console.log("✅ ═══════════════════════════════");

  return result;
}

// ─────────────────────────────────────────────
// EDIT PIPELINE — 3-stage edit path
// kimi → identifies which files change
// deepseek → rewrites them in parallel
// phi → validates edited files
// ─────────────────────────────────────────────

export async function editProjectWithPipeline(
  instructions: string,
  existingFiles: AIFileNode[]
): Promise<AIFileNode[]> {
  console.log("✏️ ═══════════════════════════════");
  console.log("✏️ EDIT PIPELINE START");
  console.log("✏️ ═══════════════════════════════");

  const fileMap = new Map(existingFiles.map((f) => [f.path, f]));

  // ── Edit Stage 1: kimi identifies affected files ──
  console.log("🧠 [EDIT 1] kimi-k2-instruct — identifying files to change");

  let pathsToEdit: string[] = [];

  try {
    const fileList = existingFiles.map((f) => `  - ${f.path}`).join("\n");

    const planCompletion = await client.chat.completions.create({
      model: "moonshotai/kimi-k2-instruct",
      temperature: 0.2,
      max_tokens: 400,
      messages: [
        {
          role: "system",
          content: `You are a React architect. Return ONLY a raw JSON array of file paths that need to change.
Example: ["src/App.css", "src/components/Header.jsx"]
No markdown. No explanation. Start with [ and end with ].`,
        },
        {
          role: "user",
          content: `Edit request: "${instructions}"\n\nProject files:\n${fileList}`,
        },
      ],
    });

    let planText = planCompletion.choices[0]?.message?.content || "[]";
    planText = planText
      .replace(/```json/gi, "")
      .replace(/```/g, "")
      .trim();

    const s = planText.indexOf("[");
    const e = planText.lastIndexOf("]");
    if (s !== -1 && e !== -1) {
      pathsToEdit = JSON.parse(planText.slice(s, e + 1)) as string[];
    }

    console.log(`✅ [EDIT 1] kimi identified: ${pathsToEdit.join(", ")}`);
  } catch (err) {
    console.warn("⚠️ [EDIT 1] kimi failed — defaulting to App.jsx + App.css:", err);
    pathsToEdit = existingFiles.some((f) => f.path === "src/App.css")
      ? ["src/App.css", "src/App.jsx"]
      : ["src/App.jsx"];
  }

  if (pathsToEdit.length === 0) {
    console.warn("⚠️ [EDIT 1] Empty file list — defaulting to App.jsx + App.css");
    pathsToEdit = ["src/App.jsx", "src/App.css"];
  }

  // ── Edit Stage 2: deepseek rewrites affected files ──
  const filesToEdit = pathsToEdit
    .map((p) => fileMap.get(p))
    .filter((f): f is AIFileNode => !!f);

  const rewritten = await rewriteFilesWithDeepseek(instructions, filesToEdit);

  // ── Edit Stage 3: phi validates edited files ──
  const validated = await validateFilesWithPhi(rewritten);

  // Merge back into the full file list (unchanged files stay intact)
  validated.forEach((f) => fileMap.set(f.path, f));

  const result = Array.from(fileMap.values());

  console.log("✅ ═══════════════════════════════");
  console.log(`✅ EDIT DONE — ${validated.length} file(s) updated`);
  console.log("✅ ═══════════════════════════════");

  return result;
}

function makePackageJson(projectName: string): string {
  return JSON.stringify(
    {
      name: projectName.toLowerCase().replace(/\s+/g, "-"),
      private: true,
      version: "0.0.0",
      type: "module",
      scripts: { dev: "vite" },
      dependencies: { react: "^18.2.0", "react-dom": "^18.2.0" },
      devDependencies: {
        vite: "^5.0.0",
        "@vitejs/plugin-react": "^4.0.0",
      },
    },
    null,
    2
  );
}

function makeIndexHtml(projectName: string): string {
  return `<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>${projectName}</title>
  </head>
  <body>
    <div id="root"></div>
    <script type="module" src="/src/main.jsx"></script>
  </body>
</html>`;
}

function makeMainJsx(): string {
  return `import React from 'react';
import ReactDOM from 'react-dom/client';
import App from './App';
import './App.css';

ReactDOM.createRoot(document.getElementById('root')).render(
  <React.StrictMode>
    <App />
  </React.StrictMode>
);`;
}

function makeViteConfig(): string {
  return `import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";

export default defineConfig({
  plugins: [react()],
});`;
}

function minimalFallback(prompt: string): AIFileNode[] {
  const name = JSON.stringify(prompt.slice(0, 40) || "AI App");
  return [
    {
      path: "package.json",
      content: makePackageJson("ai-app"),
      type: "file",
    },
    { path: "index.html", content: makeIndexHtml("AI App"), type: "file" },
    { path: "src/main.jsx", content: makeMainJsx(), type: "file" },
    { path: "vite.config.js", content: makeViteConfig(), type: "file" },
    {
      path: "src/App.jsx",
      content: `import React from 'react';\nimport './App.css';\n\nconst TITLE = ${name};\n\nfunction App() {\n  return <div className="app"><h1>{TITLE}</h1></div>;\n}\n\nexport default App;`,
      type: "file",
    },
    {
      path: "src/App.css",
      content: `.app { text-align: center; padding: 40px; font-family: sans-serif; }`,
      type: "file",
    },
  ];
}

export async function generateCodeFromAI(input: string): Promise<AIFile[]> {
  console.log("🔥 Generating code with Kimi K2...");

  let rawText = "";

  try {
    const completion = await client.chat.completions.create({
      model: "moonshotai/kimi-k2-instruct",
      messages: [
        { role: "system", content: SYSTEM_PROMPT },
        {
          role: "user",
          content: `Project Plan or Prompt:\n${input}\n\nGenerate the full project code now. Return only the JSON array.`,
        },
      ],
      temperature: 0.2,
      top_p: 0.9,
      max_tokens: 32768,
    });

    rawText = completion.choices[0]?.message?.content ?? "";

    if (!rawText.trim()) {
      console.error("❌ Empty response from model");
      return fallbackFile("Model returned an empty response.");
    }

    const cleaned = stripMarkdown(rawText);
    const jsonString = extractJsonArray(cleaned);

    if (!jsonString) {
      console.error("❌ No JSON array found in response");
      return fallbackFile("No JSON array detected in model response.");
    }

    const parsed = repairAndParse(jsonString);

    if (!parsed) {
      console.error("❌ JSON parse failed after repair attempts");
      return fallbackFile("JSON parsing failed. Model output may be malformed.");
    }

    if (!Array.isArray(parsed)) {
      return fallbackFile("Model returned JSON but it was not an array.");
    }

    if (parsed.length === 0) {
      return fallbackFile("Model returned an empty file list.");
    }

    const files: AIFile[] = [];
    const pathsSeen = new Set<string>();

    for (let i = 0; i < parsed.length; i++) {
      const item = parsed[i] as RawAIFile;

      if (typeof item !== "object" || item === null) continue;

      const filePath = sanitizePath(item.path);
      if (!filePath) continue;
      if (pathsSeen.has(filePath)) continue;
      pathsSeen.add(filePath);

      const content = sanitizeContent(item.content);
      files.push({ path: filePath, content });
    }

    if (files.length === 0) {
      return fallbackFile("All generated files failed validation.");
    }

    console.log(`✅ Generated ${files.length} file(s)`);
    return files;

  } catch (error: unknown) {
    const message = error instanceof Error ? error.message : String(error);
    console.error("❌ Fatal error during AI generation:", message);
    return fallbackFile(`AI request failed: ${message}`);
  }
}

function stripMarkdown(text: string): string {
  return text
    .replace(/^```[a-zA-Z]*\s*/gm, "")
    .replace(/^```\s*$/gm, "")
    .trim();
}

function extractJsonArray(text: string): string | null {
  const start = text.indexOf("[");
  if (start === -1) return null;

  let depth = 0;
  let inString = false;
  let escapeNext = false;

  for (let i = start; i < text.length; i++) {
    const ch = text[i];

    if (escapeNext) { escapeNext = false; continue; }
    if (ch === "\\" && inString) { escapeNext = true; continue; }
    if (ch === '"') { inString = !inString; continue; }
    if (inString) continue;

    if (ch === "[") depth++;
    else if (ch === "]") {
      depth--;
      if (depth === 0) return text.slice(start, i + 1);
    }
  }

  const last = text.lastIndexOf("]");
  if (last > start) return text.slice(start, last + 1);
  return null;
}

function repairAndParse(jsonString: string): unknown | null {
  // Strategy 1: as-is
  try { return JSON.parse(jsonString); } catch (_) {}

  // Strategy 2: remove trailing commas
  try {
    const fixed = jsonString.replace(/,\s*]/g, "]").replace(/,\s*}/g, "}");
    return JSON.parse(fixed);
  } catch (_) {}

  // Strategy 3: normalize line endings + strip control chars
  try {
    const fixed = jsonString
      .replace(/\r\n/g, "\\n")
      .replace(/\r/g, "\\n")
      .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, "")
      .replace(/,\s*]/g, "]")
      .replace(/,\s*}/g, "}");
    return JSON.parse(fixed);
  } catch (_) {}

  // Strategy 4: regex salvage
  try { return salvageFilesFromText(jsonString); } catch (_) {}

  return null;
}

function salvageFilesFromText(text: string): AIFile[] | null {
  const results: AIFile[] = [];
  const pattern = /\{\s*"path"\s*:\s*"([^"]+)"\s*,\s*"content"\s*:\s*"([\s\S]*?)"\s*\}/g;

  let match: RegExpExecArray | null;
  while ((match = pattern.exec(text)) !== null) {
    const filePath = match[1].trim();
    const content = match[2]
      .replace(/\\n/g, "\n")
      .replace(/\\t/g, "\t")
      .replace(/\\"/g, '"')
      .replace(/\\\\/g, "\\");
    if (filePath) results.push({ path: filePath, content });
  }

  return results.length > 0 ? results : null;
}

function sanitizePath(raw: unknown): string | null {
  if (typeof raw !== "string") return null;

  const p = raw
    .trim()
    .replace(/\\/g, "/")
    .replace(/^\/+/, "")
    .replace(/\.\.+\//g, "")
    .replace(/[<>:"|?*\x00-\x1F]/g, "");

  if (!p || !/\.\w+$/.test(p)) return null;
  return p;
}

function sanitizeContent(raw: unknown): string {
  if (raw === null || raw === undefined) return "";

  let content = typeof raw === "string" ? raw : String(raw);

  content = content
    .replace(/\\n/g, "\n")
    .replace(/\\t/g, "\t")
    .replace(/\\r/g, "")
    .replace(/\\"/g, '"')
    .replace(/\\\\/g, "\\");

  // Fix common AI typos
  content = content
    .replace(/<\/buutton>/g, "</button>")
    .replace(/<\/buton>/g, "</button>")
    .replace(/<\/diiv>/g, "</div>")
    .replace(/<\/spann>/g, "</span>");

  // Strip lone 'n' lines (model artifact)
  content = content.replace(/^\s*n\s*$/gm, "");

  // Collapse 3+ blank lines to 2
  content = content.replace(/\n{3,}/g, "\n\n");

  return content;
}

// ================================================================
// AUTO-FIX PIPELINE
// Runs every AI-generated file through a series of fixers before
// sending to WebContainers. Handles the most common model errors:
// malformed JSX, broken imports, missing exports, bad escaping,
// duplicate code, broken Tailwind, and more.
// ================================================================

export type AIFile = {
  path: string;
  content: string;
};

type FixResult = {
  file: AIFile;
  fixes: string[];
  warnings: string[];
};

// ================================================================
// PUBLIC ENTRY POINT
// ================================================================
export function autoFixFiles(files: AIFile[]): AIFile[] {
  // Pass 1: fix individual files
  const results = files.map(fixFile);

  // Pass 2: cross-file fixes (dedup imports, sync exports)
  const fixed = crossFileFixes(results.map((r) => r.file));

  // Log summary
  results.forEach((r) => {
    if (r.fixes.length > 0) {
      console.log(`🔧 [${r.file.path}] ${r.fixes.length} fix(es):`, r.fixes);
    }
    if (r.warnings.length > 0) {
      console.warn(`⚠️  [${r.file.path}] ${r.warnings.length} warning(s):`, r.warnings);
    }
  });

  return fixed;
}

// ================================================================
// PER-FILE FIX PIPELINE
// ================================================================
function fixFile(file: AIFile): FixResult {
  const fixes: string[] = [];
  const warnings: string[] = [];

  const ext = file.path.split(".").pop()?.toLowerCase() ?? "";
  const isJS = ["js", "jsx", "ts", "tsx"].includes(ext);
  const isCSS = ["css", "scss"].includes(ext);
  const isJSON = ext === "json";
  const isHTML = ext === "html";

  let code = file.content ?? "";

  // ── Universal fixes ──────────────────────────────────────────
  code = fixEncoding(code, fixes);
  code = fixLineEndings(code, fixes);
  code = stripBOM(code, fixes);
  code = collapseBlankLines(code, fixes);

  // ── JS/JSX/TS/TSX fixes ──────────────────────────────────────
  if (isJS) {
    code = fixEscapeArtifacts(code, fixes);
    code = fixTypos(code, fixes);
    code = fixSingleLetterLines(code, fixes);
    code = fixBrokenJSXTags(code, fixes);
    code = fixSelfClosingTags(code, fixes);
    code = fixMissingReactImport(code, file.path, fixes);
    code = fixImports(code, fixes, warnings);
    code = fixExports(code, file.path, fixes);
    code = fixUseClientDirective(code, file.path, fixes);
    code = fixHookRules(code, fixes, warnings);
    code = fixTailwindClasses(code, fixes);
    code = fixCommonPatterns(code, fixes);
    code = fixBrokenTemplateLiterals(code, fixes);
    code = fixConsoleStatements(code, fixes);
  }

  // ── CSS fixes ────────────────────────────────────────────────
  if (isCSS) {
    code = fixCSS(code, fixes);
  }

  // ── JSON fixes ───────────────────────────────────────────────
  if (isJSON) {
    code = fixJSON(code, fixes, warnings);
  }

  // ── HTML fixes ───────────────────────────────────────────────
  if (isHTML) {
    code = fixHTML(code, fixes);
  }

  return {
    file: { path: file.path, content: code },
    fixes,
    warnings,
  };
}

// ================================================================
// ENCODING / WHITESPACE
// ================================================================
function fixEncoding(code: string, fixes: string[]): string {
  // Strip BOM variants
  let fixed = code.replace(/^\uFEFF/, "");
  // Strip null bytes
  fixed = fixed.replace(/\0/g, "");
  // Strip other invisible control characters (keep \t \n \r)
  fixed = fixed.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, "");
  // Fix zero-width spaces / non-breaking spaces inside code
  fixed = fixed.replace(/[\u200B\u200C\u200D\uFEFF]/g, "");
  if (fixed !== code) fixes.push("encoding-control-chars");
  return fixed;
}

function fixLineEndings(code: string, fixes: string[]): string {
  if (!code.includes("\r")) return code;
  fixes.push("line-endings-crlf");
  return code.replace(/\r\n/g, "\n").replace(/\r/g, "\n");
}

function stripBOM(code: string, fixes: string[]): string {
  if (!code.startsWith("\uFEFF")) return code;
  fixes.push("bom-stripped");
  return code.slice(1);
}

function collapseBlankLines(code: string, fixes: string[]): string {
  const fixed = code.replace(/\n{4,}/g, "\n\n\n");
  if (fixed !== code) fixes.push("collapsed-blank-lines");
  return fixed;
}

// ================================================================
// ESCAPE ARTIFACTS
// ================================================================
function fixEscapeArtifacts(code: string, fixes: string[]): string {
  let fixed = code;

  // Double-escaped newlines from JSON serialization
  fixed = fixed.replace(/\\\\n/g, "\n");
  fixed = fixed.replace(/\\\\t/g, "\t");
  fixed = fixed.replace(/\\\\r/g, "");

  // Single-escaped if still encoded
  fixed = fixed.replace(/(?<!\\)\\n(?!["\\])/g, "\n");
  fixed = fixed.replace(/(?<!\\)\\t(?!["\\])/g, "\t");

  // Escaped quotes outside of strings (model artifact)
  // Only fix in JSX/code context, not inside string literals
  fixed = fixed.replace(/\\"([^"]*)\\"(?=\s*[,;)\]}])/g, '"$1"');

  if (fixed !== code) fixes.push("escape-artifacts");
  return fixed;
}

// TYPOS & COMMON MODEL ERRORS
// ================================================================
const TYPO_MAP: [RegExp, string][] = [
  // Closing tag typos
  [/<\/buutton>/g, "</button>"],
  [/<\/buton>/g, "</button>"],
  [/<\/diiv>/g, "</div>"],
  [/<\/spann>/g, "</span>"],
  [/<\/imput>/g, "</input>"],
  [/<\/lable>/g, "</label>"],
  [/<\/tabble>/g, "</table>"],
  [/<\/tabel>/g, "</table>"],
  [/<\/thead>/g, "</thead>"],  // already correct but normalize case
  [/<\/Div>/g, "</div>"],
  [/<\/Span>/g, "</span>"],
  [/<\/P>/g, "</p>"],
  [/<\/H1>/g, "</h1>"],
  [/<\/H2>/g, "</h2>"],
  // Opening tag typos
  [/<buutton/g, "<button"],
  [/<buton/g, "<button"],
  [/<diiv/g, "<div"],
  [/<imput/g, "<input"],
  [/<lable/g, "<label"],
  // className typos
  [/classname=/g, "className="],
  [/Classname=/g, "className="],
  [/class=(?!Name)/g, "className="],
  // onClick typos
  [/onclick=/g, "onClick="],
  [/Onclick=/g, "onClick="],
  // htmlFor
  [/for=(?!")/g, "htmlFor="],
  // Common JS typos
  [/consoloe\./g, "console."],
  [/consle\./g, "console."],
  [/fucntion /g, "function "],
  [/funtion /g, "function "],
  [/retrun /g, "return "],
  [/retrn /g, "return "],
  [/improt /g, "import "],
  [/exprot /g, "export "],
  // React hook typos
  [/userState/g, "useState"],
  [/usestate/g, "useState"],
  [/useEffekt/g, "useEffect"],
  [/useMemo\b/g, "useMemo"],
  [/useCallbakc/g, "useCallback"],
  [/useContxt/g, "useContext"],
];

function fixTypos(code: string, fixes: string[]): string {
  let fixed = code;
  let changed = false;

  for (const [pattern, replacement] of TYPO_MAP) {
    const next = fixed.replace(pattern, replacement);
    if (next !== fixed) { changed = true; fixed = next; }
  }

  if (changed) fixes.push("typos");
  return fixed;
}

  const fixed = code.replace(/^[ \t]*[nseNSE][ \t]*$/gm, "");
  if (fixed !== code) fixes.push("single-letter-artifact-lines");

// ================================================================
// BROKEN JSX TAGS
// ================================================================
function fixBrokenJSXTags(code: string, fixes: string[]): string {
  let fixed = code;
  let changed = false;

  // Unclosed JSX expression braces in attributes: className={styles.foo
  // (very hard to auto-fix reliably — just warn, don't break things)

  // Fix <Component/> → <Component /> (space before />)
  // Actually many formatters prefer without space — skip this

  // Fix stray > that breaks JSX: <div className="foo">content</div >
  const divWithSpace = fixed.replace(/<\/\s+(\w)/g, "</$1");
  if (divWithSpace !== fixed) { changed = true; fixed = divWithSpace; }

  // Fix missing = in attributes: <Button variant"primary"> → variant="primary"
  const missingEq = fixed.replace(/(\s)(\w+)"([^"]*)"(?=[\s/>])/g, '$1$2="$3"');
  if (missingEq !== fixed) { changed = true; fixed = missingEq; }

  if (changed) fixes.push("broken-jsx-tags");
  return fixed;
}

// ================================================================
// SELF-CLOSING TAGS
// HTML void elements that must be self-closing in JSX
// ================================================================
const VOID_ELEMENTS = new Set([
  "area", "base", "br", "col", "embed", "hr", "img",
  "input", "link", "meta", "param", "source", "track", "wbr",
]);

function fixSelfClosingTags(code: string, fixes: string[]): string {
  let fixed = code;
  let changed = false;

  for (const tag of VOID_ELEMENTS) {
    // <br> → <br /> and <br></br> → <br />
    const unclosed = new RegExp(`<${tag}([^/]*)(?<!/)>(?!</${tag}>)`, "gi");
    const next1 = fixed.replace(unclosed, `<${tag}$1 />`);
    if (next1 !== fixed) { changed = true; fixed = next1; }

    const withClose = new RegExp(`<${tag}([^>]*)></${tag}>`, "gi");
    const next2 = fixed.replace(withClose, `<${tag}$1 />`);
    if (next2 !== fixed) { changed = true; fixed = next2; }
  }

  if (changed) fixes.push("self-closing-void-elements");
  return fixed;
}

function fixMissingReactImport(
  code: string,
  filePath: string,
  fixes: string[]
): string {
  const hasJSX = /<[A-Z][a-zA-Z]*|<[a-z]+[\s/>]/.test(code);
  const hasReactImport = /import\s+React/.test(code);
  const isTypeFile = filePath.endsWith(".d.ts") || filePath.includes("types");

  // React 17+ doesn't need import for JSX transform,
  // but hooks still need explicit React or named imports.
  // Only add if using React.* directly or old-style JSX.
  const usesReactDirectly = /\bReact\.(use|create|memo|forwardRef|lazy|Suspense|Fragment|cloneElement|Children)/.test(code);

  if (hasJSX && usesReactDirectly && !hasReactImport && !isTypeFile) {
    fixes.push("added-react-import");
    return `import React from "react";\n${code}`;
  }

  return code;
}

// ================================================================
// IMPORTS
// ================================================================
function fixImports(code: string, fixes: string[], warnings: string[]): string {
  let fixed = code;
  let changed = false;

  const lines = fixed.split("\n");
  const importLines: string[] = [];
  const nonImportLines: string[] = [];
  let inImportBlock = true;

  // Separate import block from the rest
  // (imports should all be at the top)
  for (const line of lines) {
    const trimmed = line.trim();
    if (
      inImportBlock &&
      (trimmed.startsWith("import ") ||
        trimmed.startsWith("// ") ||
        trimmed === "" ||
        trimmed.startsWith("'use ") ||
        trimmed.startsWith('"use '))
    ) {
      importLines.push(line);
    } else {
      inImportBlock = false;
      nonImportLines.push(line);
    }
  }

  // Deduplicate import lines (exact match)
  const seenImports = new Set<string>();
  const dedupedImports = importLines.filter((line) => {
    const trimmed = line.trim();
    if (!trimmed.startsWith("import ")) return true;
    if (seenImports.has(trimmed)) { changed = true; return false; }
    seenImports.add(trimmed);
    return true;
  });

  // Merge named imports from the same source
  // e.g. import { A } from 'x' + import { B } from 'x' → import { A, B } from 'x'
  const mergedImports = mergeNamedImports(dedupedImports, fixes);

  // Fix common wrong import paths
  const correctedImports = mergedImports.map((line) => {
    let l = line;
    // Fix: import from 'react-router' → 'react-router-dom'
    l = l.replace(/from ['"]react-router['"]/g, 'from "react-router-dom"');
    // Fix: import { ... } from '@/components/ui/button' → '@/components/ui'
    // (only if the component is in our pre-built ui.jsx)
    const prebuilt = ["Button", "Input", "Textarea", "Select", "Card", "Badge",
      "Spinner", "Modal", "Skeleton", "Avatar", "Divider", "Alert",
      "Table", "Progress", "Tooltip"];
    for (const comp of prebuilt) {
      const wrongPath = new RegExp(
        `import\\s*\\{([^}]*)\\b${comp}\\b([^}]*)\\}\\s*from\\s*['"]@/components/ui/${comp.toLowerCase()}['"]`,
        "g"
      );
      if (wrongPath.test(l)) {
        l = l.replace(wrongPath, `import {$1${comp}$2} from "@/components/ui"`);
        changed = true;
      }
    }
    // Fix double slashes in paths
    l = l.replace(/from ['"]([^'"]*\/\/[^'"]*)['"]/g, (m, p) =>
      m.replace(p, p.replace(/\/\//g, "/"))
    );
    if (l !== line) changed = true;
    return l;
  });

  if (changed) {
    fixes.push("imports-deduped-merged");
    fixed = [...correctedImports, ...nonImportLines].join("\n");
  }

  // Warn about missing imports (common patterns)
  if (/\buseState\b/.test(fixed) && !/import.*useState.*from ['"]react['"]/.test(fixed)) {
    warnings.push("useState used but not imported from react");
  }
  if (/\buseEffect\b/.test(fixed) && !/import.*useEffect.*from ['"]react['"]/.test(fixed)) {
    warnings.push("useEffect used but not imported from react");
  }
  if (/\bLink\b/.test(fixed) && !/import.*Link.*from ['"]react-router/.test(fixed)) {
    warnings.push("Link used but not imported from react-router-dom");
  }

  return fixed;
}

function mergeNamedImports(lines: string[], fixes: string[]): string[] {
  // Map: source → Set of named imports
  const sourceMap = new Map<string, Set<string>>();
  const defaultImports = new Map<string, string>();
  const otherLines: string[] = [];

  const namedImportRe = /^import\s+\{([^}]+)\}\s+from\s+['"]([^'"]+)['"]/;
  const defaultImportRe = /^import\s+(\w+)\s+from\s+['"]([^'"]+)['"]/;

  for (const line of lines) {
    const trimmed = line.trim();

    const namedMatch = trimmed.match(namedImportRe);
    if (namedMatch) {
      const named = namedMatch[1].split(",").map((s) => s.trim()).filter(Boolean);
      const source = namedMatch[2];
      if (!sourceMap.has(source)) sourceMap.set(source, new Set());
      named.forEach((n) => sourceMap.get(source)!.add(n));
      continue;
    }

    const defaultMatch = trimmed.match(defaultImportRe);
    if (defaultMatch) {
      defaultImports.set(defaultMatch[2], defaultMatch[1]);
      continue;
    }

    otherLines.push(line);
  }

  const merged: string[] = [];

  // Reconstruct default imports
  for (const [source, name] of defaultImports) {
    const named = sourceMap.get(source);
    if (named && named.size > 0) {
      merged.push(`import ${name}, { ${[...named].join(", ")} } from "${source}";`);
      sourceMap.delete(source);
    } else {
      merged.push(`import ${name} from "${source}";`);
    }
  }

  // Reconstruct named imports (sources without a default import)
  for (const [source, named] of sourceMap) {
    if (named.size > 0) {
      merged.push(`import { ${[...named].join(", ")} } from "${source}";`);
    }
  }

  if (merged.length > 0) fixes.push("imports-merged-by-source");

  return [...merged, ...otherLines];
}

// ================================================================
// EXPORTS
// ================================================================
function fixExports(code: string, filePath: string, fixes: string[]): string {
  let fixed = code;

  const fileName = filePath
    .split("/")
    .pop()!
    .replace(/\.[^.]+$/, "");

  // Capitalize for component name inference
  const componentName =
    fileName.charAt(0).toUpperCase() + fileName.slice(1);

  // Case 1: `function App() {` without `export default` → add it
  const hasExportDefault = /export\s+default\s/.test(fixed);
  const funcMatch = fixed.match(/^function\s+([A-Z]\w*)\s*\(/m);

  if (!hasExportDefault && funcMatch) {
    fixed = fixed.replace(
      new RegExp(`^function\\s+${funcMatch[1]}`, "m"),
      `export default function ${funcMatch[1]}`
    );
    fixes.push(`added-export-default-to-${funcMatch[1]}`);
  }

  // Case 2: const Component = () => {} without export
  if (!hasExportDefault) {
    const constMatch = fixed.match(
      /^const\s+([A-Z]\w*)\s*=\s*(?:\([^)]*\)|[a-z_]\w*)\s*=>/m
    );
    if (constMatch) {
      fixed = fixed.replace(
        new RegExp(`^const\\s+${constMatch[1]}`, "m"),
        `export default const ${constMatch[1]}`
      );
      // Actually `export default const` is invalid — fix to add at bottom instead
      fixed = fixed.replace(
        `export default const ${constMatch[1]}`,
        `const ${constMatch[1]}`
      );
      fixed += `\n\nexport default ${constMatch[1]};\n`;
      fixes.push(`added-export-default-bottom-${constMatch[1]}`);
    }
  }

  return fixed;
}

// ================================================================
// USE CLIENT DIRECTIVE
// ================================================================
function fixUseClientDirective(
  code: string,
  filePath: string,
  fixes: string[]
): string {
  const hooksPattern =
    /\b(useState|useEffect|useRef|useCallback|useMemo|useContext|useReducer|useLayoutEffect|useId)\b/;
  const browserAPIs =
    /\b(window|document|navigator|localStorage|sessionStorage|addEventListener|removeEventListener)\b/;
  const eventHandlers = /\bon[A-Z]\w+\s*=\s*\{/;

  const needsUseClient =
    hooksPattern.test(code) ||
    browserAPIs.test(code) ||
    eventHandlers.test(code);

  const hasUseClient = /^['"]use client['"]/.test(code.trimStart());

  // Skip for non-component files
  const isPage = filePath.includes("/pages/") || filePath.includes("/app/");
  const isComponent =
    filePath.endsWith(".jsx") || filePath.endsWith(".tsx");

  if (needsUseClient && !hasUseClient && isComponent) {
    fixes.push("added-use-client");
    return `"use client";\n\n${code}`;
  }

  // Remove "use client" from files that don't need it (e.g. utility files)
  if (hasUseClient && !needsUseClient && !isPage) {
    const withoutDirective = code.replace(/^['"]use client['"];\s*\n*/m, "");
    if (withoutDirective !== code) {
      fixes.push("removed-unnecessary-use-client");
      return withoutDirective;
    }
  }

  return code;
}

// ================================================================
// HOOK RULES
// ================================================================
function fixHookRules(
  code: string,
  fixes: string[],
  warnings: string[]
): string {
  // Detect hooks called inside conditions (very rough check)
  const conditionalHook = /if\s*\([^)]*\)\s*\{[^}]*use[A-Z]\w+\s*\(/;
  if (conditionalHook.test(code)) {
    warnings.push("possible-hook-in-conditional — hooks must not be called conditionally");
  }

  // Detect hooks in loops
  const loopHook = /(?:for|while|forEach)\s*[(\[][^)\]]*\)[^{]*\{[^}]*use[A-Z]/;
  if (loopHook.test(code)) {
    warnings.push("possible-hook-in-loop — hooks must not be called inside loops");
  }

  return code;
}

// Common AI mistakes with Tailwind class names
type TailwindFix = [RegExp, string | ((match: string) => string)];

const TAILWIND_FIXES: TailwindFix[] = [
  // Deprecated classes
  [/\bflex-no-wrap\b/g, "flex-nowrap"],
  [/\bfont-hairline\b/g, "font-thin"],
  [/\bfont-extralight\b/g, "font-extralight"],
  [/\btruncate-text\b/g, "truncate"],
  [/\bbreakword\b/g, "break-words"],
  [/\bscreen-reader-only\b/g, "sr-only"],
  [/\bvisually-hidden\b/g, "sr-only"],
  // Non-existent classes AI commonly invents
  [/\btext-default\b/g, "text-base"],
  [/\bfont-regular\b/g, "font-normal"],
  [/\bcolor-primary\b/g, "text-primary-500"],
  [/\bbg-default\b/g, "bg-white"],
  [/\bborder-default\b/g, "border-gray-200"],
  [/\bpadding-(\d+)\b/g, (m: string) => m.replace("padding-", "p-")],
  [/\bmargin-(\d+)\b/g, (m: string) => m.replace("margin-", "m-")],
  // Wrong spacing for gap
  [/\bgap\s*-\s*(\d+)\b/g, "gap-$1"],
  // Common class name confusions
  [/\bfont-weight-bold\b/g, "font-bold"],
  [/\btext-align-center\b/g, "text-center"],
  [/\bdisplay-flex\b/g, "flex"],
  [/\bdisplay-block\b/g, "block"],
  [/\bdisplay-none\b/g, "hidden"],
  [/\boverflow-hidden-x\b/g, "overflow-x-hidden"],
  [/\boverflow-hidden-y\b/g, "overflow-y-hidden"],
  // Spacing typos
  [/\bp-x-(\d+)\b/g, "px-$1"],
  [/\bp-y-(\d+)\b/g, "py-$1"],
  [/\bm-x-(\d+)\b/g, "mx-$1"],
  [/\bm-y-(\d+)\b/g, "my-$1"],
];

function fixTailwindClasses(code: string, fixes: string[]): string {
  let fixed = code;
  let changed = false;

  for (const [pattern, replacement] of TAILWIND_FIXES) {
    const next = typeof replacement === "string"
      ? fixed.replace(pattern, replacement)
      : fixed.replace(pattern, replacement);
    if (next !== fixed) { changed = true; fixed = next; }
  }

  if (changed) fixes.push("tailwind-class-fixes");
  return fixed;
}

// ================================================================
// COMMON PATTERNS
// ================================================================
function fixCommonPatterns(code: string, fixes: string[]): string {
  let fixed = code;
  let changed = false;

  // Fix: key prop on root element of map (very common AI error — missing key)
  // We can't auto-add keys without knowing the data, but we can fix wrong ones
  // Fix: key={index} → warn only
  // Fix: .map(item => <div>) → add key if missing (use item.id if available)
  const mapWithoutKey = /\.map\((\w+)\s*=>\s*(<[A-Za-z])/g;
  const mapWithKey = /\.map\((\w+)\s*=>\s*<[A-Za-z][^>]*key=/g;

  if (mapWithoutKey.test(fixed) && !mapWithKey.test(fixed)) {
    // Try to add key using common patterns
    fixed = fixed.replace(
      /\.map\((\w+)\s*=>\s*(<[A-Za-z][a-zA-Z]*)([\s>])/g,
      (match, item, tag, after) => {
        // Don't add if already has key
        if (match.includes("key=")) return match;
        const keyVal = `${item}?.id ?? ${item}`;
        return `.map(${item} => ${tag} key={${keyVal}}${after}`;
      }
    );
    if (fixed !== code) { changed = true; fixes.push("added-map-keys"); }
  }

  // Fix: async function in useEffect (should use inner async)
  const asyncEffect = /useEffect\s*\(\s*async\s*\(/;
  if (asyncEffect.test(fixed)) {
    fixed = fixed.replace(
      /useEffect\s*\(\s*async\s*\(([^)]*)\)\s*=>\s*\{/g,
      "useEffect((() => { const run = async ($1) => {"
    );
    // This is a partial fix — add warning
    fixes.push("async-effect-partial-fix");
  }

  // Fix: missing semicolons on import/export lines (not critical but cleaner)
  // Skip — too risky to auto-add

  // Fix: `export default export default` (double export artifact)
  const doubleExport = /export\s+default\s+export\s+default/g;
  if (doubleExport.test(fixed)) {
    fixed = fixed.replace(doubleExport, "export default");
    changed = true;
    fixes.push("double-export-default");
  }

  // Fix: `export { default } from '...'` used as re-export inside component files
  // (usually fine, skip)

  // Fix: props spreading with wrong syntax <Component {...props,} />
  const badSpread = /\{\.\.\.(\w+),\s*\}/g;
  if (badSpread.test(fixed)) {
    fixed = fixed.replace(badSpread, "{...$1}");
    changed = true;
    fixes.push("spread-trailing-comma");
  }

  // Fix: `<>` without import (Fragment shorthand — fine in React 17+)
  // No fix needed

  // Fix: Common state setter naming issues
  // `const count, setCount = useState(0)` → `const [count, setCount] = useState(0)`
  const badDestructure = /const\s+(\w+)\s*,\s*(set\w+)\s*=\s*useState/g;
  if (badDestructure.test(fixed)) {
    fixed = fixed.replace(badDestructure, "const [$1, $2] = useState");
    changed = true;
    fixes.push("fixed-state-destructure");
  }

  return fixed;
}

// ================================================================
// TEMPLATE LITERALS
// ================================================================
function fixBrokenTemplateLiterals(code: string, fixes: string[]): string {
  let fixed = code;

  // Fix: `className={\`...\`}` with unclosed backtick
  // This is hard to auto-fix reliably — detect and warn

  // Fix: template literal used where string expected in JSX attribute
  // <div className=`foo bar`> → <div className="foo bar">
  const rawTemplateLiteral = /<[a-zA-Z][^>]*\s(\w+)=`([^`]*)`/g;
  if (rawTemplateLiteral.test(fixed)) {
    fixed = fixed.replace(rawTemplateLiteral, (m, attr, val) => {
      if (attr === "className" || attr === "style" || attr === "id") {
        return m.replace(`${attr}=\`${val}\``, `${attr}="${val}"`);
      }
      return m;
    });
    if (fixed !== code) fixes.push("template-literal-in-jsx-attr");
  }

  return fixed;
}

function fixConsoleStatements(code: string, fixes: string[]): string {

  // Just fix common artifact: stray `console.log()` on import lines
  const fixed = code.replace(/^import.*;\s*console\.log\(\)/gm, (m) =>

  );
  if (fixed !== code) fixes.push("removed-stray-console");

// ================================================================
// CSS FIXES
// ================================================================
function fixCSS(code: string, fixes: string[]): string {
  let fixed = code;
  let changed = false;

  // Fix: missing semicolons in CSS properties
  // property: value (no semicolon) → property: value;
  const missingSemicolon = /([a-zA-Z-]+\s*:\s*[^;{}]+)(\s*\n\s*[a-zA-Z-]+\s*:)/g;
  const withSemicolon = fixed.replace(missingSemicolon, "$1;$2");
  if (withSemicolon !== fixed) { changed = true; fixed = withSemicolon; }

  // Fix: vendor prefix ordering (put unprefixed last)
  // Too complex to auto-fix safely — skip

  // Fix: 0px → 0 (unnecessary unit)
  const zeroPx = fixed.replace(/:\s*0px\b/g, ": 0");
  if (zeroPx !== fixed) { changed = true; fixed = zeroPx; }

  if (changed) fixes.push("css-fixes");
  return fixed;
}

// ================================================================
// JSON FIXES
// ================================================================
function fixJSON(code: string, fixes: string[], warnings: string[]): string {
  let fixed = code.trim();

  // Remove trailing commas
  const noTrailing = fixed
    .replace(/,\s*}/g, "}")
    .replace(/,\s*]/g, "]");

  if (noTrailing !== fixed) {
    fixes.push("json-trailing-commas");
    fixed = noTrailing;
  }

  // Validate JSON
  try {
    JSON.parse(fixed);
  } catch (err) {
    warnings.push(`json-parse-error: ${String(err).slice(0, 100)}`);
  }

  return fixed;
}

// ================================================================
// HTML FIXES
// ================================================================
function fixHTML(code: string, fixes: string[]): string {
  let fixed = code;

  // Ensure UTF-8 charset meta if missing
  if (fixed.includes("<head") && !fixed.includes("charset")) {
    fixed = fixed.replace(
      /<head>/,
      '<head>\n  <meta charset="UTF-8" />'
    );
    fixes.push("added-charset-meta");
  }

  // Ensure viewport meta
  if (fixed.includes("<head") && !fixed.includes("viewport")) {
    fixed = fixed.replace(
      /<head>/,
      '<head>\n  <meta name="viewport" content="width=device-width, initial-scale=1.0" />'
    );
    fixes.push("added-viewport-meta");
  }

  return fixed;
}

// ================================================================
// CROSS-FILE FIXES
// Runs after all individual files are fixed.
// ================================================================
function crossFileFixes(files: AIFile[]): AIFile[] {
  // Ensure exactly one App.jsx/App.tsx exists
  const appFiles = files.filter(
    (f) => f.path === "src/App.jsx" || f.path === "src/App.tsx"
  );

  if (appFiles.length === 0) {
    // No App file — something went wrong, leave as-is
    return files;
  }

  if (appFiles.length > 1) {
    // Deduplicate: keep the tsx version, or the last one
    const keepPath = appFiles.find((f) => f.path.endsWith(".tsx"))?.path
      ?? appFiles[appFiles.length - 1].path;
    console.warn(`⚠️ Multiple App files found — keeping ${keepPath}`);
    return files.filter(
      (f) =>
        !(
          (f.path === "src/App.jsx" || f.path === "src/App.tsx") &&
          f.path !== keepPath
        )
    );
  }

  return files;
}

export type AIFile = {
  path: string;
  content: string;
};

export function mergeFiles(baseFiles: AIFile[], aiFiles: AIFile[]): AIFile[] {
  const fileMap = new Map<string, string>();

  // STEP 1: Load base template
  for (const file of baseFiles) {
    fileMap.set(file.path, file.content);
  }

  // STEP 2: Overlay AI files
  for (const file of aiFiles) {
    if (file.path === "package.json") {
      // Smart merge: keep base deps, add AI extra deps
      try {
        const basePkg = JSON.parse(fileMap.get("package.json") || "{}");
        const aiPkg = JSON.parse(file.content);

        const merged = {
          ...basePkg,
          ...aiPkg,
          // Always preserve name/version/type from base
          name: basePkg.name,
          version: basePkg.version,
          type: basePkg.type,
          // Merge scripts (base wins for dev/build)
          scripts: { ...aiPkg.scripts, ...basePkg.scripts },
          // AI deps extend base deps
          dependencies: { ...basePkg.dependencies, ...aiPkg.dependencies },
          devDependencies: { ...basePkg.devDependencies, ...aiPkg.devDependencies },
        };

        fileMap.set("package.json", JSON.stringify(merged, null, 2));
      } catch (err) {
        console.warn("⚠️ package.json merge failed, using base version");
        // Keep base package.json — safer than broken AI version
      }
      continue;
    }

    // For all other files: AI wins
    fileMap.set(file.path, file.content);
  }

  const result = Array.from(fileMap.entries()).map(([path, content]) => ({
    path,
    content,
  }));

  console.log(
    `🔀 Merged: ${baseFiles.length} base + ${aiFiles.length} AI = ${result.length} total files`
  );

  return result;
}

import fs from "fs";
import path from "path";

export type AIFile = {
  path: string;
  content: string;
};

export function loadBaseTemplate(): AIFile[] {
  const templateDir = path.join(
    process.cwd(),
    "sandbox-templates",
    "vite-react-base"
  );

  const files: AIFile[] = [];

  function readDir(dir: string, basePath: string) {
    const entries = fs.readdirSync(dir, { withFileTypes: true });

    for (const entry of entries) {
      const fullPath = path.join(dir, entry.name);
      const relativePath = path
        .join(basePath, entry.name)
        .replace(/\\/g, "/");

      if (entry.isDirectory()) {
        readDir(fullPath, relativePath);
      } else {
        const content = fs.readFileSync(fullPath, "utf-8");
        files.push({ path: relativePath, content });
      }
    }
  }

  try {
    readDir(templateDir, "");
    console.log(`📦 Loaded ${files.length} base template files`);
  } catch (err) {
    console.error("❌ Failed to load base template:", err);
  }

  return files;
}

"use client";

import { create } from "zustand";
import { AIFileNode } from "@/lib/ai/types/ai-project.types";

type Message = {
  role: "user" | "assistant";
  content: string;
};

interface AIBuilderState {
  projectId: string | null;
  files: AIFileNode[];
  activeFilePath: string | null;
  messages: Message[];

  setProject: (id: string) => void;
  setFiles: (files: AIFileNode[]) => void;
  setActiveFile: (path: string) => void;
  updateFile: (path: string, content: string) => void;

  addMessage: (msg: Message) => void;
  reset: () => void;
}

export const useAIBuilderStore = create<AIBuilderState>((set, get) => ({
  projectId: null,
  files: [],
  activeFilePath: null,
  messages: [],

  setProject: (id) =>
    set({
      projectId: id,
    }),

  setFiles: (files) =>
    set({
      files,
      activeFilePath: files.length ? files[0].path : null,
    }),

  setActiveFile: (path) =>
    set({
      activeFilePath: path,
    }),

  updateFile: (path, content) => {
    const { files } = get();

    const updated = files.map((file) =>
      file.path === path ? { ...file, content } : file
    );

    set({ files: updated });
  },

  addMessage: (msg) =>
    set((state) => ({
      messages: [...state.messages, msg],
    })),

  reset: () =>
    set({
      projectId: null,
      files: [],
      activeFilePath: null,
      messages: [],
    }),
}));
