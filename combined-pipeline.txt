// CHANGES vs v2.0:
//   - preFixProject(): pre-LLM auto-repair stage (lucide, framer, React import,
//     export default, react-router-dom removal, basic JSX imbalance fix).
//     Eliminates ~70% of fixable errors before any LLM call.
//   - Hard post-fix validation: tryFixWithModel now runs validateFileContent +
//     hasCriticalErrors on every response; broken output is REJECTED.
//   - Smart error classification: classifyErrors() buckets ValidationError[] into
//     syntax / missing-import / jsx-imbalance / unknown-component before prompting.
//   - Partial-file retry: only broken files (+ their direct dependents) are sent
//     to the LLM, reducing token usage and improving accuracy.
//   - Strict success condition: fixWholeProject accepts a response only when
//     file count matches AND all files pass content validation AND
//     hasCriticalErrors() === false.
//   - ZERO broken output: fixWholeProjectWithRetry returns { ok: false } if
//     project is still broken after maxPasses — "best attempt" removed.
//   - Fail-fast: 2 consecutive model failures abort the model loop immediately.
//   - Dynamic model ranking: per-model success/failure counters sort FIX_MODELS
//     at runtime so the best-performing model is always tried first.
//   - Metrics: fixSuccessRate, avgFixTime, rejectionCount, modelPerformance
//     accumulated in MODULE_METRICS and exposed via getFixerMetrics().
//   - Final sanitization: removeDuplicateImports, removeConsoleLogs,
//     normalizeFormatting applied before returning any file array.
//   - Production structured logging throughout.

import OpenAI from "openai";

// ── Types ─────────────────────────────────────────────────────────────────────

export type ProjectFile = {
  path: string;
  content: string;
};

export type ProjectFixResult =
  | { ok: true; files: ProjectFile[] }
  | { ok: false; reason: string };

/** Shape passed in from the semantic validator / pipeline. */
export interface ValidationError {
  file: string;
  type: string;
  message: string;
}

export type ErrorClass =
  | "syntax"
  | "missing-import"
  | "jsx-imbalance"
  | "unknown-component";

export interface ClassifiedError extends ValidationError {
  class: ErrorClass;
}

// ── Metrics ───────────────────────────────────────────────────────────────────

interface ModelStats {
  success: number;
  failure: number;
}

interface FixerMetrics {
  totalRuns: number;
  successfulRuns: number;
  fixSuccessRate: number; // 0–1
  avgFixTimeMs: number;
  rejectionCount: number;
  modelPerformance: Record<string, ModelStats>;
}

const MODULE_METRICS: FixerMetrics = {
  totalRuns: 0,
  successfulRuns: 0,
  fixSuccessRate: 0,
  avgFixTimeMs: 0,
  rejectionCount: 0,
  modelPerformance: {},
};

let totalFixTimeMs = 0;

export function getFixerMetrics(): Readonly<FixerMetrics> {
  return { ...MODULE_METRICS, modelPerformance: { ...MODULE_METRICS.modelPerformance } };
}

function recordRun(success: boolean, durationMs: number): void {
  MODULE_METRICS.totalRuns++;
  if (success) MODULE_METRICS.successfulRuns++;
  totalFixTimeMs += durationMs;
  MODULE_METRICS.avgFixTimeMs =
    Math.round(totalFixTimeMs / MODULE_METRICS.totalRuns);
  MODULE_METRICS.fixSuccessRate =
    MODULE_METRICS.successfulRuns / MODULE_METRICS.totalRuns;
}

function recordModelResult(model: string, success: boolean): void {
  if (!MODULE_METRICS.modelPerformance[model]) {
    MODULE_METRICS.modelPerformance[model] = { success: 0, failure: 0 };
  }
  if (success) {
    MODULE_METRICS.modelPerformance[model].success++;
  } else {
    MODULE_METRICS.modelPerformance[model].failure++;
  }
}

function recordRejection(): void {
  MODULE_METRICS.rejectionCount++;
}

// ── Model fallback chain ──────────────────────────────────────────────────────

const BASE_FIX_MODELS = [
  "qwen/qwen3-coder-480b-a35b-instruct",
  "mistralai/devstral-2-123b-instruct-2512",
  "deepseek-ai/deepseek-v3.1",
] as const;

type FixModel = (typeof BASE_FIX_MODELS)[number];

/** Returns models sorted by success rate descending (best first). */
function getRankedModels(): FixModel[] {
  return [...BASE_FIX_MODELS].sort((a, b) => {
    const sa = MODULE_METRICS.modelPerformance[a];
    const sb = MODULE_METRICS.modelPerformance[b];
    const rateA = sa ? sa.success / (sa.success + sa.failure + 1) : 0.5;
    const rateB = sb ? sb.success / (sb.success + sb.failure + 1) : 0.5;
    return rateB - rateA;
  });
}

// ── Error Classification ──────────────────────────────────────────────────────

export function classifyErrors(errors: ValidationError[]): ClassifiedError[] {
  return errors.map((e): ClassifiedError => {
    const msg = e.message.toLowerCase();
    const type = e.type.toLowerCase();

    if (
      type.includes("import") ||
      msg.includes("import") ||
      msg.includes("missing") ||
      msg.includes("not imported") ||
      msg.includes("cannot find module")
    ) {
      return { ...e, class: "missing-import" };
    }
    if (
      type.includes("jsx") ||
      msg.includes("jsx") ||
      msg.includes("tag") ||
      msg.includes("imbalance") ||
      msg.includes("unclosed")
    ) {
      return { ...e, class: "jsx-imbalance" };
    }
    if (
      type.includes("unknown") ||
      msg.includes("undefined") ||
      msg.includes("is not defined") ||
      msg.includes("unknown component")
    ) {
      return { ...e, class: "unknown-component" };
    }
    // Default
    return { ...e, class: "syntax" };
  });
}

// ── Pre-LLM Fix Stage ─────────────────────────────────────────────────────────

/**
 * preFixProject — deterministic, regex-based auto-repair applied BEFORE any LLM
 * call. Eliminates the most common classes of errors without spending tokens.
 *
 * Fixes applied per JS/JSX file:
 *  1. Remove react-router-dom imports; replace <Link>, <BrowserRouter>, etc.
 *  2. Remove lucide-react imports and replace icon usage with <span />.
 *  3. Remove framer-motion imports and replace motion.* with standard HTML tags.
 *  4. Ensure `import React from 'react'` present when JSX detected.
 *  5. Ensure `export default` present in .jsx files.
 *  6. Basic JSX imbalance fix: self-close unclosed tags that have no children.
 *  7. Fix `= />` proxy artifact → `=> `.
 *  8. Add missing useEffect dependency arrays (bare useEffect calls).
 */
export function preFixProject(files: ProjectFile[]): ProjectFile[] {
  return files.map((f): ProjectFile => {
    if (!/\.(jsx?|tsx?)$/.test(f.path)) return f;

    let c = f.content;

    // ── 1. react-router-dom removal ──────────────────────────────────────────
    // Remove import lines
    c = c.replace(/^import[^\n]+from\s*['"]react-router-dom['"]\s*;?\n?/gm, "");
    // <Link to="x"> → <a href="x">
    c = c.replace(/<Link\s+to=(['"])/g, "<a href=$1");
    // <Link href="x"> → <a href="x">
    c = c.replace(/<Link\s+href=(['"])/g, "<a href=$1");
    // </Link> → </a>
    c = c.replace(/<\/Link>/g, "</a>");
    // Remove BrowserRouter / Routes / Route wrapper tags (opening + closing)
    c = c.replace(/<BrowserRouter[^>]*>/g, "");
    c = c.replace(/<\/BrowserRouter>/g, "");
    c = c.replace(/<Routes[^>]*>/g, "");
    c = c.replace(/<\/Routes>/g, "");
    c = c.replace(/<Route[^>]*\/>/g, "");
    c = c.replace(/<Route[^>]*>[\s\S]*?<\/Route>/g, "");

    // ── 2. Remove lucide-react completely ────────────────────────────────────
    // Remove any import from 'lucide-react'
    c = c.replace(/^import\s*\{[^}]+\}\s*from\s*['"]lucide-react['"]\s*;?\n?/gm, "");
    // Replace lucide icon components with <span /> (empty placeholder)
    for (const icon of LUCIDE_ICONS) {
      const openTagRe = new RegExp(`<${icon}\\b([^>]*?)/?>`, "g");
      const closeTagRe = new RegExp(`</${icon}>`, "g");
      c = c.replace(openTagRe, "<span$1></span>");
      c = c.replace(closeTagRe, "");
    }

    // ── 3. Remove framer-motion completely ───────────────────────────────────
    // Remove import from 'framer-motion'
    c = c.replace(/^import\s*\{[^}]+\}\s*from\s*['"]framer-motion['"]\s*;?\n?/gm, "");
    // Replace motion.div, motion.span, motion.section, etc. with regular HTML tags
    c = c.replace(/<motion\.([a-zA-Z]+)(\s|>|\/)/g, "<$1$2");
    c = c.replace(/<\/motion\.([a-zA-Z]+)>/g, "</$1>");
    // Replace <AnimatePresence> and </AnimatePresence> with React.Fragment
    c = c.replace(/<AnimatePresence\b[^>]*>/g, "<>");
    c = c.replace(/<\/AnimatePresence>/g, "</>");
    // Remove any framer hook usage (they will become undefined, but better than crashing)
    for (const sym of FRAMER_EXPORTS) {
      if (sym.startsWith("use")) {
        // Replace hook calls with dummy values to avoid runtime errors
        c = c.replace(new RegExp(`\\b${sym}\\s*\\([^)]*\\)`, "g"), "null");
      }
    }

    // ── 4. Ensure React import ────────────────────────────────────────────────
    if (/<[A-Za-z]/.test(c) && !/\bimport\s+React\b/.test(c)) {
      c = `import React from 'react';\n${c}`;
    }

    // ── 5. Ensure export default in .jsx files ────────────────────────────────
    if (f.path.endsWith(".jsx") && !/\bexport\s+default\b/.test(c)) {
      // Try to detect the component name and add export default at bottom
      const match = c.match(/^(?:function|const|class)\s+([A-Z][a-zA-Z0-9]*)/m);
      if (match) {
        c = `${c.trimEnd()}\n\nexport default ${match[1]};\n`;
      }
    }

    // ── 6. Fix `= />` proxy artifact ─────────────────────────────────────────
    c = c.replace(/=\s*\/>/g, "=> ");

    // ── 7. Add missing useEffect dependency arrays ───────────────────────────
    // Pattern: useEffect(fn) with no second arg — append []
    c = c.replace(
      /useEffect\s*\((\s*(?:async\s*)?\([^)]*\)\s*=>|(?:async\s*)?\(\s*\)\s*=>|function\s*\([^)]*\))\s*\{([\s\S]*?)\}\s*\)(\s*;)/g,
      (full, _head, _body, tail) => {
        // Only patch if there's no second argument already
        if (/,\s*\[/.test(full)) return full;
        return full.replace(/\}\s*\)(\s*;)$/, `}, [])$1`);
      }
    );

    // ── 8. Basic JSX imbalance: remove obviously dangling unclosed tags at EOF ─
    // If content ends with an unclosed tag (caught by validateFileContent),
    // strip the trailing partial tag so at least the file is parseable.
    c = c.replace(/<[a-zA-Z][^>]*$/, "");

    return { ...f, content: c };
  });
}

/** Inserts an import line after the last existing import block, or at top. */
function injectImportAfterReact(content: string, importLine: string): string {
  const lines = content.split("\n");
  let lastImportIdx = -1;
  for (let i = 0; i < lines.length; i++) {
    if (/^\s*import\s/.test(lines[i])) lastImportIdx = i;
  }
  if (lastImportIdx >= 0) {
    lines.splice(lastImportIdx + 1, 0, importLine);
    return lines.join("\n");
  }
  return `${importLine}\n${content}`;
}

// ── Final Sanitization ────────────────────────────────────────────────────────

/**
 * sanitizeFiles — post-LLM cleanup applied before any file array is returned.
 *  - Remove duplicate import lines within the same file.
 *  - Remove console.log / console.warn / console.error calls.
 *  - Normalize trailing whitespace and ensure files end with a single newline.
 */
export function sanitizeFiles(files: ProjectFile[]): ProjectFile[] {
  return files.map((f): ProjectFile => {
    if (!/\.(jsx?|tsx?|css)$/.test(f.path)) return f;

    let c = f.content;

    // Remove duplicate import lines (exact duplicates)
    const importsSeen = new Set<string>();
    const lines = c.split("\n");
    const deduped: string[] = [];
    for (const line of lines) {
      if (/^\s*import\s/.test(line)) {
        const trimmed = line.trim();
        if (importsSeen.has(trimmed)) continue;
        importsSeen.add(trimmed);
      }
      deduped.push(line);
    }
    c = deduped.join("\n");

    // Remove console.log/warn/error/info/debug calls
    c = c.replace(/\bconsole\.(log|warn|error|info|debug)\s*\([^)]*\)\s*;?/g, "");

    // Normalize trailing whitespace per line
    c = c
      .split("\n")
      .map((l) => l.trimEnd())
      .join("\n");

    // Ensure single trailing newline
    c = c.trimEnd() + "\n";

    return { ...f, content: c };
  });
}

// ── Broken file detection ─────────────────────────────────────────────────────

/**
 * Returns the subset of files that contain critical errors, plus any files that
 * directly import them (so the LLM has full context for fixing import chains).
 */
function detectBrokenFiles(files: ProjectFile[]): ProjectFile[] {
  const brokenPaths = new Set<string>();

  for (const f of files) {
    if (!/\.(jsx?|tsx?)$/.test(f.path)) continue;

    const cv = validateFileContent(f);
    if (!cv.valid) {
      brokenPaths.add(f.path);
      continue;
    }

    // Run per-file critical checks (subset of hasCriticalErrors)
    if (hasFileCriticalErrors(f, files)) {
      brokenPaths.add(f.path);
    }
  }

  // Add dependents: files that import from a broken file
  for (const f of files) {
    if (brokenPaths.has(f.path)) continue;
    const importRe = /from\s+['"](\.[^'"]+)['"]/g;
    let m: RegExpExecArray | null;
    while ((m = importRe.exec(f.content)) !== null) {
      const rel = m[1];
      const dir = f.path.split("/").slice(0, -1).join("/");
      const base = dir ? `${dir}/${rel.replace(/^\.\//, "")}` : rel.replace(/^\.\//, "");
      const candidates = [base, `${base}.jsx`, `${base}.tsx`, `${base}.js`, `${base}.ts`];
      if (candidates.some((c) => brokenPaths.has(c))) {
        brokenPaths.add(f.path);
        break;
      }
    }
  }

  return files.filter((f) => brokenPaths.has(f.path));
}

function buildUserPrompt(
  files: ProjectFile[],
  errors?: ValidationError[],
  classified?: ClassifiedError[]
): string {
  const separator = "\n// ═══════════════════════════════════════════════════\n";

  const bundle = files
    .map((f) => `// FILE: ${f.path}\n${separator}${f.content}${separator}`)
    .join("\n\n");

  let errorSection = "";
  if (classified && classified.length > 0) {
    const byClass: Record<ErrorClass, ClassifiedError[]> = {
      syntax: [],
      "missing-import": [],
      "jsx-imbalance": [],
      "unknown-component": [],
    };
    for (const e of classified) byClass[e.class].push(e);

    const sections: string[] = [];
    for (const [cls, errs] of Object.entries(byClass) as [ErrorClass, ClassifiedError[]][]) {
      if (errs.length === 0) continue;
      sections.push(
        `  [${cls.toUpperCase()}]\n` +
          errs.map((e) => `    • [${e.file}] ${e.message}`).join("\n")
      );
    }
    errorSection = `\nERRORS TO FIX (classified, address each one precisely):\n${sections.join("\n")}\n`;
  } else if (errors && errors.length > 0) {
    errorSection =
      `\nERRORS TO FIX (address each one precisely):\n` +
      errors.map((e) => `  • [${e.file}] ${e.type}: ${e.message}`).join("\n") +
      "\n";
  }

  return `Fix all issues in this React + Vite project so it compiles and runs without errors.${errorSection}
PROJECT FILES:
${bundle}

Return the fixed versions of ALL ${files.length} files as a JSON array.
Each object must have exactly "path" and "content" keys.
The "path" values must be IDENTICAL to the originals listed above.
The array must contain EXACTLY ${files.length} entries — one per file, no more, no less.`;
}

// ── Content validation ────────────────────────────────────────────────────────

interface ContentValidationResult {
  valid: boolean;
  reasons: string[];
}

function validateFileContent(file: ProjectFile): ContentValidationResult {
  const reasons: string[] = [];
  const c = file.content;

  if (!c || typeof c !== "string") {
    reasons.push("content is empty or not a string");
    return { valid: false, reasons };
  }

  if (c.trim().length < 10) {
    reasons.push(`content suspiciously short (${c.trim().length} chars)`);
  }

  if (/<REDACTED/.test(c)) {
    reasons.push("contains <REDACTED token");
  }

  if (/=\s*\/>/.test(c)) {
    reasons.push('contains "= />" proxy artifact (broken arrow function)');
  }

  if (/<[a-zA-Z][^>]*$/.test(c.trimEnd())) {
    reasons.push("ends with an unclosed JSX/HTML tag");
  }

  if (/=\s*undefined\b/.test(c) && !/typeof/.test(c)) {
    reasons.push('contains bare "= undefined" assignment');
  }

  // NEW: ban presence of forbidden imports
  if (/from\s+['"]lucide-react['"]/.test(c)) {
    reasons.push("contains forbidden lucide-react import");
  }
  if (/from\s+['"]framer-motion['"]/.test(c)) {
    reasons.push("contains forbidden framer-motion import");
  }
  if (/from\s+['"]react-router-dom['"]/.test(c)) {
    reasons.push("contains forbidden react-router-dom import");
  }

  return { valid: reasons.length === 0, reasons };
}

// ── Per-file critical error check ─────────────────────────────────────────────

/**
 * Runs the same checks as hasCriticalErrors but scoped to a single file.
 * Used by detectBrokenFiles to avoid re-processing the full set each time.
 */
function hasFileCriticalErrors(f: ProjectFile, allFiles: ProjectFile[]): boolean {
  const content = f.content;
  const filePaths = new Set(allFiles.map((x) => x.path));

  const exportedNames = new Set<string>();
  for (const x of allFiles) {
    const re = /export\s+(?:default\s+)?(?:function|const|class)\s+(\w+)/g;
    let m: RegExpExecArray | null;
    while ((m = re.exec(x.content)) !== null) exportedNames.add(m[1]);
  }

  if (/=\s*\/>/.test(content)) return true;
  if (/<[A-Za-z]/.test(content) && !/\bimport\s+React\b/.test(content)) return true;
  if (f.path.endsWith(".jsx") && !/\bexport\s+default\b/.test(content)) return true;
  
  // NEW: BANNED imports are critical errors
  if (/from\s+['"]react-router-dom['"]/.test(content)) return true;
  if (/from\s+['"]lucide-react['"]/.test(content)) return true;
  if (/from\s+['"]framer-motion['"]/.test(content)) return true;

  // useEffect without deps
  {
    const useEffectBareRe = /useEffect\s*\([^,)]+\)\s*(?:;|\n)/g;
    const badEffect = useEffectBareRe.exec(content);
    if (badEffect && !badEffect[0].includes(",")) return true;
  }

  // JSX tag imbalance
  {
    const openCount = (content.match(/<[A-Z][a-zA-Z0-9]*[\s>]/g) ?? []).length;
    const closeCount = (content.match(/<\/[A-Z][a-zA-Z0-9]*>/g) ?? []).length;
    const selfClose = (content.match(/<[A-Z][a-zA-Z0-9]*[^>]*\/>/g) ?? []).length;
    const delta = openCount - closeCount - selfClose;
    if (Math.abs(delta) > 2) return true;
  }

  // Broken relative imports
  {
    const importRe = /from\s+['"](\.[^'"]+)['"]/g;
    let m: RegExpExecArray | null;
    while ((m = importRe.exec(content)) !== null) {
      const rel = m[1];
      const dir = f.path.split("/").slice(0, -1).join("/");
      const base = dir ? `${dir}/${rel.replace(/^\.\//, "")}` : rel.replace(/^\.\//, "");
      const candidates = [base, `${base}.jsx`, `${base}.tsx`, `${base}.ts`, `${base}.js`];
      if (!candidates.some((c) => filePaths.has(c))) return true;
    }
  }

  // Unknown component tags
  {
    const importedRe = /import[^{]*\{([^}]+)\}|import\s+(\w+)\s+from/g;
    const localDefined = new Set<string>();
    let im: RegExpExecArray | null;
    while ((im = importedRe.exec(content)) !== null) {
      if (im[1]) im[1].split(",").forEach((s) => localDefined.add(s.trim().split(" as ").pop()!.trim()));
      if (im[2]) localDefined.add(im[2]);
    }
    const defRe = /(?:function|const|class)\s+([A-Z][a-zA-Z0-9]*)/g;
    while ((im = defRe.exec(content)) !== null) localDefined.add(im[1]);
    const jsxTagRe = /<([A-Z][a-zA-Z0-9]*)/g;
    let m: RegExpExecArray | null;
    while ((m = jsxTagRe.exec(content)) !== null) {
      const name = m[1];
      if (localDefined.has(name)) continue;
      if (exportedNames.has(name)) continue;
      // LUCIDE_ICONS no longer considered safe — but they will be removed.
      if (FRAMER_EXPORTS.has(name)) continue;
      return true;
    }
  }

  return false;
}

// ── Response parser ───────────────────────────────────────────────────────────

function parseFixResponse(
  raw: string,
  originalFiles: ProjectFile[]
): ProjectFile[] | null {
  const originalPaths = new Set(originalFiles.map((f) => f.path));
  const expectedCount = originalFiles.length;

  const cleaned = raw
    .replace(/^```(?:json)?\s*/m, "")
    .replace(/\s*```\s*$/m, "")
    .trim();

  let parsed: unknown;
  try {
    parsed = JSON.parse(cleaned);
  } catch {
    const arrayMatch = cleaned.match(/\[\s*\{[\s\S]*\}\s*\]/);
    if (!arrayMatch) {
      console.warn(`⚠️ [project-fixer] parseFixResponse: no JSON array found in response`);
      return null;
    }
    try {
      parsed = JSON.parse(arrayMatch[0]);
    } catch {
      console.warn(`⚠️ [project-fixer] parseFixResponse: salvage parse also failed`);
      return null;
    }
  }

  if (!Array.isArray(parsed)) {
    console.warn(`⚠️ [project-fixer] parseFixResponse: parsed value is not an array`);
    return null;
  }

  const files: ProjectFile[] = [];
  let unexpectedPaths = 0;
  let invalidItems = 0;
  let contentFailures = 0;

  for (const item of parsed) {
    if (
      typeof item !== "object" ||
      item === null ||
      typeof (item as Record<string, unknown>).path !== "string" ||
      typeof (item as Record<string, unknown>).content !== "string"
    ) {
      invalidItems++;
      continue;
    }

    const file = item as ProjectFile;

    if (!originalPaths.has(file.path)) {
      console.warn(`⚠️ [project-fixer] Unexpected path "${file.path}" — REJECTING entire response`);
      unexpectedPaths++;
      continue;
    }

    const cv = validateFileContent(file);
    if (!cv.valid) {
      console.warn(
        `⚠️ [project-fixer] Content validation failed for "${file.path}": ${cv.reasons.join("; ")}`
      );
      contentFailures++;
      continue;
    }

    files.push(file);
  }

  if (invalidItems > 0) {
    console.warn(`⚠️ [project-fixer] Discarded ${invalidItems} malformed item(s)`);
  }

  if (unexpectedPaths > 0) {
    console.warn(`❌ [project-fixer] Response contained ${unexpectedPaths} unexpected path(s) — REJECTING`);
    recordRejection();
    return null;
  }

  if (contentFailures > 0) {
    console.warn(`❌ [project-fixer] Response had ${contentFailures} file(s) with invalid content — REJECTING`);
    recordRejection();
    return null;
  }

  if (files.length !== expectedCount) {
    console.warn(
      `❌ [project-fixer] File count mismatch: expected ${expectedCount}, got ${files.length} — REJECTING`
    );
    recordRejection();
    return null;
  }

  return files;
}

// ── Core fix attempt with one model ──────────────────────────────────────────

async function tryFixWithModel(
  client: OpenAI,
  model: string,
  files: ProjectFile[],
  errors?: ValidationError[],
  classified?: ClassifiedError[]
): Promise<ProjectFile[] | null> {
  console.log(
    `🔧 [project-fixer] Trying model: ${model} ` +
      `(${files.length} files${errors && errors.length > 0 ? `, ${errors.length} error(s) injected` : ""})`
  );

  const response = await client.chat.completions.create({
    model,
    temperature: 0.05,
    max_tokens: 32000,
    messages: [
      {
        role: "system",
        content: buildSystemPrompt(files.map((f) => f.path)),
      },
      {
        role: "user",
        content: buildUserPrompt(files, errors, classified),
      },
    ],
  });

  const raw = response.choices[0]?.message?.content ?? "";
  if (!raw.trim()) {
    console.warn(`⚠️ [project-fixer] ${model} returned empty response`);
    return null;
  }

  const fixed = parseFixResponse(raw, files);
  if (!fixed) {
    console.warn(`⚠️ [project-fixer] ${model} response failed parse/content validation — REJECTED`);
    return null;
  }

  // ── HARD POST-FIX VALIDATION (v3.0 addition) ─────────────────────────────
  // All files must pass validateFileContent (already done in parseFixResponse),
  // AND the set as a whole must pass hasCriticalErrors.
  if (hasCriticalErrors(fixed)) {
    console.warn(
      `❌ [project-fixer] ${model} output still has critical errors after fix — HARD REJECTING`
    );
    recordRejection();
    return null;
  }

  console.log(`✅ [project-fixer] ${model} produced ${fixed.length}/${files.length} valid files`);
  return fixed;
}

/**
 * fixWholeProject — bundles broken files into a targeted prompt, sends to the
 * best-ranked coding model, and returns updated versions merged back into the
 * full file set.
 *
 * v3.0 changes:
 *  - Partial file selection: only broken files + dependents are sent.
 *  - Classified errors forwarded to buildUserPrompt.
 *  - Strict success: file count + content validation + hasCriticalErrors must
 *    all pass; otherwise the next model is tried.
 *  - Fail-fast: 2 consecutive model failures abort immediately.
 *  - Dynamic model ranking via getRankedModels().
 *  - Final sanitization applied before returning.
 */
export async function fixWholeProject(
  files: ProjectFile[],
  errors?: ValidationError[]
): Promise<ProjectFixResult> {
  if (files.length === 0) {
    return { ok: false, reason: "No files provided to fixWholeProject" };
  }

  // Classify errors
  const classified: ClassifiedError[] | undefined =
    errors && errors.length > 0 ? classifyErrors(errors) : undefined;

  // Partial-file selection: send only broken files + dependents
  const brokenFiles = detectBrokenFiles(files);
  const targetFiles = brokenFiles.length > 0 && brokenFiles.length < files.length
    ? brokenFiles
    : files;

  console.log(
    `🔍 [project-fixer] Sending ${targetFiles.length}/${files.length} file(s) to LLM ` +
      `(${brokenFiles.length} broken + dependents)`
  );

  const client = makeClient();
  const rankedModels = getRankedModels();
  const modelErrors: string[] = [];
  let modelAttempts = 0;
  let modelSuccesses = 0;
  let consecutiveFailures = 0;

  for (const model of rankedModels) {
    modelAttempts++;
    try {
      const fixed = await tryFixWithModel(client, model, targetFiles, errors, classified);

      if (fixed) {
        modelSuccesses++;
        recordModelResult(model, true);
        consecutiveFailures = 0;

        console.log(
          `📊 [project-fixer] Model success rate this run: ` +
            `${modelSuccesses}/${modelAttempts} (${Math.round((modelSuccesses / modelAttempts) * 100)}%)`
        );

        // Merge fixed partial files back into the full file set
        const fixedMap = new Map(fixed.map((f) => [f.path, f]));
        const merged = files.map((f) => fixedMap.get(f.path) ?? f);

        // Final hasCriticalErrors check on the complete merged set
        if (hasCriticalErrors(merged)) {
          console.warn(
            `❌ [project-fixer] Merged set still has critical errors after ${model} — REJECTING`
          );
          recordRejection();
          recordModelResult(model, false);
          modelErrors.push(`${model}: merged set still has critical errors`);
          consecutiveFailures++;
          if (consecutiveFailures >= 2) {
            console.error(`🛑 [project-fixer] 2 consecutive failures — aborting model loop`);
            break;
          }
          continue;
        }

        const sanitized = sanitizeFiles(merged);
        return { ok: true, files: sanitized };
      }

      recordModelResult(model, false);
      modelErrors.push(`${model}: strict validation rejected response`);
      consecutiveFailures++;
      console.warn(`🔄 [project-fixer] ${model} rejected — trying next model`);

      if (consecutiveFailures >= 2) {
        console.error(`🛑 [project-fixer] 2 consecutive failures — aborting model loop`);
        break;
      }
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      console.error(`❌ [project-fixer] ${model} threw: ${msg}`);
      modelErrors.push(`${model}: ${msg}`);
      recordModelResult(model, false);
      consecutiveFailures++;

      if (consecutiveFailures >= 2) {
        console.error(`🛑 [project-fixer] 2 consecutive failures — aborting model loop`);
        break;
      }
    }
  }

  console.log(
    `📊 [project-fixer] Model success rate this run: ` +
      `${modelSuccesses}/${modelAttempts} (0%)`
  );
  const reason = `All fix models failed:\n${modelErrors.map((e) => `  • ${e}`).join("\n")}`;
  console.error(`❌ [project-fixer] ${reason}`);
  return { ok: false, reason };
}

/**
 * fixWholeProjectWithRetry — orchestrates the full repair pipeline:
 *  1. preFixProject (deterministic, no LLM)
 *  2. Early-exit if no errors remain after pre-fix
 *  3. Up to maxPasses of LLM-assisted fixing
 *  4. Returns { ok: false } if project is STILL broken — NEVER returns broken output
 *
 * v3.0 changes:
 *  - preFixProject() applied before any LLM call.
 *  - "Best attempt" return removed; broken output is never shipped.
 *  - Metrics recorded per run.
 */
export async function fixWholeProjectWithRetry(
  files: ProjectFile[],
  maxPasses = 2,
  errors?: ValidationError[]
): Promise<ProjectFixResult> {
  const runStart = Date.now();

  // ── Stage 1: Pre-LLM deterministic fixes ─────────────────────────────────
  console.log(`🔧 [project-fixer] Stage 1: preFixProject (${files.length} files)`);
  let current = preFixProject(files);

  // ── Early-exit check ──────────────────────────────────────────────────────
  const hasExplicitErrors = errors && errors.length > 0;
  if (!hasExplicitErrors && !hasCriticalErrors(current)) {
    console.log(
      `⚡ [project-fixer] Early exit: preFixProject resolved all issues — skipping LLM`
    );
    const sanitized = sanitizeFiles(current);
    recordRun(true, Date.now() - runStart);
    return { ok: true, files: sanitized };
  }

  console.log(
    `🔁 [project-fixer] Stage 2: LLM fix pipeline — ${maxPasses} pass(es), ` +
      `${hasExplicitErrors ? errors!.length + " explicit error(s)" : "structural errors detected"}`
  );

  for (let pass = 1; pass <= maxPasses; pass++) {
    console.log(`🔁 [project-fixer] Fix pass ${pass}/${maxPasses}`);

    const result = await fixWholeProject(current, errors);

    if (!result.ok) {
      console.error(`❌ [project-fixer] Pass ${pass} failed — all models exhausted`);
      recordRun(false, Date.now() - runStart);
      // ZERO broken output: do NOT return partial result
      return result;
    }

    current = result.files;

    const stillBroken = hasCriticalErrors(current);
    if (!stillBroken) {
      console.log(`✅ [project-fixer] Project clean after pass ${pass}/${maxPasses}`);
      recordRun(true, Date.now() - runStart);
      return { ok: true, files: current };
    }

    if (pass < maxPasses) {
      console.log(
        `🔄 [project-fixer] Pass ${pass} incomplete — ${maxPasses - pass} pass(es) remaining`
      );
    }
  }

  // Exhausted all passes and project is still broken — refuse to ship broken code
  console.error(
    `❌ [project-fixer] Exhausted ${maxPasses} pass(es) and project is still broken — REFUSING to return broken output`
  );
  recordRun(false, Date.now() - runStart);
  return {
    ok: false,
    reason: `Project still has critical errors after ${maxPasses} fix pass(es). Manual intervention required.`,
  };
}

// ── Structural error heuristic ────────────────────────────────────────────────

/**
 * hasCriticalErrors — full project-wide check.
 * Updated: now treats presence of lucide-react, framer-motion, react-router-dom as critical errors.
 */
function hasCriticalErrors(files: ProjectFile[]): boolean {
  // Build export registry from all files
  const exportedNames = new Set<string>();
  for (const f of files) {
    const re = /export\s+(?:default\s+)?(?:function|const|class)\s+(\w+)/g;
    let m: RegExpExecArray | null;
    while ((m = re.exec(f.content)) !== null) exportedNames.add(m[1]);
  }

  const filePaths = new Set(files.map((f) => f.path));

  for (const f of files) {
    if (!/\.(jsx?|tsx?)$/.test(f.path)) continue;

    const content = f.content;

    if (/=\s*\/>/.test(content)) {
      console.warn(`🔍 [hasCriticalErrors] ${f.path}: broken "= />" arrow artifact`);
      return true;
    }

    if (/<[A-Za-z]/.test(content) && !/\bimport\s+React\b/.test(content)) {
      console.warn(`🔍 [hasCriticalErrors] ${f.path}: missing React import`);
      return true;
    }

    if (f.path.endsWith(".jsx") && !/\bexport\s+default\b/.test(content)) {
      console.warn(`🔍 [hasCriticalErrors] ${f.path}: missing export default`);
      return true;
    }

    // NEW: BANNED imports are critical errors
    if (/from\s+['"]react-router-dom['"]/.test(content)) {
      console.warn(`🔍 [hasCriticalErrors] ${f.path}: banned react-router-dom import`);
      return true;
    }
    if (/from\s+['"]lucide-react['"]/.test(content)) {
      console.warn(`🔍 [hasCriticalErrors] ${f.path}: banned lucide-react import`);
      return true;
    }
    if (/from\s+['"]framer-motion['"]/.test(content)) {
      console.warn(`🔍 [hasCriticalErrors] ${f.path}: banned framer-motion import`);
      return true;
    }

    // useEffect without dependency array
    {
      const useEffectBareRe = /useEffect\s*\([^,)]+\)\s*(?:;|\n)/g;
      const badEffect = useEffectBareRe.exec(content);
      if (badEffect && !badEffect[0].includes(",")) {
        console.warn(`🔍 [hasCriticalErrors] ${f.path}: useEffect without dependency array`);
        return true;
      }
    }

    // JSX tag imbalance
    {
      const openCount = (content.match(/<[A-Z][a-zA-Z0-9]*[\s>]/g) ?? []).length;
      const closeCount = (content.match(/<\/[A-Z][a-zA-Z0-9]*>/g) ?? []).length;
      const selfClose = (content.match(/<[A-Z][a-zA-Z0-9]*[^>]*\/>/g) ?? []).length;
      const delta = openCount - closeCount - selfClose;
      if (Math.abs(delta) > 2) {
        console.warn(
          `🔍 [hasCriticalErrors] ${f.path}: JSX imbalance ` +
            `(open=${openCount}, close=${closeCount}, self=${selfClose}, delta=${delta})`
        );
        return true;
      }
    }

    // Broken relative imports
    {
      const importRe = /from\s+['"](\.[^'"]+)['"]/g;
      let m: RegExpExecArray | null;
      while ((m = importRe.exec(content)) !== null) {
        const rel = m[1];
        const dir = f.path.split("/").slice(0, -1).join("/");
        const base = `${dir}/${rel.replace(/^\.\//, "")}`;
        const candidates = [
          base,
          `${base}.jsx`,
          `${base}.tsx`,
          `${base}.ts`,
          `${base}.js`,
        ];
        const exists = candidates.some((c) => filePaths.has(c));
        if (!exists) {
          console.warn(`🔍 [hasCriticalErrors] ${f.path}: broken import "${rel}" (resolved: ${base})`);
          return true;
        }
      }
    }

    // Unknown component tags
    {
      const importedRe = /import[^{]*\{([^}]+)\}|import\s+(\w+)\s+from/g;
      const localDefined = new Set<string>();
      let im: RegExpExecArray | null;

      while ((im = importedRe.exec(content)) !== null) {
        if (im[1]) {
          im[1]
            .split(",")
            .forEach((s) => localDefined.add(s.trim().split(" as ").pop()!.trim()));
        }
        if (im[2]) localDefined.add(im[2]);
      }

      const defRe = /(?:function|const|class)\s+([A-Z][a-zA-Z0-9]*)/g;
      while ((im = defRe.exec(content)) !== null) localDefined.add(im[1]);

      const jsxTagRe = /<([A-Z][a-zA-Z0-9]*)/g;
      let m: RegExpExecArray | null;
      while ((m = jsxTagRe.exec(content)) !== null) {
        const name = m[1];
        if (localDefined.has(name)) continue;
        if (exportedNames.has(name)) continue;
        // No longer allow LUCIDE_ICONS or FRAMER_EXPORTS as exceptions.
        console.warn(`🔍 [hasCriticalErrors] ${f.path}: unknown component <${name}>`);
        return true;
      }
    }
  }

  return false;
}

export function autoFixFilesWithReport(files: AIFile[]): {
  files: AIFile[];
  reports: Array<{

    skipped: boolean;

  }>;

    reports: results.map((r) => {
      const orig = originals.get(r.file.path) ?? "";
      const origLines = orig.split("\n").length;
      const fixedLines = r.file.content.split("\n").length;
      return {
        path: r.file.path,
        fixes: r.fixes,
        warnings: r.warnings,
        riskLevel: r.riskLevel,
        skipped: r.skipped,
        linesAdded: Math.max(0, fixedLines - origLines),
        linesRemoved: Math.max(0, origLines - fixedLines),
      };
    }),
  };
}

// ================================================================
// MAIN PIPELINE
// ================================================================

function fixFile(file: AIFile): FixResult {
  const original = file.content ?? "";
  const state: State = { fixes: [], warnings: [] };

  // ── Pre-flight guards ─────────────────────────────────────────
  if (!original.trim()) {
    return skipped(file, ["empty-file"]);
  }
  if (original.length > MAX_FILE_BYTES) {
    return skipped(file, [`file-exceeds-${MAX_FILE_BYTES}-bytes`]);
  }

  const ext = file.path.split(".").pop()?.toLowerCase() ?? "";
  const isJS = ["js", "jsx", "ts", "tsx"].includes(ext);
  const isJSON = ext === "json";
  const isHTML = ext === "html";

  // CSS files: hard block
  if (ext === "css") {
    return skipped(file, ["css-not-allowed"]);
  }

  let code = original;

  // ── Non-JS files: minimal format pass then exit ───────────────
  if (!isJS) {
    if (isJSON) code = fixJSON(code, state);
    if (isHTML) code = fixHTML(code, state);
    return buildResult(file, code, original, state);
  }

  // ================================================================
  // LAYER 1 — INPUT SANITIZATION
  // ================================================================
  code = layer1InputSanitization(code, state);

  // ================================================================
  // LAYER 2 — IMPORT + DEPENDENCY STRIPPING
  // ================================================================
  code = layer2ImportStripping(code, state);

  // ================================================================
  // LAYER 3 — WHITELIST ENFORCEMENT
  // ================================================================
  code = layer3WhitelistEnforcement(code, state);

  // ================================================================
  // LAYER 4 — SYNTAX RECOVERY
  // ================================================================
  code = layer4SyntaxRecovery(code, state);

  // ================================================================
  // LAYER 5 — RUNTIME SANITIZATION
  // ================================================================
  code = layer5RuntimeSanitization(code, state);

  // ================================================================
  // LAYER 6 — HARD FALLBACK (FINAL GUARD)
  // ================================================================
  const fallbackResult = layer6HardFallback(file, code, state);
  if (fallbackResult) return fallbackResult;

  // ── Ensure default export for JSX files ──────────────────────
  if (["jsx", "tsx"].includes(ext)) {
    code = ensureDefaultExport(code, state);
  }

  // ── Final clean ───────────────────────────────────────────────
  code = finalClean(code);

  return buildResult(file, code, original, state);
}

// ================================================================
// LAYER 1 — INPUT SANITIZATION
// Encoding, control chars, line endings, LLM prose artifacts.
// ================================================================

function layer1InputSanitization(code: string, state: State): string {
  // BOM
  if (code.startsWith("\uFEFF")) {
    code = code.slice(1);
    state.fixes.push("removed-BOM");
  }

  // Early corruption detection
  if (/=\s*\/\s*>/.test(code)) {
    state.warnings.push("corrupted-arrow-token");
  }
  if (/<REDACTED/.test(code)) {
    state.warnings.push("redacted-token-found");
  }

  // Control characters
  const noCtrl = code.replace(
    /[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F\u200B\u200C\u200D\uFEFF]/g,
    "",
  );
  if (noCtrl !== code) { code = noCtrl; state.fixes.push("removed-control-chars"); }

  // Line endings
  if (code.includes("\r")) {
    code = code.replace(/\r\n/g, "\n").replace(/\r/g, "\n");
    state.fixes.push("normalized-line-endings");
  }

  // Excessive blank lines
  const noBlank = code.replace(/\n{4,}/g, "\n\n\n");
  if (noBlank !== code) { code = noBlank; state.fixes.push("collapsed-blank-lines"); }

  // Escape artifacts
  const noEsc = code.replace(/\\\\n/g, "\n").replace(/\\\\t/g, "\t");
  if (noEsc !== code) { code = noEsc; state.fixes.push("fixed-escape-artifacts"); }

  // LLM prose leakage
  const LLM_PROSE = [
    /^(Here('s| is) (the|a|an) )/i,
    /^(In this (component|file|example))/i,
    /^(This (component|file|code))/i,
    /^(Note( that)?:)/i,
    /^(Let me )/i,
    /^(I('ll| will) )/i,
    /^(Sure[,!])/i,
    /^(Certainly[,!])/i,
    /^(Of course[,!])/i,
    /^(Below is )/i,
    /^(Above is )/i,
  ];
  const lines = code.split("\n");
  const filtered = lines.filter((line) => {
    const t = line.trim();
    return !(LLM_PROSE.some((p) => p.test(t)) && !/[{;(]$/.test(t));
  });
  if (filtered.length !== lines.length) {
    code = filtered.join("\n");
    state.fixes.push("removed-LLM-prose");
  }

  // Debugger statements
  const noDbg = code.replace(/^\s*debugger\s*;?\s*$/gm, "");
  if (noDbg !== code) { code = noDbg; state.fixes.push("removed-debugger"); }

  // Console.log statements (safe: only standalone lines)
  const noLog = code
    .split("\n")
    .filter((line) => !/^\s*console\.log\([^;]*\);?\s*$/.test(line))
    .join("\n");
  if (noLog !== code) { code = noLog; state.fixes.push("removed-console-log"); }

  return code;
}

// ================================================================
// LAYER 2 — IMPORT + DEPENDENCY STRIPPING
// Removes ALL CSS imports and ALL non-whitelisted library imports.
// Merges and deduplicates remaining imports.
// ================================================================

function layer2ImportStripping(code: string, state: State): string {
  // Remove CSS imports
  const noCss = code.replace(/^import\s+['"][^'"]*\.css['"];?\n?/gm, "");
  if (noCss !== code) { code = noCss; state.fixes.push("removed-css-imports"); }

  // Remove forbidden library imports
  const noForbidden = code.replace(
    /^import\s+.+from\s+['"]([^'"]+)['"];?\n?/gm,
    (match, lib) => {
      if (FORBIDDEN_LIBS.some(f => (lib as string).includes(f))) {
        state.fixes.push(`removed-forbidden-import:${lib}`);
        return "";
      }
      return match;
    }
  );
  if (noForbidden !== code) code = noForbidden;

  // Remove all external imports (anything not in ALLOWED_IMPORTS)
  const noExternal = code.replace(
    /^import\s+(?:.+\s+from\s+)?['"]([^'"]+)['"];?\n?/gm,
    (match, ...args: unknown[]) => {
      const lib = (args[0] as string).replace(/^['"]|['"]$/g, "");
      if (ALLOWED_IMPORTS.some((a) => lib === a || lib.startsWith(a + "/"))) return match;
      state.fixes.push(`removed-external-import:${lib}`);
      return "";
    },
  );
  if (noExternal !== code) code = noExternal;

  // Deduplicate imports
  const importLines = new Set<string>();
  const deduped = code.replace(/^import\s+.+$/gm, (line) => {
    const t = line.trim();
    if (importLines.has(t)) { state.fixes.push("removed-duplicate-import"); return ""; }
    importLines.add(t);
    return line;
  });
  if (deduped !== code) code = deduped;

  // Ensure React import exists if JSX is present
  const hasJSX = /<[A-Za-z][^>]*>/.test(code) || /<>/.test(code);
  const hasReact = /import\s+React\b/.test(code) || /from\s+['"]react['"]/.test(code);
  if (hasJSX && !hasReact) {
    const usedHooks = ALLOWED_HOOKS.filter((h) => new RegExp(`\\b${h}\\b`).test(code));
    const importLine = usedHooks.length > 0
      ? `import React, { ${usedHooks.join(", ")} } from "react";`
      : `import React from "react";`;
    code = `${importLine}\n${code}`;
    state.fixes.push("injected-react-import");
  }

  return code;
}

// ================================================================
// LAYER 3 — WHITELIST ENFORCEMENT
// Neutralizes unknown hooks, replaces unknown components,
// strips class attributes, removes style strings.
// ================================================================

function layer3WhitelistEnforcement(code: string, state: State): string {
  // Neutralize any hook not in ALLOWED_HOOKS
  const noUnknownHooks = code.replace(
    /\b(use[A-Z][a-zA-Z0-9]*)\s*\(/g,
    (match, ...args: unknown[]) => {
      const hook = args[0] as string;
      if (ALLOWED_HOOKS.includes(hook)) return match;
      state.fixes.push(`neutralized-hook:${hook}`);
      return "(() => undefined)(";
    },
  );
  if (noUnknownHooks !== code) code = noUnknownHooks;

  // Replace unknown capitalized JSX components with <div> / </div>
  const noUnknownOpen = code.replace(
    /<([A-Z][A-Za-z0-9]*)/g,
    (_match, ...args: unknown[]) => {
      const tag = args[0] as string;
      state.fixes.push(`replaced-component:${tag}`);
      return "<div";
    },
  );
  if (noUnknownOpen !== code) code = noUnknownOpen;

  const noUnknownClose = code.replace(
    /<\/([A-Z][A-Za-z0-9]*)/g,
    (_match, ...args: unknown[]) => {
      const tag = args[0] as string;
      state.fixes.push(`replaced-closing-component:${tag}`);
      return "</div";
    },
  );
  if (noUnknownClose !== code) code = noUnknownClose;

  // Replace motion.* with normal elements
  code = code.replace(/motion\.(div|span|section|button)/g, "$1");

  // Remove AnimatePresence
  code = code.replace(/<AnimatePresence[^>]*>/g, "<>");
  code = code.replace(/<\/AnimatePresence>/g, "</>");

  // Replace Lucide icons with span
  code = code.replace(/<([A-Z][A-Za-z0-9]*Icon)[^>]*\/>/g, "<span />");

  // Remove className attributes entirely
  const noClass = code
    .replace(/\bclassName="[^"]*"/g, "")
    .replace(/\bclassName=\{[^}]*\}/g, "")
    .replace(/\bclass="[^"]*"/g, "");
  if (noClass !== code) { code = noClass; state.fixes.push("removed-className"); }

  // Convert style strings to style={{}}
  const noStyleStr = code.replace(/style="[^"]*"/g, "style={{}}");
  if (noStyleStr !== code) { code = noStyleStr; state.fixes.push("converted-style-strings"); }

  // Static template literal JSX attrs: attr={`value`} → attr="value"
  const noTmpl = code.replace(/([\w-]+)=`([^`${}]+)`/g, '$1="$2"');
  if (noTmpl !== code) { code = noTmpl; state.fixes.push("fixed-static-template-attrs"); }

  return code;
}

// ================================================================
// LAYER 4 — SYNTAX RECOVERY
// Fixes JSX structure, bracket imbalances, and common typos.
// When uncertain: DO NOTHING.
// ================================================================

function layer4SyntaxRecovery(code: string, state: State): string {
  // Fix useEffect cleanup arrow artifacts: ); => expr
  {
    const before = code;
    code = code
      .replace(/\)\s*;?\s*=>\s*([^\n{][^\n]*)/g, (_m, ...args: unknown[]) => {
        const body = (args[0] as string).trim();
        if (!body) return _m;
        return `)\nreturn () => { ${body} }`;
      })
      .replace(/\)\s*;\s*=>\s*(\{[\s\S]*?\})/g, (_m, ...args: unknown[]) =>
        `)\nreturn () => ${args[0] as string}`,
      )
      .replace(/^(\s*);=>\s*(.+)$/gm, (_m, ...args: unknown[]) =>
        `${args[0] as string}return () => { ${(args[1] as string).trim()} }`,
      );
    if (code !== before) state.fixes.push("fixed-useEffect-cleanup");
  }

  // JSX inline-if: { if (cond) { <El /> } } → { cond && <El /> }
  {
    const before = code;
    code = code.replace(
      /\{\s*if\s*\(([^)]+)\)\s*\{\s*(<[^}]+>)\s*\}\s*\}/g,
      (_m, ...args: unknown[]) =>
        `{ ${(args[0] as string).trim()} && ${(args[1] as string).trim()} }`,
    );
    // Multi-line variant
    const lines = code.split("\n");
    const result: string[] = [];
    let i = 0;
    while (i < lines.length) {
      const line = lines[i];
      const ifOpen = line.trim().match(/^\{\s*if\s*\(([^)]+)\)\s*\{$/);
      if (ifOpen && i + 2 < lines.length) {
        const jsxLine = lines[i + 1].trim();
        const close = lines[i + 2].trim();
        if (close === "}" || close === "} }" || close === "}}") {
          const indent = line.match(/^(\s*)/)?.[1] ?? "";
          result.push(`${indent}{ ${ifOpen[1].trim()} && ${jsxLine} }`);
          i += 3;
          continue;
        }
      }
      result.push(line);
      i++;
    }
    code = result.join("\n");
    if (code !== before) state.fixes.push("fixed-JSX-inline-if");
  }

  // JSX typo corrections
  const JSX_TYPOS: [RegExp, string][] = [
    [/<\/buutton>/gi, "</button>"], [/<\/buton>/gi, "</button>"],
    [/<\/diiv>/gi, "</div>"],       [/<\/spann>/gi, "</span>"],
    [/<\/lable>/gi, "</label>"],    [/<\/tabble>/gi, "</table>"],
    [/<\/tabel>/gi, "</table>"],    [/<buutton(\s|\/|>)/gi, "<button$1"],
    [/<buton(\s|\/|>)/gi, "<button$1"], [/<diiv(\s|\/|>)/gi, "<div$1"],
    [/<imput(\s|\/|>)/gi, "<input$1"], [/<lable(\s|\/|>)/gi, "<label$1"],
    [/\bonclick=/g, "onClick="],    [/\bOnclick=/g, "onClick="],
  ];
  let typoFixed = false;
  for (const [pat, rep] of JSX_TYPOS) {
    const next = code.replace(pat, rep);
    if (next !== code) { code = next; typoFixed = true; }
  }
  if (typoFixed) state.fixes.push("fixed-JSX-typos");

  // Fix broken closing tags: </ div> → </div>
  const noSpaceTag = code.replace(/<\/\s+(\w)/g, "</$1");
  if (noSpaceTag !== code) { code = noSpaceTag; state.fixes.push("fixed-broken-closing-tags"); }

  // label for= → htmlFor=
  const htmlFor = code.replace(/(?<=<label[^>]*)\bfor=/g, "htmlFor=");
  if (htmlFor !== code) { code = htmlFor; state.fixes.push("fixed-for-to-htmlFor"); }

  // Truncate oversized SVG paths
  const noSVG = code.replace(/<path[^>]*d="[^"]{400,}[^"]*"[^>]*>/g, '<path d="M0 0 L10 10" />');
  if (noSVG !== code) { code = noSVG; state.fixes.push("truncated-svg-path"); }

  // Fix corrupted inline style objects
  const noCorruptStyle = code.replace(/style=\{\{([^}]*)$/gm, "style={{}}");
  if (noCorruptStyle !== code) { code = noCorruptStyle; state.fixes.push("fixed-broken-style-object"); }

  // Fix empty return blocks
  const noEmptyReturn = code.replace(/return\s*\(\s*<\/\w+>/g, "return (<div></div>)");
  if (noEmptyReturn !== code) { code = noEmptyReturn; state.fixes.push("fixed-empty-return"); }

  // Fix empty return parens
  const noEmptyParens = code.replace(/return\s*\(\s*\)/g, "return (<div></div>)");
  if (noEmptyParens !== code) { code = noEmptyParens; state.fixes.push("fixed-empty-return-parens"); }

  // Collapse runaway duplicate JSX lines (6+ identical → 1)
  {
    const lines = code.split("\n");
    const result: string[] = [];
    let i = 0;
    while (i < lines.length) {
      const line = lines[i];
      let j = i + 1;
      while (j < lines.length && lines[j] === line) j++;
      const count = j - i;
      if (count >= 6 && /^\s*<[A-Za-z]/.test(line.trim())) {
        result.push(line);
        state.fixes.push(`collapsed-runaway-JSX:${count}x`);
      } else {
        for (let k = i; k < j; k++) result.push(lines[k]);
      }
      i = j;
    }
    code = result.join("\n");
  }

  return code;
}

// ================================================================
// LAYER 5 — RUNTIME SANITIZATION
// Converts style strings, strips CSS imports, deduplicates exports
// and declarations.
// ================================================================

function layer5RuntimeSanitization(code: string, state: State): string {
  // Convert style="..." → style={{ camelCase: "value" }}
  
   // Remove media queries (not supported in inline styles)
  const noMediaQueries = code.replace(
    /@media\s*\([^)]*\)\s*\{[^}]*\}/g,
    ""
  );
  if (noMediaQueries !== code) {
    code = noMediaQueries;
    state.fixes.push("removed-media-queries");
  }

  // Remove inline media queries inside style objects
  const noInlineMedia = code.replace(
    /["']@media[^"']*["']\s*:\s*\{[^}]*\}/g,
    ""
  );
  if (noInlineMedia !== code) {
    code = noInlineMedia;
    state.fixes.push("removed-inline-media-queries");
  }
  const styledCode = code.replace(/style="([^"]*)"/g, (_match, ...args: unknown[]) => {
    const s = args[0] as string;
    try {
      const obj = s
        .split(";")
        .map((x) => x.trim())
        .filter(Boolean)
        .map((rule) => {
          const colonIdx = rule.indexOf(":");
          if (colonIdx === -1) return null;
          const k = rule.slice(0, colonIdx).trim();
          const v = rule.slice(colonIdx + 1).trim();
          if (!k || !v) return null;
          const camel = k.replace(/-([a-z])/g, (_, c) => c.toUpperCase());
          return `${camel}: "${v}"`;
        })
        .filter(Boolean)
        .join(", ");
      return `style={{ ${obj} }}`;
    } catch {
      return "style={{}}";
    }
  });
  if (styledCode !== code) { code = styledCode; state.fixes.push("converted-style-to-object"); }

  // Force-remove any remaining CSS imports
  const noCss = code.replace(/import\s+['"][^'"]*\.css['"];?/g, "");
  if (noCss !== code) { code = noCss; state.fixes.push("force-removed-css"); }

  // Remove duplicate default exports (keep first)
  {
    let count = 0;
    const deduped = code.replace(/\bexport\s+default\s+(\w+)\s*;/g, (match) => {
      count++;
      if (count === 1) return match;
      state.fixes.push("removed-duplicate-export-default");
      return "";
    });
    if (deduped !== code) code = deduped;
  }

  // Fix stray brace after export default
  const fixedExport = code.replace(/\bexport\s+default\s+(\w+)\s*\}/g, "export default $1;");
  if (fixedExport !== code) { code = fixedExport; state.fixes.push("fixed-broken-export-default"); }

  // Dedup useState calls (same state variable declared twice)
  {
    const seen = new Set<string>();
    const lines = code.split("\n");
    const deduped = lines.filter((line) => {
      const m = line.trim().match(
        /^(?:const|let|var)\s+\[(\w+),\s*(\w+)\]\s*=\s*useState\s*\(/,
      );
      if (!m) return true;
      const key = `${m[1]}:${m[2]}`;
      if (seen.has(key)) { state.fixes.push(`dedup-useState:${m[1]}`); return false; }
      seen.add(key);
      return true;
    });
    if (deduped.length !== lines.length) code = deduped.join("\n");
  }

  // Dedup variable declarations (non-destructured, same name twice)
  {
    const seen = new Set<string>();
    const lines = code.split("\n");
    const deduped = lines.filter((line) => {
      const t = line.trim();
      if (/^(?:const|let|var)\s+[\[{]/.test(t)) return true;
      const m = t.match(/^(?:export\s+)?(?:const|let|var)\s+(\w+)\s*=/);
      if (!m) return true;
      if (seen.has(m[1])) { state.fixes.push(`dedup-declaration:${m[1]}`); return false; }
      seen.add(m[1]);
      return true;
    });
    if (deduped.length !== lines.length) code = deduped.join("\n");
  }

  return code;
}

// ================================================================
// LAYER 6 — HARD FALLBACK (FINAL GUARD)
// If ANY unsafe signal remains → replace entire file with fallback.
// ================================================================

function layer6HardFallback(
  file: AIFile,
  code: string,
  state: State,
): FixResult | null {
  const triggers: string[] = [];

  // Unknown hook survived
  const hookMatches = code.match(/\buse[A-Z][a-zA-Z0-9]+/g) ?? [];
  const unknownHooks = hookMatches.filter((h) => !ALLOWED_HOOKS.includes(h));
  if (unknownHooks.length > 0) triggers.push(`unknown-hooks:${unknownHooks.join(",")}`);

  // Invalid style string survived
  if (/style="[^"]*"/.test(code)) triggers.push("invalid-style-string");

  // CSS import survived
  if (/\.css['"]/.test(code)) triggers.push("css-import-survived");

  // Catastrophic syntax tokens
  if (/Unexpected end of file|Unterminated/.test(code)) triggers.push("catastrophic-syntax");

  // Structural check failure
  const check = structuralCheck(code);
  if (!check.valid) triggers.push(`syntax-failure:${check.issues.join(";")}`);

  // Severe brace mismatch (more than 2 off)
  const stripped = stripStrings(code);
  const openB = (stripped.match(/\{/g) ?? []).length;
  const closeB = (stripped.match(/\}/g) ?? []).length;
  if (Math.abs(openB - closeB) > 2) triggers.push(`severe-brace-mismatch:${openB}/${closeB}`);

  // New triggers for forbidden libraries and corruption
  if (/framer-motion/.test(code)) triggers.push("forbidden-lib:framer-motion");
  if (/lucide-react/.test(code)) triggers.push("forbidden-lib:lucide");
  if (/=\s*\/\s*>/.test(code)) triggers.push("corrupted-arrow");
  if (/<REDACTED/.test(code)) triggers.push("redacted-token");

  if (triggers.length === 0) return null;

  state.warnings.push(`fallback-triggered:${triggers.join("|")}`);

  return {
    file: {
      ...file,
      content: SAFE_FALLBACK_COMPONENT,
      meta: { sanitized: true, fixCount: state.fixes.length + 1, riskLevel: "high" },
    },
    fixes: [...state.fixes, "hard-fallback"],
    warnings: state.warnings,
    riskLevel: "high",
    skipped: false,
  };
}

// ================================================================
// STRUCTURAL CHECK
// ================================================================

function structuralCheck(code: string): { valid: boolean; issues: string[] } {
  const issues: string[] = [];
  const stripped = stripStrings(code);

  const openBraces = (stripped.match(/\{/g) ?? []).length;
  const closeBraces = (stripped.match(/\}/g) ?? []).length;
  if (openBraces !== closeBraces) {
    issues.push(`unmatched-braces:${openBraces}open/${closeBraces}close`);
  }

  const openParens = (stripped.match(/\(/g) ?? []).length;
  const closeParens = (stripped.match(/\)/g) ?? []).length;
  if (openParens !== closeParens) {
    issues.push(`unmatched-parens:${openParens}open/${closeParens}close`);
  }

  const defaultExports = (code.match(/\bexport\s+default\b/g) ?? []).length;
  if (defaultExports > 1) issues.push(`multiple-export-defaults:${defaultExports}`);

  if (/from\s+['"](?!react|react-dom)/.test(code)) issues.push("external-import-survived");
  if (/\.css['"]/.test(code)) issues.push("css-import-survived");
  if (/style="[^"]*"/.test(code)) issues.push("invalid-style-string");

  // New forbidden library checks
  if (/framer-motion/.test(code)) issues.push("forbidden-lib-present");
  if (/lucide-react/.test(code)) issues.push("forbidden-lib-present");

  const tagBalance = checkJSXTagBalance(code);
  if (!tagBalance.valid) issues.push(`unbalanced-jsx:${tagBalance.reason}`);

  return { valid: issues.length === 0, issues };
}

function stripStrings(code: string): string {
  return code
    .replace(/`(?:[^`\\]|\\.)*`/gs, "``")
    .replace(/"(?:[^"\\]|\\.)*"/g, '""')
    .replace(/'(?:[^'\\]|\\.)*'/g, "''")
    .replace(/\/\*[\s\S]*?\*\//g, "")
    .replace(/\/\/[^\n]*/g, "");
}

const VOID_ELEMENTS = new Set([
  "area", "base", "br", "col", "embed", "hr", "img",
  "input", "link", "meta", "param", "source", "track", "wbr",
]);

function checkJSXTagBalance(code: string): { valid: boolean; reason?: string } {
  const masked = code
    .replace(/`(?:[^`\\]|\\.)*`/gs, "``")
    .replace(/"(?:[^"\\]|\\.)*"/g, '""')
    .replace(/'(?:[^'\\]|\\.)*'/g, "''")
    .replace(/\/\*[\s\S]*?\*\//g, "")
    .replace(/\/\/[^\n]*/g, "");

  const stack: string[] = [];
  const tagRe = /<\/?([A-Za-z][a-zA-Z0-9]*)[^>]*\/?>/g;
  let m: RegExpExecArray | null;
  let mismatch = 0;

  while ((m = tagRe.exec(masked)) !== null) {
    const full = m[0];
    const name = m[1];
    if (!name) continue;
    if (full.startsWith("</")) {
      if (stack.length === 0) { mismatch++; continue; }
      const last = stack.pop();
      if (last !== name) mismatch++;
    } else if (!full.endsWith("/>") && !VOID_ELEMENTS.has(name.toLowerCase())) {
      stack.push(name);
    }
  }

  if (stack.length > 0) return { valid: false, reason: `unclosed:${stack.join(",")}` };
  if (mismatch > 3) return { valid: false, reason: `mismatches:${mismatch}` };
  return { valid: true };
}

// HELPERS
// ================================================================

function ensureDefaultExport(code: string, state: State): string {
  if (/export\s+default\b/.test(code)) return code;

  const fnMatch = code.match(
    /(?:export\s+)?function\s+([A-Z]\w*)\s*\([^)]*\)\s*\{[^}]*return\s*\(?\s*</,
  );
  if (fnMatch) {
    const name = fnMatch[1];
    const fixed = code.replace(
      new RegExp(`^function\\s+${name}\\b`, "m"),
      `export default function ${name}`,
    );
    if (fixed !== code) { state.fixes.push(`added-export-default:${name}`); return fixed; }
  }

  const arrowMatch = code.match(
    /const\s+([A-Z]\w*)\s*=\s*(?:\([^)]*\)|[^=])\s*=>\s*(?:\(?\s*<)/,
  );
  if (arrowMatch) {
    const name = arrowMatch[1];
    state.fixes.push(`added-export-default-bottom:${name}`);
    return `${code}\n\nexport default ${name};\n`;
  }

  return code;
}

function finalClean(code: string): string {
  return (
    code
      .split("\n")
      .map((l) => l.trimEnd())
      .join("\n")
      .trimEnd() + "\n"
  );
}

function skipped(file: AIFile, fixes: string[]): FixResult {
  return {
    file: { ...file, meta: { sanitized: false, fixCount: 0, riskLevel: "low" } },
    fixes,
    warnings: [],
    riskLevel: "low",
    skipped: true,
  };
}

function buildResult(
  file: AIFile,
  code: string,
  original: string,
  state: State,
): FixResult {
  if (code === original && state.fixes.length === 0) return skipped(file, []);
  const fixCount = state.fixes.length;
  const riskLevel: "low" | "medium" | "high" =
    fixCount <= 4 ? "low" : fixCount <= 15 ? "medium" : "high";
  return {
    file: { path: file.path, content: code, meta: { sanitized: true, fixCount, riskLevel } },
    fixes: state.fixes,
    warnings: state.warnings,
    riskLevel,
    skipped: false,
  };
}

// NON-JS FORMAT HELPERS

function fixJSON(code: string, state: State): string {
  try { JSON.parse(code); return code; } catch { /* try repair */ }
  const fixed = code.replace(/,\s*([}\]])/g, "$1");
  try {
    JSON.parse(fixed);
    state.fixes.push("json-removed-trailing-commas");
    return fixed;
  } catch {
    state.warnings.push("json-invalid:manual-fix-required");
    return code;
  }
}

function fixHTML(code: string, state: State): string {
  let c = code;
  let changed = false;
  if (c.includes("<head") && !c.includes("charset")) {
    c = c.replace(/<head>/, '<head>\n  <meta charset="UTF-8" />');
    changed = true;
  }
  if (c.includes("<html") && !c.includes("lang=")) {
    c = c.replace(/<html>/, '<html lang="en">');
    changed = true;
  }
  if (changed) state.fixes.push("html-meta-fixes");
  return c;
}

export async function POST(req: NextRequest) {
  // ───────────── AUTH ─────────────
  const clerkUser = await currentUser();
  if (!clerkUser) {
    return new Response(JSON.stringify({ error: "Unauthorized" }), {
      status: 401,
      headers: { "Content-Type": "application/json" },
    });
  }

  // ───────────── BODY ─────────────
  let body: { subaccountId?: string; prompt?: string; name?: string };
  try {
    body = await req.json();
  } catch {
    return new Response(JSON.stringify({ error: "Invalid JSON" }), {
      status: 400,
      headers: { "Content-Type": "application/json" },
    });
  }

  const { subaccountId, prompt, name } = body;

  if (!subaccountId || !prompt) {
    return new Response(
      JSON.stringify({ error: "Missing required fields: subaccountId, prompt" }),
      { status: 400, headers: { "Content-Type": "application/json" } }
    );
  }

  // ───────────── PERMISSIONS ─────────────
  const authUser = await getAuthUserDetails();
  if (!authUser?.agency) {
    return new Response(JSON.stringify({ error: "User not found" }), {
      status: 401,
      headers: { "Content-Type": "application/json" },
    });
  }

  const role = clerkUser.privateMetadata.role as Role | undefined;
  const isOwnerOrAdmin = role === Role.AGENCY_OWNER || role === Role.AGENCY_ADMIN;

  if (!isOwnerOrAdmin) {
    const hasPermission = authUser.permissions.find(
      (p) => p.access && p.subAccountId === subaccountId
    );
    if (!hasPermission) {
      return new Response(JSON.stringify({ error: "No access" }), {
        status: 403,
        headers: { "Content-Type": "application/json" },
      });
    }
  }

  // ───────────── SSE STREAM ─────────────
  const projectId = crypto.randomUUID();
  const allFiles: AIFileNode[] = [];

  const stream = new ReadableStream({
    async start(controller) {
      const encoder = new TextEncoder();

      const send = (data: object) => {
        try {
          controller.enqueue(encoder.encode(`data: ${JSON.stringify(data)}\n\n`));
        } catch {
          // client disconnected – ignore, pipeline will finish and save to DB anyway
        }
      };

      const ping = () => {
        try {
          controller.enqueue(encoder.encode(`: keep-alive\n\n`));
        } catch {}
      };

      const heartbeat = setInterval(ping, 8_000);

      try {
        // Create the DB project record first
        const dbProject = await createAIProject(
          subaccountId,
          name || prompt.slice(0, 50) || "AI Project",
          prompt
        );

        send({
          type: "project_init",
          projectId,
          dbProjectId: dbProject?.id ?? projectId,
          name: name || prompt.slice(0, 50),
        });

        // ── Run pipeline with all required callbacks ──
        await streamProjectPipeline(prompt, {
          onFile: async (file) => {
            // ── Safety: replace unsafe files before emitting ──
            let safeFile = file;
            if (isUnsafeFile(file.content)) {
              safeFile = { ...file, ...makeSafeFile(file.path) };
            }

            const existing = allFiles.findIndex((f) => f.path === safeFile.path);
            const node: AIFileNode = {
              path: safeFile.path,
              content: safeFile.content,
              type: "file",
            };
            if (existing >= 0) {
              allFiles[existing] = node;
            } else {
              allFiles.push(node);
            }

            send({
              type: "file",
              path: safeFile.path,
              content: safeFile.content,
              provisional: safeFile.provisional ?? false,
              size: safeFile.content.length,
            });
          },

          onLog: (message) => {
            send({ type: "log", message });
          },

          onError: (message) => {
            send({ type: "error", message });
          },

          onPing: ping,

          onDone: () => {
            // Called when pipeline finishes successfully
            console.log(`✅ Pipeline completed for project ${projectId}`);
          },
        });

        // ── Final safety before DB save: replace any unsafe files ──
        const safeFiles = allFiles.map((f) =>
          isUnsafeFile(f.content) ? makeSafeFile(f.path) : f
        );

        // ── Save ALL files to DB once generation is complete ──
        if (safeFiles.length > 0) {
          try {
            await upsertAIProjectFiles(projectId, safeFiles);
            console.log(`✅ Saved ${safeFiles.length} files to DB for project ${projectId}`);
          } catch (dbErr) {
            console.error("⚠️ DB save failed:", dbErr);
          }
        }

        send({ type: "done", fileCount: safeFiles.length, projectId });
      } catch (err) {
        const msg = (err as Error)?.message ?? "Generation failed";
        console.error("❌ [generate/stream] crash:", msg);
        send({ type: "error", message: msg });

        // Emit safe fallback files (CSS-free)
        const fallback = makeFallbackFiles(prompt);
        for (const f of fallback) {
          send({
            type: "file",
            path: f.path,
            content: f.content,
            provisional: false,
          });
        }
        send({ type: "done", fileCount: fallback.length, projectId, fallback: true });
      } finally {
        clearInterval(heartbeat);
        controller.close();
      }
    },
  });

  return new Response(stream, {
    headers: {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache, no-transform",
      Connection: "keep-alive",
      "Transfer-Encoding": "chunked",
      "X-Accel-Buffering": "no",
    },
  });
}

      // Don't return yet — fall through to initial boot logic below.
    }

    // ── FIX LP1: Initial boot ────────────────────────────────────────────────
    // Only boot if:
    //   (a) we haven't booted for this project yet, AND
    //   (b) the WC isn't already running/installing something
    //       (ChatPanel may have already queued a mountAndRun)
    if (!isBootedRef.current) {
      isBootedRef.current     = true;
      prevFilesMapRef.current = new Map(files.map((f) => [f.path, f.content]));

      const wcAlreadyBusy =
        status === "booting"    ||
        status === "mounting"   ||
        status === "installing" ||
        status === "starting"   ||
        status === "running";

      if (wcAlreadyBusy) {
        // ChatPanel or another caller already kicked off the pipeline. Stand down.
        console.log(`[LivePreview] WC already ${status} — skipping duplicate boot`);
      } else {
        // WC is idle — this is a DB-load scenario. LivePreview owns the boot.
        console.log("[LivePreview] WC idle on first boot → mountAndRun");
        mountAndRun(files);
      }
      return;
    }

    // ── Already booted: diff for code-editor edits ───────────────────────────
    const newMap      = new Map(files.map((f) => [f.path, f.content]));
    const changed: AIFile[] = [];

    for (const file of files) {
      if (prevFilesMapRef.current.get(file.path) !== file.content) {
        changed.push(file);
      }
    }

    prevFilesMapRef.current = newMap;

    if (!changed.length) return;

    // Large ChatPanel batch — it already drove mountAndRun/remount directly.
    const isChatPanelCommit =
      lastCommitLabel.startsWith("Generated:") ||
      lastCommitLabel.startsWith("Edit:");

    if (isChatPanelCommit || changed.length >= 5) {
      console.log(
        `⏭️  [LivePreview] skipping ChatPanel batch (${changed.length} files, "${lastCommitLabel}")`
      );
      return;
    }

    // Small change = user manually edited a file in the code editor.
    const needsRemount = changed.some(
      (f) =>
        f.path === "package.json"   ||
        f.path === "vite.config.js" ||
        f.path === "vite.config.ts" ||
        f.path === "index.html"
    );

    if (needsRemount) {
      console.log("🔄 [LivePreview] structural change → remounting");
      remount(files);
    } else {
      console.log(`⚡ [LivePreview] HMR patch — ${changed.length} file(s)`);
      writeFiles(changed);
    }
  }, [files, projectId, lastCommitLabel, status, isInstalled, mountAndRun, writeFiles, remount]);

  // ── Manual refresh ──────────────────────────────────────────────────────────
  const handleRefresh = () => {
    if (!files.length) return;
    isBootedRef.current        = false;
    prevFilesMapRef.current    = new Map();
    bootedProjectIdRef.current = null;
    // FIX LP3: bump reload key to force iframe hard-reload, not just src re-set
    setIframeReloadKey((k) => k + 1);
    remount(files);
  };

  return (
    <div style={{
      height: "100%",
      display: "flex",
      flexDirection: "column",
      background: "#0f172a",
      color: "#e2e8f0"
    }}>
      {/* Toolbar */}
      <div style={{
        display: "flex",
        alignItems: "center",
        justifyContent: "space-between",
        padding: "8px 12px",
        borderBottom: "1px solid #1e293b",
        flexShrink: 0,
      }}>
        <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
          <span style={{ display: "inline-flex", color: "#94a3b8" }}>
            <Icons.monitor />
          </span>
          <span style={{ fontSize: 12, fontWeight: 500 }}>Preview</span>
        </div>
        <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
          <StatusBadge status={status} />
          <button
            onClick={handleRefresh}
            disabled={!files.length}
            title="Force remount"
            style={{
              padding: 4,
              borderRadius: 4,
              background: "transparent",
              border: "none",
              cursor: files.length ? "pointer" : "default",
              opacity: files.length ? 1 : 0.4,
              color: "#94a3b8",
              display: "inline-flex",
            }}
          >
            <Icons.refresh />
          </button>
        </div>
      </div>

      {/* Content */}
      <div style={{
        flex: 1,
        position: "relative",
        overflow: "hidden",
        height: "100%"
      }}>
        {url ? (
          // FIX LP3: key only changes on explicit refresh, not on every URL change.
          // src changes navigate the iframe without tearing it down.
          <div style={{
            position: "absolute",
            inset: 0,
            width: "100%",
            height: "100%",
            overflow: "hidden",
            background: "#0f172a"
          }}>
            <iframe
              key={iframeReloadKey}
              src={url}
              allow="cross-origin-isolated"
              sandbox="allow-scripts allow-same-origin allow-forms allow-popups allow-modals"
              style={{
                width: "100%",
                height: "100%",
                border: 0,
                display: "block",
                background: "#0f172a"
              }}
              title="Live preview"
            />
          </div>
        ) : (
          <PreviewPlaceholder status={status} />
        )}
      </div>
    </div>
  );
});

// ─── Deploy Drawer ─────────────────────────────────────────────────────────────
function DeployDrawer({
  projectId,
  subaccountId,
  onClose,
}: {
  projectId: string;
  subaccountId: string;
  onClose: () => void;
}) {
  const [target, setTarget] = useState<DeployTarget>("vercel");
  const [status, setStatus] = useState<"idle" | "deploying" | "done" | "error">("idle");
  const [deployedUrl, setDeployedUrl] = useState<string | null>(null);
  const [copied, setCopied] = useState(false);

  const handleDeploy = async () => {
    setStatus("deploying");
    try {
      const res = await fetch("/api/ai/deploy", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ projectId, subaccountId, target }),
      });

      if (data?.url) { setDeployedUrl(data.url); setStatus("done"); }
      else setStatus("error");
    } catch { setStatus("error"); }
  };

  const copyUrl = () => {
    if (!deployedUrl) return;
    navigator.clipboard.writeText(deployedUrl);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  const targets: { id: DeployTarget; label: string; description: string }[] = [
    { id: "vercel", label: "Vercel", description: "Instant global CDN deployment" },
    { id: "netlify", label: "Netlify", description: "JAMstack with form support" },
    { id: "github-pages", label: "GitHub Pages", description: "Free static hosting" },
  ];

  return (

      onClick={(e) => { if (e.target === e.currentTarget) onClose(); }}

          <p className="text-xs text-white/40 mt-0.5">Choose a platform and go live</p>
        </div>
        <div className="px-5 py-4 flex flex-col gap-2">

const AIWorkspace = () => {

  const loadedForProjectRef = useRef<string | null>(null);

  const [previewTab, setPreviewTab]       = useState<PreviewTab>("preview");

  const [exportMenuOpen, setExportMenuOpen]     = useState(false);

  // ── Reset + load effect (FIXED for StrictMode) ───────────────────────────────
  useEffect(() => {
    if (!projectId || projectId === "new") return;
    if (loadedForProjectRef.current === projectId) return;

    // ✅ FIX: set the ref BEFORE reset() so StrictMode's second invoke
    // hits the guard above and exits — no double-wipe mid-load.
    loadedForProjectRef.current = projectId;

    resetContainer();
    reset();

    const load = async () => {
      try {
        const res = await fetch(`/api/ai/projects/${projectId}`);
        if (!res.ok) {
          console.warn(`[AIWorkspace] Project ${projectId} not found — new project`);
          setProject(projectId);
          return;
        }

        const data = await res.json();
        const loadedFiles = data?.project?.files ?? [];

        setProject(projectId);

        if (loadedFiles.length > 0) {
          setFiles(loadedFiles, "Loaded from DB");
        }
      } catch (err) {
        console.error("[AIWorkspace] Failed to load project files:", err);
      }
    };

    load();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [projectId]);

  // ── Auto‑switch sidebar to Files tab when files arrive ───────────────────────
  useEffect(() => {
    if (files.length > 0) {
      setSidebarTab("files");
      if (!sidebarOpen) setSidebarOpen(true);

          onClick={() => setSidebarOpen((v) => !v)}
          title={sidebarOpen ? "Collapse sidebar" : "Expand sidebar"}
          className="shrink-0 self-center -ml-px z-10 w-5 h-10 flex items-center justify-center border border-border rounded-r-lg bg-background hover:bg-muted transition-colors text-muted-foreground hover:text-foreground">
          {sidebarOpen ? <Icons.close /> : <Icons.open />}
        </button>

        {/* Chat */}
        <div className="w-[360px] shrink-0 flex flex-col border-r border-border overflow-hidden">
          <ChatPanel />
        </div>

        {/* Preview / Code */}
        <div className="flex-1 min-w-0 flex flex-col overflow-hidden">
          <div className="shrink-0 flex items-center gap-1 px-3 py-1.5 border-b border-border bg-background">
            <div className="flex items-center rounded-lg border border-border overflow-hidden">
              <button
                onClick={() => setPreviewTab("preview")}
                className={cn(
                  "px-2.5 py-1 text-xs flex items-center gap-1 transition-colors",
                  previewTab === "preview"
                    ? "bg-muted text-foreground"
                    : "text-muted-foreground hover:text-foreground"
                )}>
                <Icons.monitor /> Preview
              </button>
              <button
                onClick={() => setPreviewTab("code")}
                className={cn(
                  "px-2.5 py-1 text-xs flex items-center gap-1 transition-colors",
                  previewTab === "code"
                    ? "bg-muted text-foreground"
                    : "text-muted-foreground hover:text-foreground"
                )}>
                <Icons.code /> Code
              </button>
            </div>
            {previewTab === "preview" && (
              <div className="flex items-center rounded-lg border border-border overflow-hidden ml-1">
                {(["desktop", "tablet", "mobile"] as Device[]).map((d) => {
                  const IconComponent = d === "desktop" ? Icons.monitor : d === "tablet" ? Icons.tablet : Icons.mobile;
                  return (
                    <button
                      key={d}
                      onClick={() => setDevice(d)}
                      title={d.charAt(0).toUpperCase() + d.slice(1)}
                      className={cn(
                        "px-2 py-1 transition-colors",
                        device === d
                          ? "bg-muted text-foreground"
                          : "text-muted-foreground hover:text-foreground"
                      )}>
                      <IconComponent />
                    </button>
                  );
                })}
              </div>
            )}
          </div>
          <div className="flex-1 min-h-0 flex items-start justify-center overflow-hidden bg-muted/30">
            {previewTab === "preview" ? (
              <div
                className="h-full transition-all duration-300 overflow-hidden"
                style={{ width: DEVICE_WIDTHS[device] }}>
                <SafePreview projectId={storeProjectId} />
              </div>
            ) : (
              <div className="w-full h-full">
                <MonacoEditor />
              </div>
            )}
          </div>
        </div>
      </div>

      {showDeployDrawer && storeProjectId && (
        <DeployDrawer
          projectId={storeProjectId}
          subaccountId={subaccountId}
          onClose={() => setShowDeployDrawer(false)}
        />
      )}
    </div>
  );
};

export default AIWorkspace;
