// PIPELINE v19.0 — FILE‑BY‑FILE · NEW MODEL ROSTER · HARD VALIDATION · NEVER COLLAPSE
//
// ARCHITECTURE:
//   S0  → Boilerplate emit (instant, WebContainer can boot)
//   S1  → Planner   — uses PLAN_MODELS roster (kimi-k2, mistral-large, llama-4-maverick)
//   S1.5→ Spec validation + prioritization
//   S2  → Generator — file-by-file using CODEGEN_MODELS roster (qwen3-coder, devstral, llama-3.3)
//          Each file is an independent LLM call → no JSON encoding failure possible
//   S3  → Deterministic auto-fix (import resolver, JSX balancer, React injector, export enforcer)
//   S3.5→ Hard validation gate — forbidden imports, structural check, safe-replace bad files
//   S4  → Final emit — only changed files, all validated
//
// ROOT CAUSE FIX:
//   Old batch-generator asked LLM to return ALL files as ONE JSON array with code as string values.
//   LLM cannot reliably JSON-escape multiline code → truncation → parse failure → fallback.
//   Fix: ONE file per LLM call. Raw code output. No JSON. No parsing. No failure modes.
//
// MODEL ROSTER (NVIDIA API):
//   PLAN_MODELS:    kimi-k2-instruct, mistral-large-3, llama-4-maverick
//   CODEGEN_MODELS: qwen3-coder-480b, devstral-2-123b, llama-3.3-70b

class ModelHealthTracker {
  private failures = new Map<string, number>();
  private readonly SKIP_THRESHOLD = 3;

  recordFailure(model: string): void {
    this.failures.set(model, (this.failures.get(model) ?? 0) + 1);
  }

  recordSuccess(model: string): void {
    this.failures.delete(model);
  }

  isHealthy(model: string): boolean {
    return (this.failures.get(model) ?? 0) < this.SKIP_THRESHOLD;
  }

  getHealthy<T extends string>(models: readonly T[]): T[] {
    return models.filter((m) => this.isHealthy(m));
  }
}

const planHealth    = new ModelHealthTracker();
const codeHealth    = new ModelHealthTracker();

class RateLimiter {
  private timestamps: number[] = [];
  private readonly MAX_RPM     = 5;
  private readonly WINDOW_MS   = 62_000;

  async wait(label: string, onPing?: () => void): Promise<void> {
    const now = Date.now();
    this.timestamps = this.timestamps.filter((t) => now - t < this.WINDOW_MS);
    if (this.timestamps.length < this.MAX_RPM) {
      this.timestamps.push(now);
      return;
    }
    const waitMs = this.WINDOW_MS - (now - this.timestamps[0]) + 300;
    plog(`[RL] Rate limit — waiting ${(waitMs / 1000).toFixed(1)}s for ${label}`);
    await sleepPing(waitMs, onPing);
    await this.wait(label, onPing);
  }
}

const rateLimiter = new RateLimiter();

let _log: (msg: string) => void = console.log;

function plog(msg: string): void { _log(msg); }
function perr(msg: string): void { _log(`❌ ${msg}`); console.error(msg); }

function sleep(ms: number): Promise<void> {
  return new Promise((r) => setTimeout(r, ms));
}

async function sleepPing(ms: number, onPing?: () => void): Promise<void> {
  if (ms <= 0) return;
  if (!onPing || ms <= PING_INTERVAL_MS) return sleep(ms);
  const end = Date.now() + ms;
  while (Date.now() < end) {
    await sleep(Math.min(PING_INTERVAL_MS, end - Date.now()));
    if (Date.now() < end) onPing();
  }
}

function backoff(attempt: number): number {
  return Math.min(1200 * Math.pow(2, attempt), 20_000);
}

function mlabel(model: string): string {
  return model.split("/")[1] ?? model;
}

/**
 * Strip markdown fences from LLM output.
 * Models often wrap output in ```jsx or ```javascript blocks.
 */
function stripMarkdown(text: string): string {
  return text
    .replace(/^```(?:jsx?|javascript|typescript|tsx?|json)?\s*/gim, "")
    .replace(/^```\s*$/gim, "")
    .replace(/^["']use (?:client|server)["'];\s*\n*/m, "")
    .trim();
}

/**
 * Fix common token corruption patterns from NVIDIA models.
 * e.g. "= />" → "=>" (corrupted arrow function token)
 */
function fixTokenCorruption(code: string): string {
  let c = code;
  c = c.replace(/(\)|\])\s*=\s*\/\s*>/g, "$1 =>");
  c = c.replace(/(?<![=!<>])=\s*\/\s*>/g, "=>");
  c = c.replace(/\b(useState|useReducer)\s*\(\s*<REDACTED[^>]*>\s*\)/g, "$1(null)");
  c = c.replace(/=\s*\{\s*<REDACTED[^>]*>\s*\}/g, "={null}");
  c = c.replace(/\[REDACTED[^\]]*\]/g, "''");
  c = c.replace(/"<REDACTED[^>]*>"/g, "''");
  c = c.replace(/'<REDACTED[^>]*>'/g, "''");
  return c;
}

/**
 * Remove imports of banned packages.
 * Also converts <Link> to <a> if react-router Link was used.
 */
function stripBannedImports(content: string): string {
  const lines = content.split("\n");
  const stripped: string[] = [];
  const removed: string[] = [];

  for (const line of lines) {
    const t = line.trim();
    if (!t.startsWith("import ")) { stripped.push(line); continue; }

    // CSS imports — always remove
    if (/['"][^'"]*\.css['"]/.test(t)) { removed.push("*.css"); continue; }

    const fromMatch = t.match(/from\s+['"]([^'"]+)['"]/);
    if (!fromMatch) { stripped.push(line); continue; }

    const pkg = fromMatch[1];
    const banned = BANNED_PACKAGES.some((b) => pkg === b || pkg.startsWith(b + "/"));
    if (banned) { removed.push(pkg); continue; }

    stripped.push(line);
  }

  let result = stripped.join("\n");

  if (removed.length > 0) {
    plog(`[RESOLVER] Stripped banned imports: ${removed.join(", ")}`);
  }

  // Convert react-router Link to <a>
  if (/<Link[\s>]/.test(result) || /<\/Link>/.test(result)) {
    result = result
      .replace(/<Link\s+to=(['"])/g, "<a href=$1")
      .replace(/<Link\s+href=(['"])/g, "<a href=$1")
      .replace(/<Link(\s[^>]*)?>/g, "<a$1>")
      .replace(/<\/Link>/g, "</a>");
  }

  return result;
}

/**
 * Inject `import React from 'react';` if JSX is present and React is not imported.
 */
function ensureReactImport(content: string): string {
  if (/\bimport\s+React\b/.test(content)) return content;
  if (/<[A-Za-z]/.test(content)) return `import React from 'react';\n${content}`;
  return content;
}

/**
 * Ensure `export default` exists in a JSX file.
 */
function ensureDefaultExport(content: string, componentName: string): string {
  if (/\bexport\s+default\b/.test(content)) return content;
  const safe = componentName.replace(/[^a-zA-Z0-9_$]/g, "") || "Component";
  return `${content}\n\nexport default ${safe};\n`;
}

/**
 * Remove duplicate import lines (same full import statement).
 */
function deduplicateImports(content: string): string {
  const seen = new Set<string>();
  return content.split("\n").filter((line) => {
    const t = line.trim();
    if (!t.startsWith("import ")) return true;
    if (seen.has(t)) return false;
    seen.add(t);
    return true;
  }).join("\n");
}

/**
 * Strip trailing whitespace, collapse 4+ blank lines to 2, add trailing newline.
 */
function cleanOutput(content: string): string {
  return content
    .replace(/[ \t]+$/gm, "")
    .replace(/\n{4,}/g, "\n\n\n")
    .trimEnd() + "\n";
}

/**
 * Check if a file path is a CSS file (should never reach runtime).
 */
function isCssFile(path: string): boolean {
  return /\.(css|scss|sass|less)$/.test(path);
}

/**
 * Check if file content imports any banned package or CSS.
 */
function hasBannedImports(content: string): boolean {
  const lines = content.split("\n");
  for (const line of lines) {
    const t = line.trim();
    if (!t.startsWith("import ")) continue;
    if (/['"][^'"]*\.css['"]/.test(t)) return true;
    const m = t.match(/from\s+['"]([^'"]+)['"]/);
    if (!m) continue;
    const pkg = m[1];
    if (BANNED_PACKAGES.some((b) => pkg === b || pkg.startsWith(b + "/"))) return true;
  }
  return false;
}

/**
 * Hard structural validation for a single JSX/JS file.
 * Returns passed=true only if the file meets minimum production standards.
 */
function validateFile(file: StreamFile): ValidationResult {
  const reasons: string[] = [];

  // CSS files are unconditionally rejected
  if (isCssFile(file.path)) {
    return { passed: false, reasons: [`CSS file rejected: ${file.path}`] };
  }

  // Non-JSX/JS files (json, etc.) are accepted as-is
  if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js")) {
    return { passed: true, reasons: [] };
  }

  const c = file.content ?? "";

  if (c.trim().length < MIN_FILE_CHARS) {
    reasons.push(`too short (${c.trim().length} chars)`);
    return { passed: false, reasons };
  }

  // Banned imports
  if (hasBannedImports(c)) {
    reasons.push("contains banned imports");
  }

  // JSX-specific checks
  if (file.path.endsWith(".jsx")) {
    if (!/\bexport\s+default\b/.test(c)) {
      reasons.push("missing export default");
    }

    if (!/return\s*\(?\s*</.test(c)) {
      reasons.push("no JSX return statement");
    }

    // Multiple export defaults
    const exportCount = (c.match(/\bexport\s+default\b/g) ?? []).length;
    if (exportCount > 1) {
      reasons.push(`multiple export default (${exportCount})`);
    }

    // JSX tag balance check
    const openTags  = (c.match(/<[A-Z][a-zA-Z0-9]*[\s>]/g) ?? []).length;
    const closeTags = (c.match(/<\/[A-Z][a-zA-Z0-9]*>/g) ?? []).length;
    const selfClose = (c.match(/<[A-Z][a-zA-Z0-9]*[^>]*\/>/g) ?? []).length;
    const delta     = openTags - closeTags - selfClose;
    if (Math.abs(delta) > JSX_IMBALANCE_TOL) {
      reasons.push(`JSX tag imbalance (delta=${delta})`);
    }
  }

  return { passed: reasons.length === 0, reasons };
}

/**
 * Apply validation to all files.
 * Files that fail are replaced with a safe minimal component (never drop them).
 */
function applyValidationGate(files: StreamFile[], context: string): StreamFile[] {
  return files.map((file) => {
    if (BOILERPLATE_PATHS.has(file.path)) return file;
    if (isCssFile(file.path)) {
      plog(`[GATE][${context}] Dropping CSS file: ${file.path}`);
      return null as unknown as StreamFile;
    }
    const v = validateFile(file);
    if (!v.passed) {
      plog(`[GATE][${context}] ${file.path} failed — ${v.reasons.join("; ")} → safe-replacing`);
      return makeSafeComponent(file);
    }
    return file;
  }).filter(Boolean);
}

/**
 * Create a minimal valid React component as a safe fallback for a broken file.
 */
function makeSafeComponent(file: StreamFile): StreamFile {
  const rawName   = file.path.split("/").pop()?.replace(/\.[^.]+$/, "") ?? "Component";
  const safeName  = rawName.replace(/[^a-zA-Z0-9_$]/g, "").replace(/^\d+/, "") || "Component";
  // Capitalise first letter
  const compName  = safeName.charAt(0).toUpperCase() + safeName.slice(1);

  const content = cleanOutput(
    `import React from 'react';

function ${compName}() {
  return (
    <section style={{ padding: '4rem 2rem', textAlign: 'center', background: '#0a0a0f', minHeight: '200px', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
      <div>
        <h2 style={{ fontSize: '1.75rem', fontWeight: 700, color: '#f1f5f9', margin: 0 }}>${compName}</h2>
        <p style={{ color: '#64748b', marginTop: '0.75rem', fontSize: '0.95rem' }}>Section content</p>
      </div>
    </section>
  );
}

export default ${compName};`
  );

  return { ...file, content };
}

const PLANNER_SYSTEM = `You are a world-class React architect. Return ONLY raw JSON — no markdown, no backticks, no explanation, no preamble.

Schema:
{
  "projectName": "PascalCase name",
  "description": "One sentence description",
  "globalContext": "4-5 sentences about the design philosophy, visual style, and component relationships",
  "primaryColor": "#hexcode",
  "files": [
    {
      "path": "src/components/Hero.jsx",
      "description": "Full-screen hero section with animated headline, subheadline, and two CTA buttons",
      "dependencies": []
    }
  ]
}

REQUIREMENTS:
- Plan exactly 8-14 component files for a production-quality website
- ALWAYS include src/App.jsx as the last file
- ALL paths MUST start with "src/"
- Use folders: src/components/, src/sections/, src/layouts/, src/hooks/, src/utils/
- NEVER include: package.json, index.html, src/main.jsx, vite.config.js
- NEVER include ANY .css file — styling is inline style={{}} ONLY
- NEVER plan files named: Icon.jsx, Card.jsx, Button.jsx, Badge.jsx, index.jsx
- NEVER use TypeScript (.ts/.tsx)
- ONLY available packages: react@18, react-dom@18
- BANNED packages: framer-motion, lucide-react, react-router-dom, axios, three, zustand, recharts, gsap, styled-components, next, clsx
- dependencies array = sibling file paths this component imports from
- Filenames MUST NOT start with a digit
- RESPOND WITH ONLY THE JSON OBJECT — nothing else before or after`;

async function planProject(
  prompt:  string,
  onPing?: () => void,
): Promise<ProjectSpec | null> {
  const models = planHealth.getHealthy(PLAN_MODELS);
  if (models.length === 0) {
    perr("[S1] All plan models unhealthy");
    return null;
  }

  for (let mi = 0; mi < models.length; mi++) {
    const model = models[mi];
    plog(`[S1] Planning with ${mlabel(model)} (${mi + 1}/${models.length})`);

    await rateLimiter.wait(`plan/${mlabel(model)}`, onPing);
    onPing?.();

    const abortCtrl = new AbortController();
    const timeout   = setTimeout(() => abortCtrl.abort(), 120_000);

    try {
      const res = await nvidia.chat.completions.create(
        {
          model,
          temperature: 0.3,
          max_tokens:  PLAN_MAX_TOKENS,
          messages: [
            { role: "system",  content: PLANNER_SYSTEM },
            { role: "user",    content: `User request: ${prompt}\n\nPlan a complete, production-quality React website for this request. Return ONLY the JSON object.` },
          ],
        },
        { signal: abortCtrl.signal },
      );
      clearTimeout(timeout);
      planHealth.recordSuccess(model);

      const raw  = fixTokenCorruption(res.choices[0]?.message?.content ?? "");
      const text = stripMarkdown(raw);

      // Find the JSON object boundaries robustly
      const start = text.indexOf("{");
      const end   = text.lastIndexOf("}");
      if (start === -1 || end === -1) {
        throw new Error(`No JSON object found in planner response (model: ${mlabel(model)})`);
      }

      const parsed = JSON.parse(text.slice(start, end + 1)) as ProjectSpec;

      // Sanitise file list
      parsed.files = (parsed.files ?? [])
        .map((f) => ({
          path:         f.path.replace(/^\/+/, ""),
          description:  f.description || f.path,
          dependencies: Array.isArray(f.dependencies) ? f.dependencies : [],
        }))
        // Drop boilerplate files the planner hallucinated
        .filter((f) => !BOILERPLATE_PATHS.has(f.path))
        // Drop CSS files (planner should not plan them, but guard anyway)
        .filter((f) => !isCssFile(f.path))
        // Drop files whose names start with a digit
        .filter((f) => !/^[0-9]/.test(f.path.split("/").pop() ?? ""))
        // Drop TypeScript files
        .filter((f) => !f.path.endsWith(".ts") && !f.path.endsWith(".tsx"));

      // Ensure App.jsx is always present
      if (!parsed.files.some((f) => f.path === "src/App.jsx")) {
        parsed.files.push({
          path:         "src/App.jsx",
          description:  "Root app component — imports and composes all sections and components",
          dependencies: parsed.files.filter((f) => f.path.endsWith(".jsx")).map((f) => f.path),
        });
      }

      // Ensure App.jsx is last (it depends on everything else)
      parsed.files = [
        ...parsed.files.filter((f) => f.path !== "src/App.jsx"),
        ...parsed.files.filter((f) => f.path === "src/App.jsx"),
      ];

      plog(`[S1] ✓ Planned ${parsed.files.length} files — "${parsed.projectName}"`);
      return parsed;

    } catch (e: unknown) {
      clearTimeout(timeout);
      planHealth.recordFailure(model);
      const status = (e as { status?: number })?.status ?? 0;
      if (status === 401 || status === 403) {
        perr(`[S1] Auth error — cannot continue`);
        return null;
      }
      perr(`[S1] ${mlabel(model)} failed: ${String(e)}`);
      if (mi < models.length - 1) {
        plog(`[S1] Trying next model...`);
        await sleep(1000);
      }
    }
  }

  perr("[S1] All plan models failed");
  return null;
}

// SECTION 11 — FILE PRIORITY SORTER
// ═══════════════════════════════════════════════════════════════════════════════

const PRIORITY_ORDER = [
  "src/utils/",
  "src/hooks/",
  "src/data/",
  "src/lib/",
  "src/layouts/",
  "src/components/",
  "src/sections/",
  "src/features/",
  "src/pages/",
  "src/views/",
  "src/App.jsx",
];

function sortByPriority(files: FileSpec[]): FileSpec[] {
  // Always generate App.jsx last
  const appFile  = files.filter((f) => f.path === "src/App.jsx");
  const rest     = files.filter((f) => f.path !== "src/App.jsx");

  return [
    ...rest.sort((a, b) => {
      const rank = (p: string) => {
        const i = PRIORITY_ORDER.findIndex((prefix) => p.startsWith(prefix) || p === prefix);
        return i === -1 ? PRIORITY_ORDER.length : i;
      };
      return rank(a.path) - rank(b.path);
    }),
    ...appFile,
  ];
}

/**
 * Build a well-formed App.jsx that imports and renders all generated components.
 * This runs AFTER all other files are generated so we know what actually exists.
 */
function buildAppJsx(spec: ProjectSpec, generatedFiles: StreamFile[]): string {
  // Only import renderable JSX files that are not App itself and not utility/hook/data files
  const renderableFiles = generatedFiles.filter((f) =>
    f.path.endsWith(".jsx") &&
    f.path !== "src/App.jsx" &&
    f.path !== "src/main.jsx" &&
    !/\/(hooks?|utils?|data|lib|types?)\//i.test(f.path) &&
    /\bexport\s+default\b/.test(f.content)
  );

  // Sort: layouts first, then sections, then components, then pages
  const ORDER = ["layout", "section", "component", "page", "feature", "view"];
  renderableFiles.sort((a, b) => {
    const rank = (p: string) => {
      const i = ORDER.findIndex((s) => p.toLowerCase().includes(s));
      return i === -1 ? ORDER.length : i;
    };
    return rank(a.path) - rank(b.path);
  });

  const components = renderableFiles.map((f) => {
    const name    = f.path.split("/").pop()!.replace(/\.jsx$/, "");
    const relPath = "./" + f.path.replace(/^src\//, "").replace(/\.jsx$/, "") + ".jsx";
    return { name, from: relPath };
  });

  const imports = [
    "import React from 'react';",
    ...components.map((c) => `import ${c.name} from '${c.from}';`),
  ].join("\n");

  const renders = components.length > 0
    ? components.map((c) => `      <${c.name} />`).join("\n")
    : `      <div style={{ padding: '5rem 2rem', textAlign: 'center', color: '#f1f5f9' }}>
        <h1 style={{ fontSize: '3rem', fontWeight: 800, margin: '0 0 1rem' }}>${spec.projectName}</h1>
        <p style={{ color: '#64748b', fontSize: '1.1rem', margin: 0 }}>${spec.description}</p>
      </div>`;

  return cleanOutput(`${imports}

function App() {
  return (
    <div style={{
      minHeight: '100vh',
      background: '#050508',
      fontFamily: 'Inter, system-ui, -apple-system, sans-serif',
    }}>
${renders}
    </div>
  );
}

export default App;`);
}

/**

   ${BANNED_PACKAGES.join(", ")}

/**
 * Build the per-file user prompt, injecting dependency context.
 */
function makeFileUserPrompt(
  fileSpec:       FileSpec,
  spec:           ProjectSpec,
  generatedSoFar: StreamFile[],
): string {
  const genMap = new Map(generatedSoFar.map((f) => [f.path, f.content]));

  // Inject dependency file content (first 60 lines each) for context
  const depContext = (fileSpec.dependencies ?? [])
    .map((dep) => {
      const content = genMap.get(dep);
      if (!content) return null;
      const preview = content.split("\n").slice(0, 60).join("\n");
      return `// ── DEPENDENCY: ${dep} ──\n${preview}`;
    })
    .filter(Boolean)
    .join("\n\n");

  const isDataFile = /\/(utils?|data|lib)\//i.test(fileSpec.path);
  const isHookFile = /\/hooks?\//i.test(fileSpec.path);
  const isAppFile  = fileSpec.path === "src/App.jsx";

  let instructions = "";

  if (isDataFile || isHookFile) {
    instructions = `Write a complete JavaScript utility/hook file.
Export named OR default exports.
NO JSX. Pure JavaScript only.
Use real, realistic data (at least 4-6 items if it is a data file).`;
  } else if (isAppFile) {
    instructions = `Write the root App component.
Import and render each component listed in the ALREADY GENERATED section above.
Use a vertical stack layout (flexbox column).
Background: #050508. Full viewport height.`;
  } else {
    instructions = `Write a complete, production-quality React component.
Make it visually impressive with rich inline styles.
Use real content — no lorem ipsum.
Implement real interactivity where appropriate (hover states, toggles, etc.) using useState.
Use CSS keyframes via a <style> tag in JSX only if animation is needed (no external animation libraries).
The component must look like it belongs on a premium, modern website.`;
  }

  return `Write the COMPLETE file: ${fileSpec.path}
Purpose: ${fileSpec.description}

${instructions}

${depContext ? `━━━ DEPENDENCY CONTEXT ━━━\n${depContext}\n` : ""}
Start the file content NOW. First character must be the first character of the file.
Do NOT write any explanation. Do NOT use markdown fences.`;
}

const _regenCount = new Map<string, number>();

/**
 * Generate a single file using the CODEGEN_MODELS roster.
 * Returns the best result or a safe fallback — NEVER throws.
 */
async function generateFile(
  fileSpec:       FileSpec,
  spec:           ProjectSpec,
  generatedSoFar: StreamFile[],
  prompt:         string,
  onPing?:        () => void,
): Promise<StreamFile> {
  // Never generate CSS files
  if (isCssFile(fileSpec.path)) {
    plog(`[S2] Skipping CSS file: ${fileSpec.path}`);
    return makeSafeComponent({ path: fileSpec.path.replace(/\.css$/, ".jsx"), content: "" });
  }

  // Regen guard — if we've tried too many times, use safe fallback
  const regensSoFar = _regenCount.get(fileSpec.path) ?? 0;
  if (regensSoFar >= MAX_REGEN_PER_FILE) {
    plog(`[S2] ${fileSpec.path} exceeded regen limit — safe fallback`);
    return makeSafeComponent({ path: fileSpec.path, content: "" });
  }

  const models = codeHealth.getHealthy(CODEGEN_MODELS);
  if (models.length === 0) {
    perr(`[S2] All codegen models unhealthy for ${fileSpec.path}`);
    return makeSafeComponent({ path: fileSpec.path, content: "" });
  }

  const system     = makeCodegenSystemPrompt(spec, generatedSoFar);
  const userPrompt = makeFileUserPrompt(fileSpec, spec, generatedSoFar);

  for (let mi = 0; mi < models.length; mi++) {
    const model = models[mi];

    for (let attempt = 0; attempt < MAX_FILE_RETRIES; attempt++) {
      await rateLimiter.wait(`${mlabel(model)}/${fileSpec.path}`, onPing);
      onPing?.();

      const abortCtrl = new AbortController();
      const timeout   = setTimeout(() => abortCtrl.abort(), 90_000);
      const t0        = Date.now();

      try {
        // Temperature: slightly higher on retries to escape local minima
        const temperature = 0.2 + attempt * 0.08;

        plog(`[S2] ${mlabel(model)} → ${fileSpec.path} (attempt ${attempt + 1}, t=${temperature.toFixed(2)})`);

        const res = await nvidia.chat.completions.create(
          {
            model,
            temperature,
            max_tokens: CODEGEN_MAX_TOKENS,
            messages: [
              { role: "system", content: system     },
              { role: "user",   content: userPrompt },
            ],
          },
          { signal: abortCtrl.signal },
        );

        clearTimeout(timeout);
        codeHealth.recordSuccess(model);

        const latencyMs = Date.now() - t0;
        const raw       = fixTokenCorruption(res.choices[0]?.message?.content ?? "");

        // Strip markdown fences — models often wrap output even when told not to
        let content = stripMarkdown(raw);

        // Hard rejection: too short
        if (content.trim().length < MIN_FILE_CHARS) {
          throw new Error(`Output too short: ${content.trim().length} chars`);
        }

        // Hard rejection: token corruption detected
        if (/=\s*\/\s*>/.test(content) || /<REDACTED/.test(content)) {
          throw new Error("Token corruption detected in output");
        }

        // Strip any banned imports that slipped through
        content = stripBannedImports(content);

        // JSX-specific normalization
        if (fileSpec.path.endsWith(".jsx")) {
          content = ensureReactImport(content);
          const compName = fileSpec.path.split("/").pop()!.replace(".jsx", "");
          content = ensureDefaultExport(content, compName);
        }

        // Deduplicate imports
        content = deduplicateImports(content);

        // Truncate if unreasonably large
        if (content.length > MAX_FILE_CHARS) {
          const trunc   = content.slice(0, MAX_FILE_CHARS);
          const lastBrace = trunc.lastIndexOf("\n}");
          content = lastBrace > MAX_FILE_CHARS * 0.6 ? trunc.slice(0, lastBrace + 2) : trunc;
          // Re-ensure export default after truncation
          if (fileSpec.path.endsWith(".jsx")) {
            const compName = fileSpec.path.split("/").pop()!.replace(".jsx", "");
            content = ensureDefaultExport(content, compName);
          }
        }

        content = cleanOutput(content);

        // Validate the result
        const v = validateFile({ path: fileSpec.path, content });
        if (!v.passed) {
          plog(`[S2] ${mlabel(model)} ${fileSpec.path} validation failed: ${v.reasons.join("; ")}`);
          // For auto-fixable issues (missing React import, missing export default), fix and accept
          if (v.reasons.every((r) => r.includes("missing export default") || r.includes("missing React"))) {
            content = ensureReactImport(content);
            content = ensureDefaultExport(content, fileSpec.path.split("/").pop()!.replace(".jsx", ""));
            plog(`[S2] Auto-fixed: ${fileSpec.path}`);
          } else {
            // Structural failure — retry
            if (attempt < MAX_FILE_RETRIES - 1) {
              await sleepPing(backoff(attempt), onPing);
              continue;
            }
            break; // Try next model
          }
        }

        _regenCount.delete(fileSpec.path);
        plog(`[S2] ✓ ${fileSpec.path} (${content.length}c, ${latencyMs}ms)`);
        return { path: fileSpec.path, content };

      } catch (e: unknown) {
        clearTimeout(timeout);
        codeHealth.recordFailure(model);

        const status = (e as { status?: number })?.status ?? 0;
        const msg    = (e as Error)?.message ?? String(e);

        if (status === 401 || status === 403) {
          perr(`[S2] Auth error — cannot continue`);
          return makeSafeComponent({ path: fileSpec.path, content: "" });
        }

        plog(`[S2] ${mlabel(model)} ${fileSpec.path} attempt ${attempt + 1} failed: ${msg}`);

        if (attempt < MAX_FILE_RETRIES - 1) {
          await sleepPing(backoff(attempt), onPing);
        }
      }
    }

    plog(`[S2] ${mlabel(model)} exhausted for ${fileSpec.path} — trying next model`);
  }

  perr(`[S2] All models failed for ${fileSpec.path} — safe fallback`);
  _regenCount.set(fileSpec.path, regensSoFar + 1);
  return makeSafeComponent({ path: fileSpec.path, content: "" });
}

/**
 * Generate all files in priority order, with PARALLEL_BATCH_SIZE concurrency.
 * App.jsx is always generated last after all other files exist for context.
 */
async function generateAllFiles(
  spec:      ProjectSpec,
  prompt:    string,
  smartEmit: (f: StreamFile) => Promise<void>,
  onPing?:   () => void,
): Promise<StreamFile[]> {
  const sortedSpecs = sortByPriority(spec.files);
  const nonAppSpecs = sortedSpecs.filter((f) => f.path !== "src/App.jsx");
  const appSpec     = sortedSpecs.find((f) => f.path === "src/App.jsx");

  const generated: StreamFile[] = [];

  // Generate all non-App files in batches
  for (let i = 0; i < nonAppSpecs.length; i += PARALLEL_BATCH_SIZE) {
    const batch = nonAppSpecs.slice(i, i + PARALLEL_BATCH_SIZE);
    plog(`[S2] Batch ${Math.floor(i / PARALLEL_BATCH_SIZE) + 1}: files ${i + 1}–${Math.min(i + PARALLEL_BATCH_SIZE, nonAppSpecs.length)} of ${nonAppSpecs.length}`);

    const results = await Promise.all(
      batch.map((fs) => generateFile(fs, spec, generated, prompt, onPing))
    );

    for (const file of results) {
      generated.push(file);
      await smartEmit({ ...file, provisional: true });
    }
  }

  // Generate App.jsx last — now it can see all generated files
  if (appSpec) {
    plog(`[S2] Generating src/App.jsx (final — has full context of ${generated.length} files)`);
    // First: build a deterministic App.jsx so the user sees something fast
    const deterministicApp = buildAppJsx(spec, generated);
    const deterministicFile: StreamFile = { path: "src/App.jsx", content: deterministicApp };
    generated.push(deterministicFile);
    await smartEmit({ ...deterministicFile, provisional: true });
  }

  plog(`[S2] ✓ Generated ${generated.length} files`);
  return generated;
}

/**
 * Try to auto-fix JSX tag imbalance by wrapping the return block in a <div>.
 * This handles the most common case: missing outer wrapper.
 */
function fixJsxImbalance(content: string): string {
  const openTags  = (content.match(/<[A-Z][a-zA-Z0-9]*[\s>]/g) ?? []).length;
  const closeTags = (content.match(/<\/[A-Z][a-zA-Z0-9]*>/g) ?? []).length;
  const selfClose = (content.match(/<[A-Z][a-zA-Z0-9]*[^>]*\/>/g) ?? []).length;
  const delta     = openTags - closeTags - selfClose;

  if (Math.abs(delta) <= JSX_IMBALANCE_TOL) return content;

  // Wrap the return body in a fragment
  const returnMatch = content.match(/(\breturn\s*\()([\s\S]*?)(\)\s*;?\s*\n)/);
  if (!returnMatch) return content;

  const [full, open, body, close] = returnMatch;
  const wrapped = `${open}\n    <>\n${body}\n    </>\n  ${close}`;
  return content.replace(full, wrapped);
}

/**
 * Fix useEffect calls missing dependency arrays.
 * Pattern: useEffect(() => { ... }) → useEffect(() => { ... }, [])
 */
function fixUseEffectDeps(content: string): string {
  return content
    .replace(
      /\buseEffect\s*\(\s*(\(\s*\)\s*=>\s*\{(?:[^{}]|\{[^{}]*\})*\})\s*\)(?!\s*,)/g,
      (_, fn) => `useEffect(${fn}, [])`
    )
    .replace(
      /\buseEffect\s*\(\s*(function\s*\(\s*\)\s*\{(?:[^{}]|\{[^{}]*\})*\})\s*\)(?!\s*,)/g,
      (_, fn) => `useEffect(${fn}, [])`
    );
}

/**
 * Run all deterministic fixes on a collection of files.
 * These are safe transformations that cannot make a valid file invalid.
 */
function deterministicFixPass(files: StreamFile[]): StreamFile[] {
  return files.map((file) => {
    if (BOILERPLATE_PATHS.has(file.path)) return file;
    if (isCssFile(file.path)) return file;
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js")) return file;

    let content = file.content;

    // 1. Strip any banned imports that survived generation
    content = stripBannedImports(content);

    // 2. Ensure React import for JSX files
    if (file.path.endsWith(".jsx")) {
      content = ensureReactImport(content);
    }

    // 3. Ensure export default for JSX files
    if (file.path.endsWith(".jsx")) {
      const compName = file.path.split("/").pop()!.replace(".jsx", "");
      content = ensureDefaultExport(content, compName);
    }

    // 4. Fix multiple export defaults (keep only the last one)
    const exportDefaultMatches = [...content.matchAll(/\bexport\s+default\b/g)];
    if (exportDefaultMatches.length > 1) {
      // Remove all but the last export default
      let count = 0;
      content = content.replace(/\bexport\s+default\b/g, (match) => {
        count++;
        return count < exportDefaultMatches.length ? "" : match;
      });
    }

    // 5. Fix useEffect missing dep arrays
    content = fixUseEffectDeps(content);

    // 6. Try to fix JSX imbalance
    if (file.path.endsWith(".jsx")) {
      content = fixJsxImbalance(content);
    }

    // 7. Deduplicate imports
    content = deduplicateImports(content);

    // 8. Fix token corruption
    content = fixTokenCorruption(content);

    // 9. Clean output
    content = cleanOutput(content);

    return { ...file, content };
  });
}

/**
 * Emit a file only if its content has changed since last emit.
 * Applies final validation before emit — bad files are safe-replaced.
 */
async function smartEmitFile(
  file:           StreamFile,
  onFile:         (f: StreamFile) => Promise<void>,
  emittedContent: Map<string, string>,
): Promise<void> {
  if (isCssFile(file.path)) {
    plog(`[EMIT] CSS file dropped: ${file.path}`);
    return;
  }

  let fileToEmit = file;

  // Final validation before emit
  if (!BOILERPLATE_PATHS.has(file.path) && file.path.endsWith(".jsx")) {
    const v = validateFile(file);
    if (!v.passed) {
      plog(`[EMIT] ${file.path} invalid at emit gate: ${v.reasons.join("; ")} → safe-replacing`);
      fileToEmit = makeSafeComponent(file);
    }
  }

  const prev = emittedContent.get(fileToEmit.path);
  if (prev === fileToEmit.content) return;

  emittedContent.set(fileToEmit.path, fileToEmit.content);
  try {
    await onFile(fileToEmit);
  } catch (e) {
    perr(`[EMIT] onFile threw for ${fileToEmit.path}: ${String(e)}`);
  }
}

export async function streamProjectPipeline(
  prompt:    string,
  callbacks: PipelineCallbacks,
): Promise<void> {
  const { onFile, onDone, onError, onLog, onPing } = callbacks;

  _log = (msg: string) => { console.log(msg); onLog?.(msg); };

  _regenCount.clear();

  const emittedContent = new Map<string, string>();

  const smartEmit = (f: StreamFile) =>
    smartEmitFile(f, onFile, emittedContent);

  const safeDone  = () => { try { onDone();        } catch {} };
  const safeError = (m: string) => { try { onError?.(m); } catch {} };

  plog("🚀 ══════════════════════════════════════════════════════════");
  plog("🚀 PIPELINE v19.0 — File-by-File · New Model Roster · Hard Validation");
  plog(`🚀 Prompt: "${prompt.slice(0, 80)}"`);
  plog(`🚀 Plan models:    ${PLAN_MODELS.map(mlabel).join(", ")}`);
  plog(`🚀 Codegen models: ${CODEGEN_MODELS.map(mlabel).join(", ")}`);
  plog("🚀 ══════════════════════════════════════════════════════════");

  // ── Stage 0: Emit boilerplate immediately ───────────────────────────────────
  plog("[S0] Emitting boilerplate...");
  try {
    for (const f of makeBoilerplate(prompt)) {
      await smartEmit(f);
    }
    plog("[S0] ✓ Boilerplate emitted — WebContainer can boot");
  } catch (e) {
    perr(`[S0] Boilerplate emit failed: ${String(e)}`);
    // Non-fatal — continue
  }

  // ── Stage 1: Plan project structure ─────────────────────────────────────────
  plog("[S1] Planning project...");
  let spec: ProjectSpec | null = null;
  try {
    spec = await planProject(prompt, onPing);
  } catch (e) {
    perr(`[S1] Planning threw: ${String(e)}`);
  }

  if (!spec || spec.files.length === 0) {
    perr("[S1] Planning failed — emitting minimal fallback");
    const fallback = makeSpinnerApp(prompt).replace(
      "Generating your app",
      "Generation failed — please retry"
    );
    await smartEmit({ path: "src/App.jsx", content: fallback, provisional: false });
    safeError("Planning failed");
    safeDone();
    return;
  }

  plog(`[S1] ✓ ${spec.files.length} files planned`);

  // ── Stage 2: Generate all files ─────────────────────────────────────────────
  plog(`[S2] Generating ${spec.files.length} files (batch size: ${PARALLEL_BATCH_SIZE})...`);
  let generatedFiles: StreamFile[] = [];
  try {
    generatedFiles = await generateAllFiles(spec, prompt, smartEmit, onPing);
    plog(`[S2] ✓ ${generatedFiles.length} files generated`);
  } catch (e) {
    perr(`[S2] Generation error: ${String(e)}`);
    safeError(`Generation failed: ${String(e)}`);
    safeDone();
    return;
  }

  // ── Stage 3: Deterministic fix pass ─────────────────────────────────────────
  plog(`[S3] Running deterministic fix pass on ${generatedFiles.length} files...`);
  let fixedFiles = deterministicFixPass(generatedFiles);
  plog("[S3] ✓ Deterministic fixes applied");

  // ── Stage 3.5: Hard validation gate ─────────────────────────────────────────
  plog("[S3.5] Applying hard validation gate...");
  fixedFiles = applyValidationGate(fixedFiles, "POST-FIX");
  plog(`[S3.5] ✓ ${fixedFiles.length} files passed gate`);

  // ── Stage 4: Emit final files ─────────────────────────────────────────────
  plog(`[S4] Emitting ${fixedFiles.length} final files...`);
  let emitCount = 0;
  for (const file of fixedFiles) {
    const prev = emittedContent.get(file.path);
    if (prev !== file.content) {
      await smartEmit({ ...file, provisional: false, lastFixed: true });
      emitCount++;
    }
  }
  plog(`[S4] ✓ Emitted ${emitCount} updated files`);

  plog("✅ ══════════════════════════════════════════════════════════");
  plog(`✅ PIPELINE v19.0 DONE — ${fixedFiles.length} files`);
  plog("✅ ══════════════════════════════════════════════════════════");

  safeDone();
}

// src/lib/ai/engine/ai-orchestrator.ts

import type {
  AIFileNode,
  AIProject,
  AIProjectPlan,
} from "../types/ai-project.types";

import { AIProjectError } from "../types/ai-project.types";
import { CodeGenerator } from "./code-generator";
import { CodePatcher } from "./code-patcher";

// 🔥 GLOBAL STORE (safe for HMR)
const globalStore = globalThis as unknown as {
  _aiFileStore?: Map<string, AIFileNode[]>;
  _aiMetaStore?: Map<string, AIProject>;
};

if (!globalStore._aiFileStore) {
  globalStore._aiFileStore = new Map<string, AIFileNode[]>();
}

if (!globalStore._aiMetaStore) {
  globalStore._aiMetaStore = new Map<string, AIProject>();
}

const inMemoryProjects = globalStore._aiFileStore;
const projectMetadata = globalStore._aiMetaStore;

export class AIOrchestrator {
  // 🚀 CREATE PROJECT (fallback only)
  static createProject(
    projectId: string,
    prompt: string,
    plan?: AIProjectPlan
  ): AIFileNode[] {
    try {
      let files: AIFileNode[];

      if (plan) {
        files = CodeGenerator.generateFromPlan(projectId, plan);
      } else {
        files = CodeGenerator.generate(projectId, prompt);
      }

      inMemoryProjects.set(projectId, files);

      projectMetadata.set(projectId, {
        id: projectId,
        name: plan?.projectName || prompt.slice(0, 50),
        subaccountId: "",
        files,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });

      return files;
    } catch (error: unknown) {
      const message =
        error instanceof Error ? error.message : String(error);

      throw new AIProjectError(
        `Failed to create project: ${message}`,
        "CREATE_FAILED",
        500
      );
    }
  }

  // 🔥 MAIN ENTRY USED BY API
  static setProjectFiles(projectId: string, files: AIFileNode[]) {
    if (!Array.isArray(files) || files.length === 0) {
      throw new AIProjectError(
        "Invalid files provided",
        "INVALID_FILES",
        400
      );
    }

    inMemoryProjects.set(projectId, files);

    projectMetadata.set(projectId, {
      id: projectId,
      name: "AI Project",
      subaccountId: "",
      files,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
    });

    console.log(`✅ Project ${projectId} stored (${files.length} files)`);
  }

  // ✏️ EDIT
  static async editProject(
    projectId: string,
    instructions: string,
    useAI: boolean = false,
    aiClient?: any
  ): Promise<AIFileNode[]> {
    try {
      const existing = inMemoryProjects.get(projectId);

      if (!existing) {
        throw new AIProjectError(
          `Project ${projectId} not found`,
          "PROJECT_NOT_FOUND",
          404
        );
      }

      let updated: AIFileNode[];

      if (useAI && aiClient) {
        updated = await CodePatcher.patchWithAI(
          existing,
          instructions,
          aiClient
        );
      } else {
        updated = CodePatcher.patch(existing, instructions);
      }

      inMemoryProjects.set(projectId, updated);

      const meta = projectMetadata.get(projectId);
      if (meta) {
        meta.updatedAt = new Date().toISOString();
        meta.files = updated;
        projectMetadata.set(projectId, meta);
      }

      return updated;
    } catch (error: unknown) {
      if (error instanceof AIProjectError) throw error;

      const message =
        error instanceof Error ? error.message : String(error);

      throw new AIProjectError(
        `Failed to edit project: ${message}`,
        "EDIT_FAILED",
        500
      );
    }
  }

  // 🔄 MERGE FILES
  static updateProjectFiles(
    projectId: string,
    newFiles: AIFileNode[]
  ): AIFileNode[] {
    const existing = inMemoryProjects.get(projectId);

    if (!existing) {
      throw new AIProjectError(
        `Project ${projectId} not found`,
        "PROJECT_NOT_FOUND",
        404
      );
    }

    const map = new Map<string, AIFileNode>();

    existing.forEach((f) => map.set(f.path, f));
    newFiles.forEach((f) => {
      if (f.path && typeof f.content === "string") {
        map.set(f.path, f);
      }
    });

    const updated = Array.from(map.values());

    inMemoryProjects.set(projectId, updated);

    const meta = projectMetadata.get(projectId);
    if (meta) {
      meta.updatedAt = new Date().toISOString();
      meta.files = updated;
      projectMetadata.set(projectId, meta);
    }

    return updated;
  }

  // 📂 GET FILES
  static getProjectFiles(projectId: string): AIFileNode[] {
    const files = inMemoryProjects.get(projectId);

    if (!files) {
      throw new AIProjectError(
        `Project ${projectId} not found`,
        "PROJECT_NOT_FOUND",
        404
      );
    }

    return files;
  }

  // 📊 META
  static getProjectMetadata(projectId: string): AIProject | undefined {
    return projectMetadata.get(projectId);
  }

  static getAllProjects(): AIProject[] {
    return Array.from(projectMetadata.values());
  }

  // ❌ DELETE
  static deleteProject(projectId: string): boolean {
    projectMetadata.delete(projectId);
    return inMemoryProjects.delete(projectId);
  }

  // 📈 STATS
  static getProjectStats(projectId: string) {
    const files = this.getProjectFiles(projectId);

    const fileTypes: Record<string, number> = {};
    let totalSize = 0;

    files.forEach((file) => {
      const ext = file.path.split(".").pop() || "unknown";
      fileTypes[ext] = (fileTypes[ext] || 0) + 1;
      totalSize += new Blob([file.content]).size;
    });

    return {
      fileCount: files.length,
      totalSize,
      fileTypes,
    };
  }
}

export function validateGeneratedFile(file: FileContent): ValidationResult {
  const errors: string[]   = [];
  const warnings: string[] = [];
  const c = file.content;
  const p = file.path;

  if (!c || c.length < 20) {
    return { valid: false, errors: ['File content is empty or too short'], warnings };
  }

  // REJECT CSS FILES — pipeline does not allow CSS
  if (p.endsWith('.css')) {
    return { valid: false, errors: ['CSS files are not allowed'], warnings: [] };
  }

  if (p.endsWith('.jsx') || p.endsWith('.js')) {
    errors.push(...checkReactImport(c, p));
    errors.push(...checkExportDefault(c, p));
    errors.push(...checkJSXBalance(c, p));
    errors.push(...checkBraceBalance(c, p));
    errors.push(...checkDuplicateConsts(c, p));
    errors.push(...checkHookSyntax(c, p));
    errors.push(...checkProxyCorruption(c, p));
    errors.push(...checkRedactedTokens(c, p));
    errors.push(...checkNumericIdentifiers(c, p));
    errors.push(...checkBannedPackages(c, p));
    errors.push(...checkIfInJSX(c, p));
    errors.push(...checkMissingKeyProps(c, p));
    // Tailwind validations removed — now allowed
    errors.push(...checkCssImports(c, p));
    warnings.push(...checkAsyncUseEffect(c, p));
    warnings.push(...checkMissingDepsArray(c, p));
    warnings.push(...checkProseAfterExport(c, p));
    warnings.push(...checkTruncation(c, p));

    if (p.endsWith('.jsx')) {
      errors.push(...checkTypeScriptLeaks(c, p));
    }
  }

  if (p.endsWith('.json')) {
    errors.push(...checkJSON(c, p));
  }

  return { valid: errors.length === 0, errors, warnings };
}

function checkReactImport(c: string, p: string): string[] {
  if (!p.endsWith('.jsx')) return [];
  const hasImport = c.includes("from 'react'") || c.includes('from "react"');
  return hasImport ? [] : [`[${p}] Missing React import`];
}

function checkExportDefault(c: string, p: string): string[] {
  if (!p.endsWith('.jsx')) return [];
  const count = (c.match(/\bexport\s+default\b/g) ?? []).length;
  if (count === 0) return [`[${p}] Missing export default`];
  if (count > 1)  return [`[${p}] Multiple export default (${count}x) — only one allowed`];
  return [];
}

function checkJSXBalance(c: string, p: string): string[] {
  const TAGS = [
    'div','section','main','nav','header','footer','ul','ol','li',
    'span','p','h1','h2','h3','h4','h5','h6','form','article',
    'aside','table','thead','tbody','tr','td','th','button','a',
    'label','select','textarea','figure','figcaption',
  ];
  const errors: string[] = [];
  for (const tag of TAGS) {
    const open  = (c.match(new RegExp(`<${tag}[\\s>]`, 'g')) ?? []).length;
    const close = (c.match(new RegExp(`</${tag}>`, 'g')) ?? []).length;
    if (Math.abs(open - close) > 1) {
      errors.push(`[${p}] JSX tag imbalance <${tag}>: ${open} open, ${close} close`);
    }
  }
  return errors;
}

function checkBraceBalance(c: string, p: string): string[] {
  const stripped = c
    .replace(/\/\/[^\n]*/g, '')
    .replace(/\/\*[\s\S]*?\*\//g, '')
    .replace(/"(?:[^"\\]|\\.)*"/g, '""')
    .replace(/'(?:[^'\\]|\\.)*'/g, "''")
    .replace(/`(?:[^`\\]|\\.)*`/g, '``');
  const open  = (stripped.match(/\{/g) ?? []).length;
  const close = (stripped.match(/\}/g) ?? []).length;
  if (Math.abs(open - close) > 3) {
    return [`[${p}] Unbalanced braces: ${open} open, ${close} close`];
  }
  return [];
}

function checkDuplicateConsts(c: string, p: string): string[] {
  const counts = new Map<string, number>();
  for (const m of c.matchAll(/\bconst\s+([a-zA-Z_$][a-zA-Z0-9_$]*)\s*=/g)) {
    counts.set(m[1], (counts.get(m[1]) ?? 0) + 1);
  }
  return [...counts.entries()]
    .filter(([, n]) => n > 1)
    .map(([name, n]) => `[${p}] Duplicate const '${name}' declared ${n}x`);
}

function checkHookSyntax(c: string, p: string): string[] {
  const errors: string[] = [];

  if (/useMemo\s*\(\s*\([^)]*,\s*\[/.test(c)) {
    errors.push(
      `[${p}] CRITICAL: useMemo dep array is INSIDE the function argument. ` +
      `Fix: useMemo(() => value, [deps]) — two separate args.`
    );
  }
  if (/useCallback\s*\(\s*\([^)]*,\s*\[/.test(c)) {
    errors.push(
      `[${p}] CRITICAL: useCallback dep array is INSIDE the function argument. ` +
      `Fix: useCallback((x) => fn(x), [deps]) — two separate args.`
    );
  }
  if (/useMemo\s*\(\s*\)/.test(c)) {
    errors.push(`[${p}] useMemo called with no arguments`);
  }
  if (/useCallback\s*\(\s*\)/.test(c)) {
    errors.push(`[${p}] useCallback called with no arguments`);
  }
  return errors;
}

function checkProxyCorruption(c: string, p: string): string[] {
  return /=\s*\/\s*>/.test(c)
    ? [`[${p}] Proxy-corrupted arrow '= />' found — replace with '=>'`]
    : [];
}

function checkRedactedTokens(c: string, p: string): string[] {
  return /<REDACTED_EMAIL>|\[REDACTED(?:_EMAIL)?\]/.test(c)
    ? [`[${p}] Proxy redaction token found — replace with '' or null`]
    : [];
}

function checkNumericIdentifiers(c: string, p: string): string[] {
  const errors: string[] = [];
  if (/import\s+\d\w*\s+from/.test(c)) {
    errors.push(`[${p}] Import identifier starts with a digit (invalid JS)`);
  }
  const base = p.split('/').pop()?.split('.')[0] ?? '';
  if (/^\d/.test(base)) {
    errors.push(`[${p}] Filename starts with a digit (invalid JS identifier)`);
  }
  return errors;
}

function checkBannedPackages(c: string, p: string): string[] {
  const BANNED = [
    'react-router-dom','axios','three','@react-three/fiber','@react-three/drei',
    'gsap','react-intersection-observer','@tanstack/react-query','zustand',
    'recharts','preact','styled-components','@emotion/react','@emotion/styled',
    'classnames','clsx'
  ];
  return BANNED
    .filter((pkg) => c.includes(`'${pkg}'`) || c.includes(`"${pkg}"`))
    .map((pkg) => `[${p}] Banned package '${pkg}' (not installed)`);
}

function checkIfInJSX(c: string, p: string): string[] {
  return /\{[\s\n]*if\s*\(/.test(c)
    ? [`[${p}] if-statement inside JSX {} — use ternary or && operator instead`]
    : [];
}

function checkMissingKeyProps(c: string, p: string): string[] {
  if (/\.map\(\s*(?:\w+|\([^)]*\))\s*=>\s*\(?\s*<[A-Za-z][^>]*(?<!\bkey=\{[^}]*\})[^/]>/.test(c)) {
    return [`[${p}] .map() may render JSX without key prop`];
  }
  return [];
}

function checkCssImports(c: string, p: string): string[] {
  if (/import\s+['"`][^'"`]*\.css['"`]/.test(c)) {
    return [`[${p}] CSS import detected. CSS files are not allowed.`];
  }
  return [];
}

function checkTypeScriptLeaks(c: string, p: string): string[] {
  const errors: string[] = [];
  if (/:\s*React\.(FC|ReactNode|ReactElement|ComponentType)/.test(c)) {
    errors.push(`[${p}] TypeScript React type annotation in .jsx file`);
  }
  if (/\buseState<[A-Za-z]/.test(c) || /\buseRef<[A-Za-z]/.test(c)) {
    errors.push(`[${p}] TypeScript generic in .jsx (useState<T>/useRef<T>)`);
  }
  if (/\binterface\s+\w+\s*\{/.test(c)) {
    errors.push(`[${p}] TypeScript interface in .jsx file`);
  }
  return errors;
}

function checkAsyncUseEffect(c: string, p: string): string[] {
  return /useEffect\s*\(\s*async\s*(?:\(\s*\)|function)/.test(c)
    ? [`[${p}] WARNING: async useEffect — use IIFE: useEffect(() => { (async () => { ... })(); }, [])`]
    : [];
}

function checkMissingDepsArray(c: string, p: string): string[] {
  const warnings: string[] = [];
  if (/useMemo\s*\(/.test(c) && !/useMemo\s*\([^)]+\)\s*,\s*\[/.test(c)) {
    warnings.push(`[${p}] WARNING: useMemo may be missing dep array as second argument`);
  }
  if (/useEffect\s*\(/.test(c) && !/useEffect\s*\([^)]+\)\s*,\s*\[/.test(c)) {
    warnings.push(`[${p}] WARNING: useEffect may be missing dep array as second argument`);
  }
  return warnings;
}

function checkProseAfterExport(c: string, p: string): string[] {
  const pats = [
    /export\s+default\s+\w+\s*;\s*\n+\s*I\s+(have|added|removed)/i,
    /export\s+default\s+\w+\s*;\s*\n+\s*Note\s+that/i,
    /export\s+default\s+\w+\s*;\s*\n+\s*Here('s| is)/i,
  ];
  return pats.some((re) => re.test(c))
    ? [`[${p}] WARNING: LLM prose after export default`]
    : [];
}

function checkTruncation(c: string, p: string): string[] {
  if (!p.endsWith('.jsx')) return [];
  const open   = (c.match(/\(/g) ?? []).length;
  const close  = (c.match(/\)/g) ?? []).length;
  const openB  = (c.match(/\{/g) ?? []).length;
  const closeB = (c.match(/\}/g) ?? []).length;
  const w: string[] = [];
  if (open  - close  > 4) w.push(`[${p}] WARNING: ${open - close} unclosed parens — possible truncation`);
  if (openB - closeB > 5) w.push(`[${p}] WARNING: ${openB - closeB} unclosed braces — possible truncation`);
  return w;
}

function checkJSON(c: string, p: string): string[] {
  try { JSON.parse(c); return []; }
  catch (e) { return [`[${p}] Invalid JSON: ${String(e).slice(0, 80)}`]; }
}

  static validateBatch(files: FileContent[]): Map<string, ValidationResult> {
    const results = new Map<string, ValidationResult>();
    for (const file of files) results.set(file.path, validateGeneratedFile(file));
    return results;
  }

  static summariseValidation(results: Map<string, ValidationResult>): string {
    let valid = 0; let invalid = 0;
    const lines: string[] = [];
    for (const [path, result] of results) {
      if (result.valid) { valid++; }
      else {
        invalid++;
        lines.push(`  ❌ ${path}`);
        for (const err of result.errors) lines.push(`     • ${err}`);
      }
      for (const warn of result.warnings) lines.push(`  ⚠️  ${warn}`);
    }
    return [`Validation: ${valid} valid, ${invalid} invalid`, ...lines].join('\n');
  }
}

export function countJSXTags(
  content: string,
  tagName: string
): { open: number; close: number; balanced: boolean } {
  const open  = (content.match(new RegExp(`<${tagName}[\\s/>]`, 'g')) ?? []).length;
  const close = (content.match(new RegExp(`</${tagName}>`, 'g')) ?? []).length;
  return { open, close, balanced: Math.abs(open - close) <= 1 };
}

export function countBraces(content: string): { open: number; close: number; balanced: boolean } {
  const stripped = content
    .replace(/\/\/[^\n]*/g, '')
    .replace(/\/\*[\s\S]*?\*\//g, '')
    .replace(/"(?:[^"\\]|\\.)*"/g, '""')
    .replace(/'(?:[^'\\]|\\.)*'/g, "''")
    .replace(/`(?:[^`\\]|\\.)*`/g, '``');
  const open  = (stripped.match(/\{/g) ?? []).length;
  const close = (stripped.match(/\}/g) ?? []).length;
  return { open, close, balanced: Math.abs(open - close) <= 3 };
}

export function findUnusedImports(content: string): string[] {
  const unused: string[]  = [];
  const importLines       = content.split('\n').filter((l) => l.trim().startsWith('import '));
  const body              = content.replace(/^import[^\n]+$/gm, '');

  for (const line of importLines) {
    const namedMatch = line.match(/\{([^}]+)\}/);
    if (namedMatch) {
      for (const spec of namedMatch[1].split(',')) {
        const name = spec.trim().split(' as ').pop()?.trim() ?? '';
        if (name && name !== 'React' && !new RegExp(`\\b${name}\\b`).test(body)) unused.push(name);
      }
    }
    const defMatch = line.match(/import\s+([A-Za-z_$][A-Za-z0-9_$]*)\s+from/);
    if (defMatch && defMatch[1] !== 'React') {
      if (!new RegExp(`\\b${defMatch[1]}\\b`).test(body)) unused.push(defMatch[1]);
    }
  }
  return unused;
}

export function auditHookCallSites(content: string, filePath: string): string[] {
  const errors: string[] = [];

  if (/useMemo\s*\(\s*\([^)]*,\s*\[/.test(content)) {
    errors.push(
      `[${filePath}] useMemo: dep array is inside the first arg. ` +
      `Correct: useMemo(() => value, [deps])`
    );
  }
  if (/useCallback\s*\(\s*\([^)]*,\s*\[/.test(content)) {
    errors.push(
      `[${filePath}] useCallback: dep array is inside the first arg. ` +
      `Correct: useCallback((x) => fn(x), [deps])`
    );
  }
  if (/useMemo\s*\(\s*\(\s*\)\s*=>\s*[^,)]+\s*\)(?!\s*,)/.test(content)) {
    errors.push(`[${filePath}] useMemo missing dep array as second argument`);
  }
  if (/useEffect\s*\(\s*async\s/.test(content)) {
    errors.push(`[${filePath}] async useEffect — use IIFE pattern instead`);
  }
  return errors;
}

export function detectIdentifierTypos(content: string): { hasTypos: boolean; found: Array<[string, string]> } {
  const TYPO_MAP: Array<[RegExp, string]> = [
    [/\bhandlleScroll\b/g, 'handleScroll'],
    [/\bscrolllTo\b/g,     'scrollTo'],
    [/\buserState\b/g,     'useState'],
    [/\busestate\b/g,      'useState'],
    [/\buseEffekt\b/g,     'useEffect'],
    [/\buseCallbakc\b/g,   'useCallback'],
    [/\buseContxt\b/g,     'useContext'],
    [/\bclassname\b/g,     'className'],
    [/\bClassname\b/g,     'className'],
    [/\bonclick\b/g,       'onClick'],
    [/\bconsoloe\b/g,      'console'],
    [/\bfucntion\b/g,      'function'],
    [/\bretrun\b/g,        'return'],
    [/\bimprot\b/g,        'import'],
    [/\bdefualt\b/g,       'default'],
  ];
  const found: Array<[string, string]> = [];
  for (const [re, correct] of TYPO_MAP) {
    if (re.test(content)) found.push([re.source.replace(/\\b/g, ''), correct]);
  }
  return { hasTypos: found.length > 0, found };
}

export function findMissingComponentFiles(

    'React','Fragment','Suspense','StrictMode','ErrorBoundary',
    'Router','Route','Routes','Navigate','Link','Switch','Provider','Consumer','Canvas',

  return usedComponents.filter((name) => !defined.has(name) && !BUILTINS.has(name));

export function sanitiseComponentFilename(filename: string): string {
  const ext  = filename.includes('.') ? '.' + filename.split('.').pop() : '';
  const base = filename.slice(0, filename.length - ext.length);
  if (!/^\d/.test(base)) return filename;
  const m = base.match(/^(\d+)(.*)/);
  if (!m) return filename;
  const digits   = m[1];
  const rest     = m[2];
  const safeName = rest ? rest.charAt(0).toUpperCase() + rest.slice(1) + digits : `Component${digits}`;
  return safeName + ext;
}

export async function planProjectV2(
  prompt:  string,
  ds:      DesignSystem,
  onPing?: () => void,
): Promise<ProjectPlan | null> {
  modelFailures.clear();

  const models = getHealthyModels();
  if (models.length === 0) {
    console.error("[planProjectV2] All plan models unhealthy");
    return null;
  }

  for (let mi = 0; mi < models.length; mi++) {
    const model = models[mi];
    console.log(`[planProjectV2] Planning with ${mlabel(model)} (${mi + 1}/${models.length})`);

    const abortCtrl = new AbortController();
    const timeout   = setTimeout(() => abortCtrl.abort(), PLAN_TIMEOUT_MS);

    try {
      const res = await client.chat.completions.create(
        {
          model,
          temperature: 0.3,
          max_tokens:  PLAN_MAX_TOKENS,
          messages: [
            { role: "system", content: PLANNER_SYSTEM },
            {
              role:    "user",
              content: `Project request: ${prompt}

Design system context:
- Primary color: ${ds.colors.primary}
- Background: ${ds.colors.bg}
- Text color: ${ds.colors.text}
- Font: ${ds.fonts}

Plan a complete, production-quality React website. Return ONLY the JSON object.`,
            },
          ],
        },
        { signal: abortCtrl.signal },
      );

      clearTimeout(timeout);
      recordSuccess(model);

      const raw  = fixTokenCorruption(res.choices[0]?.message?.content ?? "");
      const text = stripMarkdown(raw);

      // Find JSON object boundaries — handles models that add preamble/postamble
      const start = text.indexOf("{");
      const end   = text.lastIndexOf("}");

      if (start === -1 || end === -1) {
        throw new Error(`No JSON object found in response (model: ${mlabel(model)})`);
      }

      const jsonStr = text.slice(start, end + 1);
      const parsed  = JSON.parse(jsonStr) as Omit<ProjectPlan, "designSystem">;

      // Sanitise component list
      let components = (parsed.components ?? []).map((c) => ({
        path:         (c.path ?? "").replace(/^\/+/, ""),
        name:         c.name || pathToName(c.path ?? ""),
        description:  c.description || c.path || "Component",
        props:        c.props || "{}",
        dependencies: Array.isArray(c.dependencies) ? c.dependencies : [],
      }));

      // Filter out invalid files
      components = components
        .filter((c) => c.path.startsWith("src/"))
        .filter((c) => !c.path.endsWith(".css") && !c.path.endsWith(".scss") && !c.path.endsWith(".sass") && !c.path.endsWith(".less"))
        .filter((c) => !c.path.endsWith(".ts") && !c.path.endsWith(".tsx"))
        .filter((c) => !startsWithDigit(c.path))
        .filter((c) => !["package.json", "index.html", "src/main.jsx", "vite.config.js"].includes(c.path));

      // Ensure App.jsx exists
      if (!components.some((c) => c.path === "src/App.jsx")) {
        components.push({
          path:         "src/App.jsx",
          name:         "App",
          description:  "Root component — imports and composes all sections and components",
          props:        "{}",
          dependencies: components.filter((c) => c.path.endsWith(".jsx")).map((c) => c.path),
        });
      }

      // App.jsx must be last (it depends on everything else)
      components = [
        ...components.filter((c) => c.path !== "src/App.jsx"),
        ...components.filter((c) => c.path === "src/App.jsx"),
      ];

      const plan: ProjectPlan = {
        projectName:  parsed.projectName  || "Project",
        description:  parsed.description  || prompt.slice(0, 100),
        appStructure: parsed.appStructure || "App.jsx renders all components in a vertical layout.",
        designSystem: ds,
        components,
      };

      console.log(`[planProjectV2] ✓ Planned ${components.length} components — "${plan.projectName}"`);
      return plan;

    } catch (e: unknown) {
      clearTimeout(timeout);
      recordFailure(model);

      const status = (e as { status?: number })?.status ?? 0;
      const msg    = (e as Error)?.message ?? String(e);

      if (status === 401 || status === 403) {
        console.error("[planProjectV2] Auth error — cannot continue");
        return null;
      }

      console.warn(`[planProjectV2] ${mlabel(model)} failed: ${msg}`);

      if (mi < models.length - 1) {
        console.log(`[planProjectV2] Trying next model...`);
        await sleepWithPing(1500, onPing);
      }
    }
  }

  console.error("[planProjectV2] All plan models failed");
  return null;
}

// ─── Model health tracking ────────────────────────────────────────────────────

const modelFailures = new Map<string, number>();
const FAILURE_THRESHOLD = 3;

function recordModelSuccess(model: string): void {
  modelFailures.delete(model);
}

function recordModelFailure(model: string): void {
  modelFailures.set(model, (modelFailures.get(model) ?? 0) + 1);
}

function isModelHealthy(model: string): boolean {
  return (modelFailures.get(model) ?? 0) < FAILURE_THRESHOLD;
}

function getHealthyModels(): CodegenModel[] {
  return CODEGEN_MODELS.filter(isModelHealthy);
}

// ─── Simple rate limiter ──────────────────────────────────────────────────────

const rlTimestamps: number[] = [];
const RL_MAX_RPM  = 5;
const RL_WINDOW   = 62_000;

async function waitForRateLimit(label: string, onPing?: () => void): Promise<void> {
  const now = Date.now();
  const trimmed = rlTimestamps.filter((t) => now - t < RL_WINDOW);
  rlTimestamps.length = 0;
  rlTimestamps.push(...trimmed);

  if (rlTimestamps.length < RL_MAX_RPM) {
    rlTimestamps.push(now);
    return;
  }

  const waitMs = RL_WINDOW - (now - rlTimestamps[0]) + 300;
  console.log(`[batch-generator][RL] Waiting ${(waitMs / 1000).toFixed(1)}s for ${label}`);
  await sleepWithPing(waitMs, onPing);
  await waitForRateLimit(label, onPing);
}

// ─── Utilities ────────────────────────────────────────────────────────────────

function sleep(ms: number): Promise<void> {
  return new Promise((r) => setTimeout(r, ms));
}

async function sleepWithPing(ms: number, onPing?: () => void): Promise<void> {
  if (!onPing || ms <= PING_INTERVAL_MS) return sleep(ms);
  const end = Date.now() + ms;
  while (Date.now() < end) {
    await sleep(Math.min(PING_INTERVAL_MS, end - Date.now()));
    if (Date.now() < end) onPing();
  }
}

function backoffMs(attempt: number): number {
  return Math.min(1200 * Math.pow(2, attempt), 20_000);
}

function mlabel(model: string): string {
  return model.split("/")[1] ?? model;
}

/**
 * Strip markdown fences. Models often wrap output in ```jsx even when instructed not to.
 */
function stripMarkdown(text: string): string {
  return text
    .replace(/^```(?:jsx?|javascript|typescript|tsx?|json)?\s*/gim, "")
    .replace(/^```\s*$/gim, "")
    .replace(/^["']use (?:client|server)["'];\s*\n*/m, "")
    .trim();
}

/**
 * Fix common token corruption from NVIDIA models.
 * e.g. ") = />" → ") =>" (arrow function token mangled)
 */
function fixTokenCorruption(code: string): string {
  let c = code;
  c = c.replace(/(\)|\])\s*=\s*\/\s*>/g, "$1 =>");
  c = c.replace(/(?<![=!<>])=\s*\/\s*>/g, "=>");
  c = c.replace(/\b(useState|useReducer)\s*\(\s*<REDACTED[^>]*>\s*\)/g, "$1(null)");
  c = c.replace(/=\s*\{\s*<REDACTED[^>]*>\s*\}/g, "={null}");
  c = c.replace(/\[REDACTED[^\]]*\]/g, "''");
  c = c.replace(/"<REDACTED[^>]*>"/g, "''");
  c = c.replace(/'<REDACTED[^>]*>'/g, "''");
  return c;
}

/**
 * Remove imports of banned packages from generated code.
 */
function stripBannedImports(content: string): string {
  const lines: string[] = [];

  for (const line of content.split("\n")) {
    const t = line.trim();
    if (!t.startsWith("import ")) { lines.push(line); continue; }

    // Always remove CSS imports
    if (/['"][^'"]*\.css['"]/.test(t)) continue;

    const fromMatch = t.match(/from\s+['"]([^'"]+)['"]/);
    if (!fromMatch) { lines.push(line); continue; }

    const pkg    = fromMatch[1];
    const banned = BANNED_PACKAGES.some((b) => pkg === b || pkg.startsWith(b + "/"));
    if (banned) continue;

    lines.push(line);
  }

  let result = lines.join("\n");

  // Convert react-router <Link> to <a> if it slipped through
  if (/<Link[\s>]/.test(result) || /<\/Link>/.test(result)) {
    result = result
      .replace(/<Link\s+to=(['"])/g, "<a href=$1")
      .replace(/<Link\s+href=(['"])/g, "<a href=$1")
      .replace(/<Link(\s[^>]*)?>/g, "<a$1>")
      .replace(/<\/Link>/g, "</a>");
  }

  return result;
}

/**
 * Inject React import if JSX is present but React is not imported.
 */
function ensureReactImport(content: string): string {
  if (/\bimport\s+React\b/.test(content)) return content;
  if (/<[A-Za-z]/.test(content)) return `import React from 'react';\n${content}`;
  return content;
}

/**
 * Ensure exactly one export default in a JSX file.
 */
function ensureDefaultExport(content: string, componentName: string): string {
  const matches = (content.match(/\bexport\s+default\b/g) ?? []).length;

  if (matches === 1) return content;

  if (matches === 0) {
    const safe = componentName.replace(/[^a-zA-Z0-9_$]/g, "").replace(/^\d+/, "") || "Component";
    return `${content}\n\nexport default ${safe};\n`;
  }

  // Multiple exports — remove all but last
  let count = 0;
  return content.replace(/\bexport\s+default\b/g, (match) => {
    count++;
    return count < matches ? "" : match;
  });
}

/**
 * Remove duplicate import statements (identical lines).
 */
function deduplicateImports(content: string): string {
  const seen = new Set<string>();
  return content.split("\n").filter((line) => {
    const t = line.trim();
    if (!t.startsWith("import ")) return true;
    if (seen.has(t)) return false;
    seen.add(t);
    return true;
  }).join("\n");
}

/**
 * Normalize whitespace and add trailing newline.
 */
function cleanOutput(content: string): string {
  return content
    .replace(/[ \t]+$/gm, "")
    .replace(/\n{4,}/g, "\n\n\n")
    .trimEnd() + "\n";
}

/**
 * Create a minimal safe fallback component for a file that failed generation.
 */
function makeFallback(spec: ComponentSpec): StreamFile {
  const rawName  = spec.path.split("/").pop()?.replace(/\.[^.]+$/, "") ?? "Component";
  const safeName = rawName.replace(/[^a-zA-Z0-9_$]/g, "").replace(/^\d+/, "") || "Component";
  const compName = safeName.charAt(0).toUpperCase() + safeName.slice(1);

  const content = cleanOutput(
    `import React from 'react';

function ${compName}() {
  return (
    <section style={{
      padding: '4rem 2rem',
      textAlign: 'center',
      background: '#0a0a0f',
      minHeight: '200px',
      display: 'flex',
      alignItems: 'center',
      justifyContent: 'center',
    }}>
      <div>
        <h2 style={{ fontSize: '1.75rem', fontWeight: 700, color: '#f1f5f9', margin: 0 }}>
          ${compName}
        </h2>
        <p style={{ color: '#64748b', marginTop: '0.75rem', fontSize: '0.95rem', margin: '0.75rem 0 0' }}>
          ${spec.description}
        </p>
      </div>
    </section>
  );
}

export default ${compName};`
  );

  return { path: spec.path, content };
}

function buildFileUserPrompt(
  spec:           ComponentSpec,
  plan:           ProjectPlan,
  generatedSoFar: StreamFile[],
): string {
  const genMap      = new Map(generatedSoFar.map((f) => [f.path, f.content]));
  const isDataFile  = /\/(utils?|data|lib)\//i.test(spec.path);
  const isHookFile  = /\/hooks?\//i.test(spec.path);
  const isAppFile   = spec.path === "src/App.jsx";

  // Inject dependency snippets (first 50 lines each)
  const depContext = (spec.dependencies ?? [])
    .map((dep) => {
      const content = genMap.get(dep);
      if (!content) return null;
      return `// ── DEP: ${dep} ──\n${content.split("\n").slice(0, 50).join("\n")}`;
    })
    .filter(Boolean)
    .join("\n\n");

  let task = "";

  if (isAppFile) {
    const renderables = generatedSoFar
      .filter((f) => f.path.endsWith(".jsx") && f.path !== "src/App.jsx" && !isDataFile && !isHookFile)
      .map((f) => {
        const name    = f.path.split("/").pop()!.replace(".jsx", "");
        const relPath = "./" + f.path.replace(/^src\//, "").replace(".jsx", "") + ".jsx";
        return { name, relPath };
      });
    const imports = renderables.map((c) => `import ${c.name} from '${c.relPath}';`).join("\n");
    const renders = renderables.map((c) => `      <${c.name} />`).join("\n");
    task = `Write src/App.jsx that composes all components.
Imports to use:
${imports}

Render them in order:
${renders}

Wrap in a <div> with style={{ minHeight: '100vh', background: '${plan.designSystem.colors.bg}', fontFamily: '${plan.designSystem.fonts}' }}.`;
  } else if (isDataFile || isHookFile) {
    task = `Write a complete JavaScript ${isHookFile ? "custom hook" : "utility/data"} file.
Export named OR default exports. NO JSX. Pure JavaScript only.
${isDataFile ? "Use realistic data — at least 5-8 items." : ""}`;
  } else {
    task = `Write a complete, production-quality React component.
Make it visually impressive with rich inline styles.
Use the project's color palette: primary=${plan.designSystem.colors.primary}, bg=${plan.designSystem.colors.bg}.
Use real, specific content — not placeholder text.
Add real interactivity with useState where it makes sense (hover states, toggles, counters).
The component must look premium and polished.`;
  }

  return `Write the COMPLETE file: ${spec.path}
Component name: ${spec.name}
Purpose: ${spec.description}
Props interface: ${spec.props}

${task}

${depContext ? `━━━ DEPENDENCY FILES ━━━\n${depContext}\n` : ""}
Start the file content NOW. No explanation, no markdown, no backticks.`;
}

/**
 * Generate a single file using the model roster.
 * Returns a StreamFile — either the generated content or a safe fallback.
 * NEVER throws.
 */
async function generateSingleFile(
  spec:           ComponentSpec,
  plan:           ProjectPlan,
  generatedSoFar: StreamFile[],
  onPing?:        () => void,
): Promise<StreamFile> {
  const models = getHealthyModels();
  if (models.length === 0) {
    console.warn("[batch-generator] All models unhealthy — using fallback");
    return makeFallback(spec);
  }

  const system     = buildSystemPrompt(plan, generatedSoFar);
  const userPrompt = buildFileUserPrompt(spec, plan, generatedSoFar);

  for (let mi = 0; mi < models.length; mi++) {
    const model = models[mi];

    for (let attempt = 0; attempt < MAX_FILE_RETRIES; attempt++) {
      await waitForRateLimit(`${mlabel(model)}/${spec.path}`, onPing);
      onPing?.();

      const abortCtrl = new AbortController();
      const timeout   = setTimeout(() => abortCtrl.abort(), 90_000);

      try {
        const temperature = 0.2 + attempt * 0.08; // Slightly higher temp on retries

        console.log(`[batch-generator] ${mlabel(model)} → ${spec.path} (attempt ${attempt + 1}, t=${temperature.toFixed(2)})`);

        const res = await client.chat.completions.create(
          {
            model,
            temperature,
            max_tokens: CODEGEN_MAX_TOKENS,
            messages: [
              { role: "system", content: system     },
              { role: "user",   content: userPrompt },
            ],
          },
          { signal: abortCtrl.signal },
        );

        clearTimeout(timeout);
        recordModelSuccess(model);

        const raw     = fixTokenCorruption(res.choices[0]?.message?.content ?? "");
        let   content = stripMarkdown(raw);

        // Hard reject: too short
        if (content.trim().length < MIN_FILE_CHARS) {
          throw new Error(`Output too short: ${content.trim().length} chars`);
        }

        // Hard reject: token corruption survived fix
        if (/=\s*\/\s*>/.test(content) || /<REDACTED/.test(content)) {
          throw new Error("Token corruption detected");
        }

        // Strip banned imports
        content = stripBannedImports(content);

        // Normalize JSX files
        if (spec.path.endsWith(".jsx")) {
          content = ensureReactImport(content);
          content = ensureDefaultExport(content, spec.name);
        }

        // Deduplicate imports
        content = deduplicateImports(content);

        // Truncate oversized files (LLM sometimes generates huge files)
        if (content.length > MAX_FILE_CHARS) {
          const trunc     = content.slice(0, MAX_FILE_CHARS);
          const lastBrace = trunc.lastIndexOf("\n}");
          content = lastBrace > MAX_FILE_CHARS * 0.6 ? trunc.slice(0, lastBrace + 2) : trunc;
          if (spec.path.endsWith(".jsx")) {
            content = ensureDefaultExport(content, spec.name);
          }
        }

        content = cleanOutput(content);

        // Basic validation
        if (spec.path.endsWith(".jsx")) {
          if (!/\bexport\s+default\b/.test(content)) {
            throw new Error("No export default in output");
          }
          if (!/return\s*\(?\s*</.test(content)) {
            throw new Error("No JSX return in output");
          }
        }

        console.log(`[batch-generator] ✓ ${spec.path} (${content.length}c)`);
        return { path: spec.path, content };

      } catch (e: unknown) {
        clearTimeout(timeout);
        recordModelFailure(model);

        const status = (e as { status?: number })?.status ?? 0;
        const msg    = (e as Error)?.message ?? String(e);

        if (status === 401 || status === 403) {
          console.error("[batch-generator] Auth error — cannot continue");
          return makeFallback(spec);
        }

        console.warn(`[batch-generator] ${mlabel(model)} ${spec.path} attempt ${attempt + 1} failed: ${msg}`);

        if (attempt < MAX_FILE_RETRIES - 1) {
          await sleepWithPing(backoffMs(attempt), onPing);
        }
      }
    }

    console.warn(`[batch-generator] ${mlabel(model)} exhausted for ${spec.path}`);
  }

  console.error(`[batch-generator] All models failed for ${spec.path} — fallback`);
  return makeFallback(spec);
}

/**
 * Generate all files in the project plan.
 *
 * WHY FILE-BY-FILE (not batch JSON):
 *   The old batch approach returned ALL files as a single JSON array with code as string values.
 *   LLMs cannot reliably JSON-escape multiline code (newlines, quotes, backslashes).
 *   A single escape error makes the entire JSON unparseable → ALL files fall back → blank UI.
 *   File-by-file means each file is independent — one failure does not affect others.
 *
 * Files are generated in dependency order with PARALLEL_BATCH concurrency.
 * App.jsx is always generated last so it has full context of all other components.
 */
export async function generateAllFiles(
  plan:    ProjectPlan,
  _ds:     DesignSystem,   // kept for API compat — design system is inside plan
  onPing?: () => void,
): Promise<StreamFile[]> {
  // Reset model health for this run
  modelFailures.clear();

  const specs = plan.components;

  // Split: App.jsx last, everything else in priority order
  const appSpec    = specs.find((s) => s.path === "src/App.jsx");
  const nonAppSpecs = specs.filter((s) => s.path !== "src/App.jsx");

  const generated: StreamFile[] = [];

  console.log(`[batch-generator] Generating ${specs.length} files (batch=${PARALLEL_BATCH}, models=${getHealthyModels().map(mlabel).join(",")})`);

  // Generate non-App files in parallel batches
  for (let i = 0; i < nonAppSpecs.length; i += PARALLEL_BATCH) {
    const batch = nonAppSpecs.slice(i, i + PARALLEL_BATCH);
    console.log(`[batch-generator] Batch ${Math.floor(i / PARALLEL_BATCH) + 1}: ${batch.map((s) => s.path).join(", ")}`);

    const results = await Promise.all(
      batch.map((spec) => generateSingleFile(spec, plan, generated, onPing))
    );

    generated.push(...results);
  }

  // Generate App.jsx last — now it can reference all generated components
  if (appSpec) {
    console.log(`[batch-generator] Generating src/App.jsx (final, has context of ${generated.length} files)`);
    const appFile = await generateSingleFile(appSpec, plan, generated, onPing);
    generated.push(appFile);
  }

  console.log(`[batch-generator] ✓ Done — ${generated.length} files`);
  return generated;
}

// src/lib/ai/engine/project-fixer.ts
// STAGE 4b: Surgical LLM patch — broken files only.
// v18.0 — Clean reduction: only sanitizeFiles and surgicalPatch remain.

import OpenAI from "openai";

// ─── Types ─────────────────────────────────────────────────────────────────────

export interface StreamFile {
  path: string;
  content: string;
}

export interface ValidationError {
  file: string;
  type: string;
  message: string;
}

export type ErrorClass =
  | "syntax"
  | "missing-import"
  | "jsx-imbalance"
  | "unknown-component";

export interface ClassifiedError extends ValidationError {
  class: ErrorClass;
}

// ─── Constants ─────────────────────────────────────────────────────────────────

const PING_INTERVAL_MS = 8_000;
const FIX_MODEL = "mistralai/devstral-2-123b-instruct-2512";

function makeClient(): OpenAI {
  return new OpenAI({
    apiKey: process.env.NVIDIA_API_KEY!,
    baseURL: "https://integrate.api.nvidia.com/v1",
  });
}

// ─── Helpers ───────────────────────────────────────────────────────────────────

function stripMarkdown(text: string): string {
  return text
    .replace(/^```(?:json|jsx?|javascript|typescript|tsx?)?\s*/gim, "")
    .replace(/^```\s*$/gim, "")
    .trim();
}

function resolveRelative(dir: string, rel: string): string {
  const parts = (dir ? `${dir}/${rel}` : rel).split("/");
  const out: string[] = [];
  for (const p of parts) {
    if (p === "..") out.pop();
    else if (p !== ".") out.push(p);
  }
  return out.join("/");
}

// ─── Final Sanitization (kept unchanged) ───────────────────────────────────────

export function sanitizeFiles(files: StreamFile[]): StreamFile[] {
  return files.map((f): StreamFile => {
    if (!/\.(jsx?|tsx?|css)$/.test(f.path)) return f;

    let c = f.content;

    // Remove duplicate import lines (exact duplicates)
    const importsSeen = new Set<string>();
    const lines = c.split("\n");
    const deduped: string[] = [];
    for (const line of lines) {
      if (/^\s*import\s/.test(line)) {
        const trimmed = line.trim();
        if (importsSeen.has(trimmed)) continue;
        importsSeen.add(trimmed);
      }
      deduped.push(line);
    }
    c = deduped.join("\n");

    // Remove console.log/warn/error/info/debug calls
    c = c.replace(/\bconsole\.(log|warn|error|info|debug)\s*\([^)]*\)\s*;?/g, "");

    // Normalize trailing whitespace per line
    c = c
      .split("\n")
      .map((l) => l.trimEnd())
      .join("\n");

    // Ensure single trailing newline
    c = c.trimEnd() + "\n";

    return { ...f, content: c };
  });
}

// ─── Surgical Patch ────────────────────────────────────────────────────────────

/**
 * Surgical patch: only broken files + their direct dependents are sent to the LLM.
 * Single call, no retries, no metrics.
 */
export async function surgicalPatch(
  allFiles: StreamFile[],
  brokenPaths: string[],
  errors: ClassifiedError[],
  onPing?: () => void,
): Promise<StreamFile[]> {
  if (brokenPaths.length === 0) return allFiles;

  console.log(`[S4b] Surgical LLM patch on ${brokenPaths.length} broken file(s): ${brokenPaths.join(", ")}`);

  // 1. Determine target files: broken files + dependents
  const brokenSet = new Set(brokenPaths);
  const targetPaths = new Set(brokenPaths);

  for (const file of allFiles) {
    if (brokenSet.has(file.path)) continue;
    const relRe = /from\s+['"](\.[^'"]+)['"]/g;
    let m: RegExpExecArray | null;
    while ((m = relRe.exec(file.content)) !== null) {
      const dir = file.path.split("/").slice(0, -1).join("/");
      const resolved = resolveRelative(dir, m[1]);
      const candidates = [resolved, `${resolved}.jsx`, `${resolved}.js`];
      if (candidates.some((p) => brokenSet.has(p))) {
        targetPaths.add(file.path);
        break;
      }
    }
  }

  const targetFiles = allFiles.filter((f) => targetPaths.has(f.path));

  // 2. Build error summary
  const errorSummary = errors
    .filter((e) => brokenPaths.includes(e.file))
    .map((e) => `[${e.file}] ${e.class}: ${e.message}`)
    .join("\n");

  // 3. Build file bundle
  const fileBundle = targetFiles
    .map((f) => `// ═══ FILE: ${f.path} ═══\n${f.content}`)
    .join("\n\n");

  const pingTimer = onPing ? setInterval(onPing, PING_INTERVAL_MS) : null;

  try {
    const client = makeClient();
    const res = await client.chat.completions.create({
      model: FIX_MODEL,
      temperature: 0.05,
      max_tokens: 16000,
      messages: [
        {
          role: "system",
          content: `You are a React expert fixing specific errors in project files.

Return ONLY a JSON array: [{"path":"...","content":"..."}]
The array must contain EXACTLY ${targetFiles.length} files with IDENTICAL paths.
Fix ONLY the listed errors. Preserve all logic, styling, and functionality.
Every .jsx file must have: import React from 'react' AND export default.
Use Tailwind className for styling. framer-motion and lucide-react are allowed.`,
        },
        {
          role: "user",
          content: `ERRORS TO FIX:\n${errorSummary}\n\nFILES:\n${fileBundle}\n\nReturn all ${targetFiles.length} fixed files as JSON array.`,
        },
      ],
    });

    if (pingTimer) clearInterval(pingTimer);

    const raw = res.choices[0]?.message?.content ?? "";
    const text = stripMarkdown(raw);
    const start = text.indexOf("[");
    const end = text.lastIndexOf("]");
    if (start === -1 || end === -1) {
      throw new Error("No JSON array in patch response");
    }

    const parsed = JSON.parse(text.slice(start, end + 1)) as Array<{ path: string; content: string }>;

    // 4. Validate fixed files
    const validPaths = new Set(targetFiles.map((f) => f.path));
    const fixedMap = new Map<string, string>();

    for (const item of parsed) {
      if (
        typeof item.path === "string" &&
        typeof item.content === "string" &&
        validPaths.has(item.path) &&
        item.content.trim().length > 30
      ) {
        const hasExport = /\bexport\s+default\b/.test(item.content);
        const hasJSX = /return\s*\(?\s*</.test(item.content) || /return\s+</.test(item.content);
        if (item.path.endsWith(".jsx") && (!hasExport || !hasJSX)) continue;
        fixedMap.set(item.path, item.content);
      }
    }

    // 5. Merge fixes
    const result = allFiles.map((f) => {
      if (!fixedMap.has(f.path)) return f;
      return { ...f, content: fixedMap.get(f.path)! };
    });

    console.log(`[S4b] ✓ Surgical patch applied to ${fixedMap.size}/${targetFiles.length} files`);
    return result;
  } catch (error) {
    if (pingTimer) clearInterval(pingTimer);
    console.error(`[S4b] Surgical patch failed: ${String(error)} — keeping deterministic fixes`);
    return allFiles;
  }
}

// src/lib/ai/sanitize/auto-fix.ts
// v18.0 — Minimal safe non-destructive cleaning.
//
// Only performs:
//   - Duplicate import removal
//   - Fix "= />" proxy artifact → "=>"
//   - Ensure React import if JSX present
//   - Ensure export default in .jsx files
//   - Trim trailing whitespace
//   - Ensure final newline

export interface StreamFile {
  path: string;
  content: string;
}

/**
 * Apply safe, non-destructive fixes to all files.
 */
export function autoFixFiles(files: StreamFile[]): StreamFile[] {
  return files.map((file) => {
    // Only process JavaScript/JSX files
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js")) {
      return file;
    }

    let content = file.content;

    // 1. Remove duplicate import lines (exact matches)
    const seenImports = new Set<string>();
    const lines = content.split("\n");
    const dedupedLines: string[] = [];
    for (const line of lines) {
      const trimmed = line.trim();
      if (trimmed.startsWith("import ")) {
        if (seenImports.has(trimmed)) continue;
        seenImports.add(trimmed);
      }
      dedupedLines.push(line);
    }
    content = dedupedLines.join("\n");

    // 2. Fix broken arrow function artifact "= />" → "=>"
    content = content.replace(/=\s*\/\s*>/g, "=>");

    // 3. Ensure React import if JSX is used
    const hasJSX = /<[A-Za-z]/.test(content);
    const hasReactImport = /\bimport\s+React\b/.test(content);
    if (hasJSX && !hasReactImport) {
      content = `import React from 'react';\n${content}`;
    }

    // 4. Ensure export default in .jsx files
    if (file.path.endsWith(".jsx") && !/\bexport\s+default\b/.test(content)) {
      const componentName = extractComponentName(content, file.path);
      if (componentName) {
        content = content.trimEnd() + `\n\nexport default ${componentName};\n`;
      }
    }

    // 5. Trim trailing whitespace per line
    content = content
      .split("\n")
      .map((line) => line.trimEnd())
      .join("\n");

    // 6. Ensure single trailing newline
    content = content.trimEnd() + "\n";

    return { ...file, content };
  });
}

/**
 * Try to extract the component name from content or fallback to path.
 */
function extractComponentName(content: string, path: string): string | null {
  // Look for function ComponentName() or const ComponentName = () =>
  const match = content.match(
    /(?:export\s+)?(?:function|const)\s+([A-Z][a-zA-Z0-9_]*)/,
  );
  if (match) return match[1];

  // Fallback to filename without extension
  const base = path.split("/").pop() ?? "";
  const name = base.replace(/\.[^.]+$/, "");
  if (/^[A-Z]/.test(name)) return name;

  return null;
}

  static patch(existingFiles: AIFileNode[], instructions: string): AIFileNode[] {
    if (!existingFiles || existingFiles.length === 0) {

    }

    const files = structuredClone(existingFiles);

    // Validate first
    const validation = this.validateFiles(files);
    if (!validation.isValid) {

    }

    const metadata = {
      timestamp: new Date().toISOString(),

      editedBy: "AI Assistant",
    };

    // Patch App.jsx safely
    const index = files.findIndex((f) => f.path === "src/App.jsx");

    if (index !== -1) {
      files[index] = this.patchAppComponent(files[index], metadata);
    }

    // Update or create changelog
    this.upsertChangelog(files, metadata);

    return files;
  }

  // ─────────────────────────────────────────────────────────────
  // AI PATCH (REAL, NOT FAKE)
  // ─────────────────────────────────────────────────────────────

  ): Promise<AIFileNode[]> {

    if (!aiClient) {

        prompt: instructions,
        files: existingFiles,
      });

      if (!response || !Array.isArray(response.files)) {
        throw new Error("Invalid AI response");
      }

      return response.files;
    } catch (err) {
      console.warn("⚠️ AI patch failed, fallback to basic patch");
      return this.patch(existingFiles, instructions);
    }
  }

  // ─────────────────────────────────────────────────────────────
  // SAFE PATCHING
  // ─────────────────────────────────────────────────────────────

  private static patchAppComponent(
    file: AIFileNode,
    metadata: { timestamp: string; instructions: string }
  ): AIFileNode {

    if (!file.content.includes("export default")) {
      return file; // avoid corrupting non-component files
    }

    const header = `
// ============================================
// 🔧 AI EDIT: ${metadata.timestamp}
// Instructions: ${metadata.instructions}
// ============================================

`;

    // Prevent duplicate headers
    if (file.content.includes("🔧 AI EDIT")) {
      return file;
    }

    return {
      ...file,
      content: header + file.content,
    };
  }

  // ─────────────────────────────────────────────────────────────
  // CHANGELOG HANDLING
  // ─────────────────────────────────────────────────────────────

  private static upsertChangelog(
    files: AIFileNode[],
    metadata: { timestamp: string; instructions: string; editedBy: string }
  ) {

    const existing = files.find((f) => f.path === "CHANGELOG.md");

    const entry = `
## [${metadata.timestamp}]
- Instructions: ${metadata.instructions}
- Edited by: ${metadata.editedBy}
`;

    if (existing) {
      existing.content += entry;
    } else {
      files.push({
        path: "CHANGELOG.md",
        content: `# Changelog\n${entry}`,
        type: "file",
      });
    }
  }

  // VALIDATION
  // ─────────────────────────────────────────────────────────────

  static validateFiles(files: AIFileNode[]): AIFileValidation {
    const errors: string[] = [];
    const warnings: string[] = [];

    const seen = new Set<string>();

    for (const file of files) {
      if (!file.path) {
        errors.push("File missing path");
      }

      if (seen.has(file.path)) {
        errors.push(`Duplicate file path: ${file.path}`);
      }
      seen.add(file.path);

      if (file.content == null) {
        errors.push(`Missing content: ${file.path}`);
      }

      if (file.content && file.content.length > 1024 * 1024) {
        warnings.push(`Large file: ${file.path}`);
      }
    }

    return {
      isValid: errors.length === 0,
      errors,
      warnings,
    };
  }

  // ─────────────────────────────────────────────────────────────
  // DIFF TRACKING
  // ─────────────────────────────────────────────────────────────

  static getChanges(
    oldFiles: AIFileNode[],
    newFiles: AIFileNode[]
  ): { added: string[]; modified: string[]; deleted: string[] } {

    const oldMap = new Map(oldFiles.map((f) => [f.path, f.content]));
    const newMap = new Map(newFiles.map((f) => [f.path, f.content]));

    const added: string[] = [];
    const modified: string[] = [];
    const deleted: string[] = [];

    for (const [path] of newMap) {
      if (!oldMap.has(path)) {
        added.push(path);
      } else if (oldMap.get(path) !== newMap.get(path)) {
        modified.push(path);
      }
    }

    for (const [path] of oldMap) {
      if (!newMap.has(path)) {
        deleted.push(path);
      }
    }

    return { added, modified, deleted };
  }

  // ─────────────────────────────────────────────────────────────
  // SANITIZATION
  // ─────────────────────────────────────────────────────────────

  private static safeText(text: string): string {
    return text.replace(/[`$<>]/g, "").slice(0, 500);
  }

// src/lib/ai/engine/deterministic-fixer.ts
// Stage 3: Deterministic fixer — pure mechanical corrections. NO LLM.

export interface StreamFile {
  path: string;
  content: string;
}

/**
 * Resolve a relative import path from a directory.
 */
function resolveRelative(dir: string, rel: string): string {
  const parts = (dir ? `${dir}/${rel}` : rel).split("/");
  const out: string[] = [];
  for (const p of parts) {
    if (p === "..") out.pop();
    else if (p !== ".") out.push(p);
  }
  return out.join("/");
}

/**
 * Extract component name from file path.
 */
function componentName(path: string): string {
  const base = path.split("/").pop() ?? "";
  return base.replace(/\.[^.]+$/, "");
}

/**
 * Apply deterministic fixes to all files.
 */
export function deterministicFix(files: StreamFile[]): StreamFile[] {
  const pathSet = new Set(files.map((f) => f.path));

  return files.map((file) => {
    // Only process JS/JSX files
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js")) {
      return file;
    }

    let content = file.content;

    // ─── 1. Fix import paths — add .jsx extension where missing ─────────────────
    content = content.replace(
      /from\s+(['"])(\.\.?\/[^'"]+)(['"])/g,
      (match, quote1: string, importPath: string, quote2: string) => {
        // Already has an extension
        if (/\.(jsx?|tsx?|js|css|svg|png|json)$/.test(importPath)) {
          return match;
        }
        const withJsx = importPath + ".jsx";
        const dir = file.path.split("/").slice(0, -1).join("/");
        const resolved = resolveRelative(dir, withJsx);
        if (pathSet.has(resolved)) {
          return `from ${quote1}${importPath}.jsx${quote2}`;
        }
        // Keep as is if file not found
        return match;
      }
    );

    // ─── 2. Ensure React import ─────────────────────────────────────────────────
    if (/<[A-Za-z]/.test(content) && !/\bimport\s+React\b/.test(content)) {
      content = `import React from 'react';\n${content}`;
    }

    // ─── 3. Ensure export default ───────────────────────────────────────────────
    if (file.path.endsWith(".jsx") && !/\bexport\s+default\b/.test(content)) {
      const fnMatch = content.match(/^(?:function|const)\s+([A-Z][a-zA-Z0-9_]*)/m);
      if (fnMatch) {
        content = content.trimEnd() + `\n\nexport default ${fnMatch[1]};\n`;
      } else {
        const name = componentName(file.path);
        content += `\n\nexport default function ${name}() {\n  return <div>${name}</div>;\n}\n`;
      }
    }

    // ─── 4. Fix useEffect missing dependency arrays ────────────────────────────
    content = content.replace(
      /(\buseEffect\s*\(\s*(?:async\s*)?\(\s*\)\s*=>|useEffect\s*\(\s*function\s*\(\s*\))(\s*\{(?:[^{}]|\{[^{}]*\})*\})\s*\)(\s*;?)/g,
      (full) => {
        if (/,\s*\[/.test(full)) return full; // already has dep array
        return full.replace(/\}\s*\)(\s*;?)$/, `}, [])$1`);
      }
    );

    // ─── 5. Deduplicate imports ────────────────────────────────────────────────
    const seen = new Set<string>();
    content = content
      .split("\n")
      .filter((line) => {
        const trimmed = line.trim();
        if (!trimmed.startsWith("import ")) return true;
        if (seen.has(trimmed)) return false;
        seen.add(trimmed);
        return true;
      })
      .join("\n");

    // ─── 6. Balance JSX tags (heuristic) ───────────────────────────────────────
    const openCount = (content.match(/<[A-Z][a-zA-Z0-9]*[\s>]/g) ?? []).length;
    const closeCount = (content.match(/<\/[A-Z][a-zA-Z0-9]*>/g) ?? []).length;
    const selfCount = (content.match(/<[A-Z][a-zA-Z0-9]*[^>]*\/>/g) ?? []).length;
    const delta = openCount - closeCount - selfCount;

    if (Math.abs(delta) > 3) {
      // Wrap the return JSX in a fragment
      content = content.replace(
        /(\breturn\s*\(\s*)([\s\S]*?)(\s*\)\s*;)/,
        (_m, pre, jsx, post) => `${pre}<>${jsx}</>${post}`
      );
    }

    // ─── 7. Fix broken arrow function proxy artifact ───────────────────────────
    content = content.replace(/(\)|\])\s*=\s*\/\s*>/g, "$1 =>");
    content = content.replace(/(?<![=!<>])=\s*\/\s*>/g, "=>");

    return { ...file, content };
  });
}

// src/lib/ai/merge/merge-files.ts
// v18.0 — Minimal safe merge of base and AI-generated files.
// No validation, no fallback generation, no React checks.

export interface StreamFile {
  path: string;
  content: string;
}

export function mergeFiles(baseFiles: StreamFile[], aiFiles: StreamFile[]): StreamFile[] {
  const fileMap = new Map<string, StreamFile>();

  // Start with base files
  for (const file of baseFiles) {
    fileMap.set(file.path, { ...file });
  }

  // Merge AI files
  for (const aiFile of aiFiles) {
    if (aiFile.path === "package.json") {
      const baseFile = fileMap.get("package.json");
      try {
        const basePkg = baseFile ? JSON.parse(baseFile.content) : {};
        const aiPkg = JSON.parse(aiFile.content);

        const merged = {
          ...basePkg,
          ...aiPkg,
          name: basePkg.name ?? aiPkg.name,
          version: basePkg.version ?? aiPkg.version,
          type: basePkg.type ?? aiPkg.type,
          scripts: { ...basePkg.scripts, ...aiPkg.scripts },
          dependencies: { ...basePkg.dependencies, ...aiPkg.dependencies },
          devDependencies: { ...basePkg.devDependencies, ...aiPkg.devDependencies },
        };

        fileMap.set("package.json", {
          path: "package.json",
          content: JSON.stringify(merged, null, 2),
        });
      } catch {
        // If parsing fails, keep the base package.json (if exists) unchanged
      }
      continue;
    }

    // For all other files, simply overwrite base with AI version
    fileMap.set(aiFile.path, { ...aiFile });
  }

  // Final cleanup: remove empty files and trim content
  const result: StreamFile[] = [];
  for (const file of fileMap.values()) {
    const trimmed = file.content?.trim();
    if (!trimmed) continue;
    result.push({
      path: file.path,
      content: trimmed + "\n",
    });
  }

  return result;
}

// src/lib/ai/validation/semantic-validator.ts
// v18.0 — Pure analysis validator. No mutations, no auto-fix, no caching.
//
// Detects:
//   - missing / multiple export default (error)
//   - unresolvable relative imports (error)
//   - JSX tag imbalance (error)
//   - missing imports for used components (warning)
//   - useEffect without dependency array (warning)
//
// Returns a ValidationResult with errors and a list of broken files.

import type { StreamFile } from "@/lib/ai/types/stream.types";

export interface ValidationError {
  file: string;
  type:
    | "missing-export-default"
    | "multiple-export-default"
    | "invalid-import-path"
    | "jsx-imbalance"
    | "missing-import"
    | "hook-missing-deps";
  message: string;
  severity: "error" | "warning";
}

export interface ValidationResult {
  errors: ValidationError[];
  brokenFiles: string[];
  clean: boolean;
}

// =============================================================================
// UTILITIES
// =============================================================================

function stripCommentsAndStrings(content: string): string {
  return content
    .replace(/\/\/[^\n]*/g, "")
    .replace(/\/\*[\s\S]*?\*\//g, "")
    .replace(/`(?:[^`\\]|\\.)*`/gs, "``")
    .replace(/"(?:[^"\\]|\\.)*"/g, '""')
    .replace(/'(?:[^'\\]|\\.)*'/g, "''");
}

function resolveRelativePath(dir: string, relativePath: string): string {
  const base = dir ? `${dir}/${relativePath}` : relativePath;
  const parts = base.split("/");
  const resolved: string[] = [];
  for (const p of parts) {
    if (p === "..") resolved.pop();
    else if (p !== ".") resolved.push(p);
  }
  return resolved.join("/");
}

function pathExists(importPath: string, allPaths: Set<string>): boolean {
  if (allPaths.has(importPath)) return true;
  for (const ext of [".jsx", ".tsx", ".js", ".ts"]) {
    if (allPaths.has(importPath + ext)) return true;
    if (allPaths.has(importPath + "/index" + ext)) return true;
  }
  return false;
}

// =============================================================================
// JSX TAG BALANCE CHECK
// =============================================================================

function checkJsxBalance(content: string): { valid: boolean; message?: string } {
  const stripped = stripCommentsAndStrings(content);
  const stack: string[] = [];
  let fragmentDepth = 0;
  let mismatchCount = 0;

  const tagRe = /<\/?([A-Za-z][a-zA-Z0-9]*)[^>]*\/?>/g;
  let m: RegExpExecArray | null;

  while ((m = tagRe.exec(stripped)) !== null) {
    const full = m[0];
    const tagName = m[1];

    // Fragment handling
    if (full === "<>") {
      fragmentDepth++;
      continue;
    }
    if (full === "</>") {
      if (fragmentDepth > 0) fragmentDepth--;
      else mismatchCount++;
      continue;
    }

    const isClosing = full.startsWith("</");
    const isSelfClosing = full.endsWith("/>");

    if (isClosing) {
      if (stack.length === 0) {
        mismatchCount++;
      } else {
        const last = stack.pop();
        if (last !== tagName) mismatchCount++;
      }
    } else if (!isSelfClosing) {
      stack.push(tagName);
    }
  }

  const openCount = stack.length + fragmentDepth + mismatchCount;
  if (openCount > 3) {
    return {
      valid: false,
      message: `JSX imbalance: ${stack.length} unclosed tags, ${fragmentDepth} fragments, ${mismatchCount} mismatches`,
    };
  }
  return { valid: true };
}

// =============================================================================
// COMPONENT USAGE EXTRACTION
// =============================================================================

function extractComponentUsages(stripped: string): string[] {
  const seen = new Set<string>();
  const re = /(?<![.\w])<([A-Z][a-zA-Z0-9]*)(?=[\s/>])/g;
  let m: RegExpExecArray | null;
  while ((m = re.exec(stripped)) !== null) {
    seen.add(m[1]);
  }
  return Array.from(seen);
}

// =============================================================================
// IMPORT EXTRACTION
// =============================================================================

interface ExtractedImport {
  source: string;
  names: string[];
}

function extractImports(content: string): ExtractedImport[] {
  const imports: ExtractedImport[] = [];
  const importRe = /^import\s+(.+?)\s+from\s+["']([^"']+)["']/gm;
  let m: RegExpExecArray | null;

  while ((m = importRe.exec(content)) !== null) {
    const clause = m[1].trim();
    const source = m[2];
    const names: string[] = [];

    // Parse import clause
    if (clause.startsWith("*")) {
      const nsMatch = clause.match(/^\*\s+as\s+(\w+)/);
      if (nsMatch) names.push(nsMatch[1]);
    } else {
      const defaultMatch = clause.match(/^(\w+)/);
      if (defaultMatch && !clause.includes("{")) {
        names.push(defaultMatch[1]);
      } else {
        if (defaultMatch && defaultMatch[1] !== "type") {
          names.push(defaultMatch[1]);
        }
        const namedMatch = clause.match(/\{([^}]*)\}/);
        if (namedMatch) {
          for (const spec of namedMatch[1].split(",")) {
            const parts = spec.trim().split(/\s+as\s+/);
            const local = parts[parts.length - 1]?.trim();
            if (local) names.push(local);
          }
        }
      }
    }
    imports.push({ source, names: names.filter(Boolean) });
  }
  return imports;
}

// =============================================================================
// MAIN VALIDATION
// =============================================================================

export function runSemanticValidation(files: StreamFile[]): ValidationResult {
  const errors: ValidationError[] = [];
  const brokenFiles = new Set<string>();
  const allPaths = new Set(files.map((f) => f.path));

  for (const file of files) {
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".tsx")) continue;

    const content = file.content;
    const stripped = stripCommentsAndStrings(content);

    // 1. Missing export default
    const hasExportDefault = /\bexport\s+default\b/.test(content);
    if (!hasExportDefault) {
      errors.push({
        file: file.path,
        type: "missing-export-default",
        message: "Missing export default",
        severity: "error",
      });
      brokenFiles.add(file.path);
    }

    // 2. Multiple export default
    const exportMatches = stripped.match(/\bexport\s+default\b/g);
    if (exportMatches && exportMatches.length > 1) {
      errors.push({
        file: file.path,
        type: "multiple-export-default",
        message: `Multiple export default (${exportMatches.length})`,
        severity: "error",
      });
      brokenFiles.add(file.path);
    }

    // 3. Invalid import paths
    const importLines = extractImports(content);
    for (const imp of importLines) {
      if (imp.source.startsWith(".")) {
        const dir = file.path.split("/").slice(0, -1).join("/");
        const resolved = resolveRelativePath(dir, imp.source);
        if (!pathExists(resolved, allPaths)) {
          errors.push({
            file: file.path,
            type: "invalid-import-path",
            message: `Unresolvable import: "${imp.source}" → "${resolved}"`,
            severity: "error",
          });
          brokenFiles.add(file.path);
        }
      }
    }

    // 4. JSX imbalance
    const balance = checkJsxBalance(content);
    if (!balance.valid) {
      errors.push({
        file: file.path,
        type: "jsx-imbalance",
        message: balance.message!,
        severity: "error",
      });
      brokenFiles.add(file.path);
    }

    // 5. Missing imports (warning)
    const importedNames = new Set<string>();
    for (const imp of importLines) {
      imp.names.forEach((n) => importedNames.add(n));
    }
    const locallyDefined = new Set<string>();
    const localRe = /(?:function|const|class)\s+([A-Z][a-zA-Z0-9]*)/g;
    let lm: RegExpExecArray | null;
    while ((lm = localRe.exec(content)) !== null) {
      locallyDefined.add(lm[1]);
    }
    const usedComponents = extractComponentUsages(stripped);
    for (const comp of usedComponents) {
      if (!importedNames.has(comp) && !locallyDefined.has(comp)) {
        errors.push({
          file: file.path,
          type: "missing-import",
          message: `Component <${comp}> used but not imported or defined`,
          severity: "warning",
        });
      }
    }

    // 6. useEffect missing dependency array (warning)
    const useEffectRe = /useEffect\s*\(\s*(?:\(\)\s*=>|function\s*\(\))\s*\{[^}]*\}\s*\)(?!\s*,)/g;
    if (useEffectRe.test(content)) {
      errors.push({
        file: file.path,
        type: "hook-missing-deps",
        message: "useEffect missing dependency array",
        severity: "warning",
      });
    }
  }

  const clean = errors.filter((e) => e.severity === "error").length === 0;
  return {
    errors,
    brokenFiles: Array.from(brokenFiles),
    clean,
  };
}

// src/lib/ai/validation/syntax-checker.ts
// v18.0 — Pure syntax checker. No semantics, no imports, no hooks, no React.

import type { StreamFile } from "@/lib/ai/stream/stream-pipeline";

export interface SyntaxError {
  path: string;
  message: string;
}

export function checkSyntax(files: StreamFile[]): SyntaxError[] {
  const errors: SyntaxError[] = [];

  for (const file of files) {
    const content = file.content;

    // 1. Empty / too short file
    if (!content || content.trim().length < 10) {
      errors.push({ path: file.path, message: "File empty or too short" });
      continue;
    }

    // Only apply further checks to JS/JSX files
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js") && !file.path.endsWith(".tsx") && !file.path.endsWith(".ts")) {
      continue;
    }

    // 2. Multiple export default
    const exportCount = (content.match(/\bexport\s+default\b/g) ?? []).length;
    if (exportCount > 1) {
      errors.push({ path: file.path, message: "Multiple export default statements" });
    }

    // 3. Broken arrow artifact
    if (/=\s*\/\s*>/.test(content)) {
      errors.push({ path: file.path, message: "Corrupted arrow => operator" });
    }

    // 4. Unbalanced braces
    const openBraces = (content.match(/\{/g) ?? []).length;
    const closeBraces = (content.match(/\}/g) ?? []).length;
    if (Math.abs(openBraces - closeBraces) > 2) {
      errors.push({ path: file.path, message: "Unbalanced braces" });
    }

    // 5. Unbalanced parentheses
    const openParens = (content.match(/\(/g) ?? []).length;
    const closeParens = (content.match(/\)/g) ?? []).length;
    if (Math.abs(openParens - closeParens) > 2) {
      errors.push({ path: file.path, message: "Unbalanced parentheses" });
    }

    // 6. Basic JSX tag imbalance
    const openTags = (content.match(/<[A-Za-z][^>]*?>/g) ?? []).filter(t => !t.startsWith("</") && !t.endsWith("/>")).length;
    const closeTags = (content.match(/<\/[A-Za-z][^>]*?>/g) ?? []).length;
    if (Math.abs(openTags - closeTags) > 3) {
      errors.push({ path: file.path, message: "JSX tag imbalance" });
    }

    // 7. Redacted / corrupted token
    if (/<REDACTED/.test(content)) {
      errors.push({ path: file.path, message: "Redacted token detected" });
    }

    // 8. Truncated file (too many open parens)
    if (openParens - closeParens > 4) {
      errors.push({ path: file.path, message: "Likely truncated file" });
    }
  }

  return errors;
}


========== src/lib/ai/validation/build-simulator.ts ==========

// src/lib/ai/validation/build-simulator.ts
// v18.0 — Lightweight build validator. Checks required files and import resolution.

import type { StreamFile } from "@/lib/ai/stream/stream-pipeline";

export function simulateBuild(files: StreamFile[]): {
  success: boolean;
  logs: string[];
} {
  const logs: string[] = [];

  if (!files.length) {
    return { success: false, logs: ["No files to build"] };
  }

  const fileMap = new Map(files.map(f => [f.path, f.content]));
  const allPaths = new Set(fileMap.keys());

  // 1. Required files check
  const requiredFiles = ["src/App.jsx", "src/main.jsx"];
  const missingRequired = requiredFiles.filter(p => !fileMap.has(p));
  if (missingRequired.length > 0) {
    logs.push(`Missing required files: ${missingRequired.join(", ")}`);
    return { success: false, logs };
  }

  let hasImportErrors = false;

  // 2. Relative import resolution
  for (const file of files) {
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js")) continue;

    const importRegex = /import\s+.*?\s+from\s+['"](\.\.?\/[^'"]+)['"]/g;
    let match;
    while ((match = importRegex.exec(file.content)) !== null) {
      const rawImport = match[1];
      const dir = file.path.split("/").slice(0, -1).join("/");
      const resolved = resolveRelativePath(dir, rawImport);

      // Check with possible extensions
      const possiblePaths = [
        resolved,
        `${resolved}.jsx`,
        `${resolved}.js`,
        `${resolved}/index.jsx`,
        `${resolved}/index.js`,
      ];

      const exists = possiblePaths.some(p => allPaths.has(p));
      if (!exists) {
        logs.push(`[${file.path}] Unresolved import: "${rawImport}" (resolved to ${resolved})`);
        hasImportErrors = true;
      }
    }
  }

  // 3. Basic JSX return check (warning only)
  for (const file of files) {
    if (!file.path.endsWith(".jsx")) continue;
    const content = file.content;
    const hasReturn = /return\s*\(?\s*</.test(content) || /return\s+</.test(content);
    if (!hasReturn) {
      logs.push(`[${file.path}] Warning: No JSX return statement found`);
    }
  }

  const success = !hasImportErrors && missingRequired.length === 0;
  if (success) logs.push("Build simulation passed");

  return { success, logs };
}

function resolveRelativePath(dir: string, relativePath: string): string {
  const parts = (dir ? `${dir}/${relativePath}` : relativePath).split("/");
  const resolved: string[] = [];
  for (const part of parts) {
    if (part === "..") resolved.pop();
    else if (part !== ".") resolved.push(part);
  }
  return resolved.join("/");
}


========== src/lib/ai/validation/error-classifier.ts ==========

// src/lib/ai/validation/error-classifier.ts
// Stage 4: Semantic validator — classify, don't reject.

export interface StreamFile {
  path: string;
  content: string;
}

export interface ClassifiedError {
  file: string;
  class: "syntax" | "missing-import" | "jsx-imbalance" | "runtime";
  message: string;
  severity: "error" | "warning";
}

export interface ClassificationResult {
  errors: ClassifiedError[];
  brokenFiles: string[];
  clean: boolean;
}

/**
 * Resolve a relative import path from a directory.
 */
function resolveRelative(dir: string, rel: string): string {
  const parts = (dir ? `${dir}/${rel}` : rel).split("/");
  const out: string[] = [];
  for (const p of parts) {
    if (p === "..") out.pop();
    else if (p !== ".") out.push(p);
  }
  return out.join("/");
}

/**
 * Extract imported identifiers from a file content.
 */
function extractImports(content: string): Map<string, string[]> {
  const importMap = new Map<string, string[]>();
  const importRegex = /^import\s+(?:{([^}]+)}|([A-Za-z_$][\w$]*))\s+from\s+['"]([^'"]+)['"]/gm;
  let match;
  while ((match = importRegex.exec(content)) !== null) {
    const namedImports = match[1]?.split(",").map(s => s.trim().split(" as ").pop()!.trim()) ?? [];
    const defaultImport = match[2] ? [match[2]] : [];
    const modulePath = match[3];
    const existing = importMap.get(modulePath) ?? [];
    importMap.set(modulePath, [...existing, ...namedImports, ...defaultImport]);
  }
  return importMap;
}

/**
 * Extract component names used in JSX.
 */
function extractJsxComponents(content: string): Set<string> {
  const components = new Set<string>();
  const jsxTagRegex = /<([A-Z][a-zA-Z0-9_]*)/g;
  let match;
  while ((match = jsxTagRegex.exec(content)) !== null) {
    components.add(match[1]);
  }
  return components;
}

/**
 * Extract locally defined components (function declarations/const arrow functions).
 */
function extractLocalDefinitions(content: string): Set<string> {
  const defined = new Set<string>();
  const defRegex = /(?:function|const|class)\s+([A-Z][a-zA-Z0-9_]*)/g;
  let match;
  while ((match = defRegex.exec(content)) !== null) {
    defined.add(match[1]);
  }
  return defined;
}

export function classifyErrors(files: StreamFile[]): ClassificationResult {
  const errors: ClassifiedError[] = [];
  const brokenFiles = new Set<string>();
  const pathSet = new Set(files.map(f => f.path));

  for (const file of files) {
    if (!file.path.endsWith(".jsx") && !file.path.endsWith(".js")) continue;
    const content = file.content;

    // 1. Syntax errors
    const exportCount = (content.match(/\bexport\s+default\b/g) ?? []).length;
    if (exportCount === 0) {
      errors.push({
        file: file.path,
        class: "syntax",
        message: "Missing export default",
        severity: "error",
      });
      brokenFiles.add(file.path);
    } else if (exportCount > 1) {
      errors.push({
        file: file.path,
        class: "syntax",
        message: "Multiple export default statements",
        severity: "error",
      });
      brokenFiles.add(file.path);
    }

    // 2. JSX issues
    const hasJsxReturn = /return\s*\(?\s*</.test(content) || /return\s+</.test(content);
    if (!hasJsxReturn) {
      errors.push({
        file: file.path,
        class: "jsx-imbalance",
        message: "No JSX return found",
        severity: "error",
      });
      brokenFiles.add(file.path);
    } else {
      const openCount = (content.match(/<[A-Z][a-zA-Z0-9]*[\s>]/g) ?? []).length;
      const closeCount = (content.match(/<\/[A-Z][a-zA-Z0-9]*>/g) ?? []).length;
      const selfCount = (content.match(/<[A-Z][a-zA-Z0-9]*[^>]*\/>/g) ?? []).length;
      const delta = Math.abs(openCount - closeCount - selfCount);
      if (delta > 3) {
        errors.push({
          file: file.path,
          class: "jsx-imbalance",
          message: `Severe tag imbalance (open: ${openCount}, close: ${closeCount}, self: ${selfCount})`,
          severity: "error",
        });
        brokenFiles.add(file.path);
      }
    }

    // 3. Missing imports
    const imports = extractImports(content);
    const importedNames = new Set<string>();
    // FIX: Cannot spread array into Set.add() - use forEach instead
    for (const names of imports.values()) {
      names.forEach(name => importedNames.add(name));
    }
    const localDefs = extractLocalDefinitions(content);
    const usedComponents = extractJsxComponents(content);

    for (const comp of usedComponents) {
      if (!importedNames.has(comp) && !localDefs.has(comp)) {
        errors.push({
          file: file.path,
          class: "missing-import",
          message: `<${comp}> used but not imported or defined locally`,
          severity: "warning",
        });
      }
    }

    // 4. Unresolvable relative imports
    const relImportRegex = /from\s+['"](\.[^'"]+)['"]/g;
    let relMatch;
    while ((relMatch = relImportRegex.exec(content)) !== null) {
      const relPath = relMatch[1];
      if (relPath.endsWith(".css")) continue;
      const dir = file.path.split("/").slice(0, -1).join("/");
      const base = resolveRelative(dir, relPath);
      const candidates = [base, `${base}.jsx`, `${base}.js`];
      if (!candidates.some(p => pathSet.has(p))) {
        errors.push({
          file: file.path,
          class: "missing-import",
          message: `Unresolvable import: ${relPath}`,
          severity: "error",
        });
        brokenFiles.add(file.path);
      }
    }

    // 5. Runtime issues (warnings)
    const useEffectMissingDep = /useEffect\s*\(\s*(?:\(\)\s*=>|function\s*\(\))\s*\{[^}]*\}\s*\)(?!\s*,)/g;
    if (useEffectMissingDep.test(content)) {
      errors.push({
        file: file.path,
        class: "runtime",
        message: "useEffect without dependency array",
        severity: "warning",
      });
    }

    const hookInsideCondition = /if\s*\([^)]*\)\s*\{\s*use[A-Z]/g;
    if (hookInsideCondition.test(content)) {
      errors.push({
        file: file.path,
        class: "runtime",
        message: "React Hook used inside conditional (may break rules of hooks)",
        severity: "warning",
      });
    }
  }

  const broken = Array.from(brokenFiles);
  return {
    errors,
    brokenFiles: broken,
    clean: errors.filter(e => e.severity === "error").length === 0,
  };
}

export function getBrokenFiles(result: ClassificationResult): string[] {
  return result.brokenFiles;
}

// src/lib/ai/llm/editor.ts
// v18.0 — Clean editor: modify only necessary files, preserve existing libraries and styling.

import OpenAI from "openai";
import type { StreamFile } from "@/lib/ai/stream/stream-pipeline";

const client = new OpenAI({
  apiKey: process.env.NVIDIA_API_KEY!,
  baseURL: "https://integrate.api.nvidia.com/v1",
});

const MODEL = "meta/llama-3.1-70b-instruct";
const PING_INTERVAL_MS = 8_000;

function stripMarkdown(text: string): string {
  return text
    .replace(/^```(?:json)?\s*/m, "")
    .replace(/\s*```\s*$/m, "")
    .trim();
}

export async function generateEdit(
  prompt: string,
  files: StreamFile[],
  onPing?: () => void
): Promise<StreamFile[]> {
  const pingTimer = onPing ? setInterval(onPing, PING_INTERVAL_MS) : null;

  try {
    const fileBundle = files
      .map((f) => `// ═══ ${f.path} ═══\n${f.content}`)
      .join("\n\n");

    const systemPrompt = `You are a senior React engineer editing an existing project.

Return ONLY a valid JSON array of modified files. No markdown, no explanation.

Format: [{"path": "src/...", "content": "..."}]

RULES:
- Modify ONLY the files that need changes.
- Keep the existing project structure and styling.
- Tailwind CSS is allowed (className with Tailwind utilities).
- framer-motion is allowed (import { motion, AnimatePresence } from "framer-motion").
- lucide-react is allowed (import icons from "lucide-react").
- Preserve all existing functionality and imports.
- DO NOT remove working code unless requested.
- DO NOT introduce new libraries unless necessary.
- Every .jsx file must have import React from 'react' and export default.`;

    const userPrompt = `User request: ${prompt}

Current project files:
${fileBundle}

Return the updated versions of ONLY the files that changed.`;

    const completion = await client.chat.completions.create({
      model: MODEL,
      temperature: 0.2,
      max_tokens: 2000, // NVIDIA-safe limit
      messages: [
        { role: "system", content: systemPrompt },
        { role: "user", content: userPrompt },
      ],
    });

    if (pingTimer) clearInterval(pingTimer);

    const raw = completion.choices[0]?.message?.content ?? "";
    const cleaned = stripMarkdown(raw);

    const start = cleaned.indexOf("[");
    const end = cleaned.lastIndexOf("]");

    if (start === -1 || end === -1) {
      console.warn("[editor] No JSON array found in response");
      return files;
    }

    const parsed = JSON.parse(cleaned.slice(start, end + 1));

    if (!Array.isArray(parsed)) {
      console.warn("[editor] Parsed response is not an array");
      return files;
    }

    const updatedMap = new Map<string, string>();

    for (const item of parsed) {
      if (
        typeof item === "object" &&
        item !== null &&
        typeof item.path === "string" &&
        typeof item.content === "string" &&
        item.content.trim().length > 20
      ) {
        updatedMap.set(item.path, item.content);
      }
    }

    if (updatedMap.size === 0) {
      console.warn("[editor] No valid file updates in response");
      return files;
    }

    const result = files.map((f) => {
      const newContent = updatedMap.get(f.path);
      return newContent ? { ...f, content: newContent } : f;
    });

    return result;
  } catch (error) {
    if (pingTimer) clearInterval(pingTimer);
    console.error("❌ [editor] LLM call failed:", error);
    return files;
  }
}

// src/lib/ai/router/model-router.ts
import { AIModelId } from "@/config/ai.config";

export const resolveModel = (model?: string): AIModelId => {
  if (!model) return "gpt-5";

  const normalized = model.toLowerCase();

  if (normalized.includes("gemini")) return "gemini";
  if (normalized.includes("deepseek")) return "deepseek";

  return "gpt-5";
};
