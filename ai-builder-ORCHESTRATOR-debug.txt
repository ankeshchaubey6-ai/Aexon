// src/lib/ai/engine/ai-orchestrator.ts

import {
  AIFileNode,
  AIProject,
  AIProjectPlan,
  AIProjectError,
} from "../types/ai-project.types";

import { CodeGenerator } from "./code-generator";
import { CodePatcher } from "./code-patcher";

// =========================
// 🔥 GLOBAL STORE (survives Next.js HMR reloads)
// =========================
const g = global as typeof global & {
  _aiFileStore?: Map<string, AIFileNode[]>;
  _aiMetaStore?: Map<string, AIProject>;
};

if (!g._aiFileStore) g._aiFileStore = new Map<string, AIFileNode[]>();
if (!g._aiMetaStore) g._aiMetaStore = new Map<string, AIProject>();

const inMemoryProjects = g._aiFileStore;
const projectMetadata = g._aiMetaStore;

export class AIOrchestrator {
  // =========================
  // 🚀 CREATE PROJECT (OPTIONAL USE)
  // =========================
  static createProject(
    projectId: string,
    prompt: string,
    plan?: AIProjectPlan
  ): AIFileNode[] {
    try {
      let files: AIFileNode[];

      if (plan) {
        files = CodeGenerator.generateFromPlan(projectId, plan);
      } else {
        files = CodeGenerator.generate(projectId, prompt);
      }

      inMemoryProjects.set(projectId, files);

      projectMetadata.set(projectId, {
        id: projectId,
        name: plan?.projectName || prompt.slice(0, 50),
        subaccountId: "",
        files,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      });

      return files;
    } catch (error: unknown) {
      const message = error instanceof Error ? error.message : String(error);
      throw new AIProjectError(
        `Failed to create project: ${message}`,
        "CREATE_FAILED",
        500
      );
    }
  }

  // =========================
  // 🔥 DIRECT SET (used by generate route)
  // =========================
  static setProjectFiles(projectId: string, files: AIFileNode[]) {
    if (!Array.isArray(files) || files.length === 0) {
      throw new AIProjectError(
        "Invalid files provided",
        "INVALID_FILES",
        400
      );
    }

    inMemoryProjects.set(projectId, files);

    projectMetadata.set(projectId, {
      id: projectId,
      name: "AI Project",
      subaccountId: "",
      files,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
    });

    console.log(`✅ Project ${projectId} stored with ${files.length} files`);
    console.log(`📦 Total projects in store: ${inMemoryProjects.size}`);
  }

  // =========================
  // ✏️ EDIT PROJECT
  // =========================
  static async editProject(
    projectId: string,
    instructions: string,
    useAI: boolean = false,
    aiClient?: any
  ): Promise<AIFileNode[]> {
    try {
      const existing = inMemoryProjects.get(projectId);

      if (!existing) {
        console.error(
          `❌ Project ${projectId} not found. Store has ${inMemoryProjects.size} projects:`,
          Array.from(inMemoryProjects.keys())
        );
        throw new AIProjectError(
          `Project ${projectId} not found`,
          "PROJECT_NOT_FOUND",
          404
        );
      }

      let updated: AIFileNode[];

      if (useAI && aiClient) {
        updated = await CodePatcher.patchWithAI(existing, instructions, aiClient);
      } else {
        updated = CodePatcher.patch(existing, instructions);
      }

      inMemoryProjects.set(projectId, updated);

      const metadata = projectMetadata.get(projectId);
      if (metadata) {
        metadata.updatedAt = new Date().toISOString();
        metadata.files = updated;
        projectMetadata.set(projectId, metadata);
      }

      return updated;
    } catch (error: unknown) {
      if (error instanceof AIProjectError) throw error;
      const message = error instanceof Error ? error.message : String(error);
      throw new AIProjectError(
        `Failed to edit project: ${message}`,
        "EDIT_FAILED",
        500
      );
    }
  }

  // =========================
  // 🔄 UPDATE FILES (SAFE MERGE)
  // =========================
  static updateProjectFiles(
    projectId: string,
    newFiles: AIFileNode[]
  ): AIFileNode[] {
    try {
      const existing = inMemoryProjects.get(projectId);

      if (!existing) {
        throw new AIProjectError(
          `Project ${projectId} not found`,
          "PROJECT_NOT_FOUND",
          404
        );
      }

      const fileMap = new Map<string, AIFileNode>();

      existing.forEach((f) => fileMap.set(f.path, f));
      newFiles.forEach((f) => {
        if (f.path && typeof f.content === "string") {
          fileMap.set(f.path, f);
        }
      });

      const updated = Array.from(fileMap.values());

      inMemoryProjects.set(projectId, updated);

      const metadata = projectMetadata.get(projectId);
      if (metadata) {
        metadata.updatedAt = new Date().toISOString();
        metadata.files = updated;
        projectMetadata.set(projectId, metadata);
      }

      console.log(`✅ Updated project ${projectId} with ${newFiles.length} new files`);
      return updated;
    } catch (error: unknown) {
      if (error instanceof AIProjectError) throw error;
      const message = error instanceof Error ? error.message : String(error);
      throw new AIProjectError(
        `Failed to update files: ${message}`,
        "UPDATE_FAILED",
        500
      );
    }
  }

  // =========================
  // 📂 GET FILES
  // =========================
  static getProjectFiles(projectId: string): AIFileNode[] {
    const files = inMemoryProjects.get(projectId);

    if (!files) {
      console.error(
        `❌ Project ${projectId} not found. Store has ${inMemoryProjects.size} projects:`,
        Array.from(inMemoryProjects.keys())
      );
      throw new AIProjectError(
        `Project ${projectId} not found`,
        "PROJECT_NOT_FOUND",
        404
      );
    }

    return files;
  }

  // =========================
  // 📊 METADATA
  // =========================
  static getProjectMetadata(projectId: string): AIProject | undefined {
    return projectMetadata.get(projectId);
  }

  static getAllProjects(): AIProject[] {
    return Array.from(projectMetadata.values());
  }

  // =========================
  // ❌ DELETE
  // =========================
  static deleteProject(projectId: string): boolean {
    const deleted = inMemoryProjects.delete(projectId);
    projectMetadata.delete(projectId);
    return deleted;
  }

  // =========================
  // ✅ VALIDATE
  // =========================
  static validateProject(files: AIFileNode[]): boolean {
    const validation = CodePatcher.validateFiles(files);

    if (!validation.isValid) {
      console.error("❌ Validation failed:", validation.errors);
      return false;
    }

    if (validation.warnings.length > 0) {
      console.warn("⚠️ Warnings:", validation.warnings);
    }

    return true;
  }

  // =========================
  // 📈 STATS
  // =========================
  static getProjectStats(projectId: string) {
    const files = this.getProjectFiles(projectId);

    const fileTypes: Record<string, number> = {};
    let totalSize = 0;

    files.forEach((file) => {
      const ext = file.path.split(".").pop() || "unknown";
      fileTypes[ext] = (fileTypes[ext] || 0) + 1;
      totalSize += new Blob([file.content]).size;
    });

    return {
      fileCount: files.length,
      totalSize,
      fileTypes,
    };
  }
}

import { planProject } from "@/lib/ai/llm/pipeline";

export interface BasicPlan {
  name: string;
  pages: string[];
}

export class ProjectPlanner {
  // ─────────────────────────────────────────────
  // Full AI plan via kimi-k2-instruct (Stage 1)
  // Returns the full ProjectSpec used by the pipeline,
  // or null if planning fails.
  // ─────────────────────────────────────────────
  static async planFromPrompt(prompt: string) {
    return planProject(prompt);
  }

  // ─────────────────────────────────────────────
  // Quick local heuristic — zero-latency fallback.
  // Used when you need an instant structural hint
  // without waiting for a network call.
  // ─────────────────────────────────────────────
  static plan(prompt: string): BasicPlan {
    const lower = prompt.toLowerCase();

    if (lower.includes("landing")) {
      return {
        name: "ai-landing-app",
        pages: ["home", "pricing", "contact"],
      };
    }

    if (lower.includes("dashboard")) {
      return {
        name: "ai-dashboard-app",
        pages: ["dashboard", "analytics", "settings"],
      };
    }

    if (lower.includes("blog")) {
      return {
        name: "ai-blog-app",
        pages: ["home", "post", "about"],
      };
    }

    if (lower.includes("portfolio")) {
      return {
        name: "ai-portfolio-app",
        pages: ["home", "projects", "contact"],
      };
    }

    if (lower.includes("ecommerce") || lower.includes("shop")) {
      return {
        name: "ai-shop-app",
        pages: ["home", "products", "cart", "checkout"],
      };
    }

    return {
      name: "ai-generated-app",
      pages: ["home"],
    };
  }
}

import { NextRequest } from "next/server";
import { db } from "@/lib/db";
import { streamProjectPipeline } from "@/lib/ai/stream/stream-pipeline";

type FileType = {
  path: string;
  content: string;
};

export async function POST(req: NextRequest) {
  const { prompt, subAccountId } = await req.json();

  const project = await db.aIProject.create({
    data: {
      name: prompt.slice(0, 40),
      subAccountId,
    },
  });

  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      // ✅ SEND INITIAL EVENT (prevents blocking)
      controller.enqueue(
        encoder.encode(
          `data: ${JSON.stringify({ type: "start" })}\n\n`
        )
      );

      try {
        // 🚀 DO NOT AWAIT → allow streaming
        streamProjectPipeline({
          prompt,

          onFile: async (file: FileType) => {
            await db.aIProjectFile.create({
              data: {
                projectId: project.id,
                path: file.path,
                content: file.content,
              },
            });

            controller.enqueue(
              encoder.encode(
                `data: ${JSON.stringify({
                  type: "file",
                  file,
                })}\n\n`
              )
            );
          },

          onDone: () => {
            controller.enqueue(
              encoder.encode(
                `data: ${JSON.stringify({ type: "done" })}\n\n`
              )
            );
            controller.close();
          },
        }).catch((err) => {
          console.error("❌ Pipeline Error:", err);
          controller.close();
        });

      } catch (err) {
        console.error("❌ Stream Error:", err);
        controller.close();
      }
    },
  });

  return new Response(stream, {
    headers: {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache",
      Connection: "keep-alive",
    },
  });
}

import { planProject } from "@/lib/ai/llm/plan-project";
import { generateFilesInParallel } from "@/lib/ai/llm/deepseek-generator";

type FileType = {
  path: string;
  content: string;
};

export async function streamProjectPipeline({
  prompt,
  onFile,
  onDone,
}: {
  prompt: string;
  onFile: (file: FileType) => Promise<void>;
  onDone: () => void;
}) {
  const spec = await planProject(prompt);

  if (!spec || !spec.files) {
    onDone();
    return;
  }

  const queue = [...spec.files];

  const workers = Array(3)
    .fill(null)
    .map(async () => {
      while (queue.length) {
        const file = queue.shift();
        if (!file) break;

        const result = await generateFilesInParallel({
          ...spec,
          files: [file],
        });

        const generated = result[0];

        await onFile({
          path: generated.path,
          content: generated.content,
        });
      }
    });

  await Promise.all(workers);

  onDone();
}

// src/lib/ai/llm/pipeline.ts
//
// Full multi-model pipeline — 5 stages for generation, 3 stages for editing.
//
//  Stage 1 — kimi-k2-instruct       → JSON project spec (planner)
//  Stage 2 — your code              → split spec into per-file tasks
//  Stage 3 — deepseek-v3.2          → parallel codegen (one call per file)
//  Stage 4 — phi-3.5-mini-instruct  → syntax fix on broken JSX only
//  Stage 5 — your code              → assemble + ensureBoilerplate
//
//  Edit path (only affected files):
//  kimi → identifies which files change
//  deepseek → rewrites them in parallel
//  phi → validates the edited files
//

import OpenAI from "openai";
import { AIFileNode } from "../types/ai-project.types";
import {
  generateFilesInParallel,
  rewriteFilesWithDeepseek,
  type ProjectSpec,
} from "./deepseek-generator";
import { validateFilesWithPhi } from "./phi-fixer";

const client = new OpenAI({
  apiKey: process.env.NVIDIA_API_KEY!,
  baseURL: "https://integrate.api.nvidia.com/v1",
});

// ─────────────────────────────────────────────
// STAGE 1 — PLANNER  (kimi-k2-instruct)
// Small output, pure JSON. kimi excels at structured planning.
// max_tokens: 2000  temp: 0.2
// ─────────────────────────────────────────────

export async function planProject(
  userPrompt: string
): Promise<ProjectSpec | null> {
  console.log("🧠 [STAGE 1] kimi-k2-instruct — planning");

  try {
    const completion = await client.chat.completions.create({
      model: "moonshotai/kimi-k2-instruct",
      temperature: 0.2,
      max_tokens: 2000,
      messages: [
        {
          role: "system",
          content: `You are a React architect. Return ONLY raw JSON — no markdown, no backticks, no explanation. Start with { and end with }.

{
  "projectName": "PascalCase no spaces",
  "description": "one sentence",
  "globalContext": "2-3 sentences: shared design system, color palette, font, spacing used across all files",
  "files": [
    {
      "path": "src/components/Header.jsx",
      "description": "sticky nav with logo, links, CTA button — glassmorphism style",
      "dependencies": []
    }
  ]
}

Rules:
- Always include: src/App.jsx, src/App.css, src/main.jsx, index.html, package.json, vite.config.js
- One component per file; break every visual section into its own component
- No react-router-dom; no libraries besides react and react-dom
- For a landing page: 8–15 component files. For a large site: up to 30 files
- Each description must be specific enough to code from on its own
- List ALL files you want generated`,
        },
        {
          role: "user",
          content: userPrompt,
        },
      ],
    });

    let text = completion.choices[0]?.message?.content || "";
    text = text.replace(/```json/gi, "").replace(/```/g, "").trim();

    const start = text.indexOf("{");
    const end = text.lastIndexOf("}");

    if (start === -1 || end === -1) {
      console.error("❌ [STAGE 1] No JSON object found in planner response");
      return null;
    }

    try {
      const spec = JSON.parse(text.slice(start, end + 1)) as ProjectSpec;
      console.log(
        `✅ [STAGE 1] Plan: ${spec.files.length} files — ${spec.files.map((f) => f.path).join(", ")}`
      );
      return spec;
    } catch {
      // Attempt trailing-comma repair
      const fixed = text
        .slice(start, end + 1)
        .replace(/,(\s*[}\]])/g, "$1");
      const spec = JSON.parse(fixed) as ProjectSpec;
      console.log(`✅ [STAGE 1] Plan recovered: ${spec.files.length} files`);
      return spec;
    }
  } catch (err) {
    console.error("❌ [STAGE 1] Planner failed:", err);
    return null;
  }
}

// ─────────────────────────────────────────────
// STAGE 2 — COMPONENT SPLITTER  (your code)
// Splits the JSON spec into per-file task descriptors.
// No API call. Validates the spec and normalises paths.
// ─────────────────────────────────────────────

function splitPlanIntoTasks(spec: ProjectSpec): ProjectSpec {
  console.log(`📦 [STAGE 2] Splitting ${spec.files.length} file tasks`);

  const normalised: ProjectSpec = {
    ...spec,
    files: spec.files.map((f) => ({
      path: f.path.replace(/^\/+/, ""), // strip leading slashes
      description: f.description || `${f.path} component`,
      dependencies: Array.isArray(f.dependencies) ? f.dependencies : [],
    })),
  };

  console.log(
    `✅ [STAGE 2] Tasks ready: ${normalised.files.map((f) => f.path).join(", ")}`
  );
  return normalised;
}

// ─────────────────────────────────────────────
// STAGE 5 — ASSEMBLER  (your code)
// Merges generated files, writes package.json, vite.config,
// index.html, auto-builds App.jsx if missing, and — crucially —
// scans every JSX file for local CSS imports and stubs any that
// don't exist yet (prevents Vite "does the file exist?" crashes).
// ─────────────────────────────────────────────

/**
 * Parse all `import "./Foo.css"` and `import "../bar.css"` statements
 * from a JSX file and resolve them to project-relative paths.
 *
 * e.g. file at "src/components/Header.jsx" importing "./Header.css"
 *      → "src/components/Header.css"
 */
function extractLocalCssImports(file: AIFileNode): string[] {
  // Match: import "./Something.css"  or  import '../Something.css'
  const importRegex = /import\s+['"](\.[^'"]+\.css)['"]/g;
  const dir = file.path.split("/").slice(0, -1).join("/");
  const resolved: string[] = [];

  let match: RegExpExecArray | null;
  while ((match = importRegex.exec(file.content)) !== null) {
    const rel = match[1]; // e.g. "./Header.css"

    // Resolve relative to the file's directory
    const segments = `${dir}/${rel}`.split("/");
    const normalised: string[] = [];
    for (const seg of segments) {
      if (seg === "..") normalised.pop();
      else if (seg !== ".") normalised.push(seg);
    }
    resolved.push(normalised.join("/"));
  }
  return resolved;
}

/**
 * Generate a minimal stub CSS for a component based on its name.
 * Produces a scoped class reset + a couple of sensible defaults so
 * the component renders without layout breakage.
 */
function stubCss(cssPath: string): string {
  // Derive a BEM-style root class from the filename, e.g. "header"
  const name = cssPath
    .split("/")
    .pop()!
    .replace(".css", "")
    .toLowerCase();

  return `/* ${cssPath} — auto-generated stub */
.${name} {
  box-sizing: border-box;
}

.${name} * {
  box-sizing: border-box;
}
`;
}

function ensureBoilerplate(
  files: AIFileNode[],
  projectName: string
): AIFileNode[] {
  const map = new Map(files.map((f) => [f.path, f]));

  const ensure = (path: string, content: string) => {
    if (!map.has(path)) {
      map.set(path, { path, content, type: "file" });
      console.log(`➕ [STAGE 5] Ensured: ${path}`);
    }
  };

  ensure("package.json", makePackageJson(projectName));
  ensure("index.html", makeIndexHtml(projectName));
  ensure("src/main.jsx", makeMainJsx());
  ensure("vite.config.js", makeViteConfig());

  // Auto-build App.jsx from available components if missing
  if (!map.has("src/App.jsx")) {
    const components = Array.from(map.keys())
      .filter(
        (p) => p.startsWith("src/components/") && p.endsWith(".jsx")
      )
      .map((p) => ({
        name: p.split("/").pop()!.replace(".jsx", ""),
        importPath: "./" + p.replace("src/", "").replace(".jsx", ""),
      }));

    const imports = components
      .map((c) => `import ${c.name} from '${c.importPath}';`)
      .join("\n");
    const renders = components
      .map((c) => `      <${c.name} />`)
      .join("\n");

    map.set("src/App.jsx", {
      path: "src/App.jsx",
      content: `import React from 'react';\nimport './App.css';\n${imports}\n\nfunction App() {\n  return (\n    <div className="app">\n${renders}\n    </div>\n  );\n}\n\nexport default App;`,
      type: "file",
    });
    console.log("➕ [STAGE 5] Auto-built src/App.jsx");
  }

  if (!map.has("src/App.css")) {
    map.set("src/App.css", {
      path: "src/App.css",
      content: `* { margin: 0; padding: 0; box-sizing: border-box; }\nbody { font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif; }\n.app { min-height: 100vh; }`,
      type: "file",
    });
  }

  // ── CSS stub pass ─────────────────────────────────────────────────────
  // deepseek freely writes  import "./Header.css"  inside components but
  // the planner never lists those CSS files in the spec, so they're never
  // generated → Vite crashes with "does the file exist?".
  //
  // Solution: scan every JSX file, resolve every local .css import, and
  // create a stub for any path that isn't already in the map.
  // Zero extra API calls — pure string processing.
  const jsxFiles = Array.from(map.values()).filter((f) =>
    f.path.endsWith(".jsx")
  );

  let stubbedCount = 0;
  for (const jsx of jsxFiles) {
    const cssPaths = extractLocalCssImports(jsx);
    for (const cssPath of cssPaths) {
      if (!map.has(cssPath)) {
        map.set(cssPath, {
          path: cssPath,
          content: stubCss(cssPath),
          type: "file",
        });
        stubbedCount++;
        console.log(`🎨 [STAGE 5] Stubbed missing CSS: ${cssPath}`);
      }
    }
  }

  if (stubbedCount > 0) {
    console.log(`🎨 [STAGE 5] Created ${stubbedCount} CSS stub(s) for missing imports`);
  }
  // ─────────────────────────────────────────────────────────────────────

  const result = Array.from(map.values());
  console.log(`✅ [STAGE 5] Final project: ${result.length} files`);
  return result;
}

export async function generateProjectWithPipeline(
  userPrompt: string
): Promise<AIFileNode[]> {
  console.log("🚀 ═══════════════════════════════");
  console.log("🚀 PIPELINE START — GENERATION");
  console.log("🚀 ═══════════════════════════════");

  // ── Stage 1: Plan ──────────────────────────
  const rawSpec = await planProject(userPrompt);
  if (!rawSpec) {
    console.error("❌ Planning failed — using minimal fallback");
    return minimalFallback(userPrompt);
  }

  // ── Stage 2: Split ─────────────────────────
  const spec = splitPlanIntoTasks(rawSpec);

  // ── Stage 3: Parallel codegen (deepseek) ───
  const generated = await generateFilesInParallel(spec);

  // ── Stage 4: Validate / fix (phi) ──────────
  
const validated = generated;
  // ── Stage 5: Assemble ──────────────────────
  const result = ensureBoilerplate(validated, spec.projectName);

  console.log("✅ ═══════════════════════════════");
  console.log(`✅ PIPELINE DONE — ${result.length} files`);
  console.log("✅ ═══════════════════════════════");

  return result;
}

// ─────────────────────────────────────────────
// EDIT PIPELINE — 3-stage edit path
// kimi → identifies which files change
// deepseek → rewrites them in parallel
// phi → validates edited files
// ─────────────────────────────────────────────

export async function editProjectWithPipeline(
  instructions: string,
  existingFiles: AIFileNode[]
): Promise<AIFileNode[]> {
  console.log("✏️ ═══════════════════════════════");
  console.log("✏️ EDIT PIPELINE START");
  console.log("✏️ ═══════════════════════════════");

  const fileMap = new Map(existingFiles.map((f) => [f.path, f]));

  // ── Edit Stage 1: kimi identifies affected files ──
  console.log("🧠 [EDIT 1] kimi-k2-instruct — identifying files to change");

  let pathsToEdit: string[] = [];

  try {
    const fileList = existingFiles.map((f) => `  - ${f.path}`).join("\n");

    const planCompletion = await client.chat.completions.create({
      model: "moonshotai/kimi-k2-instruct",
      temperature: 0.2,
      max_tokens: 400,
      messages: [
        {
          role: "system",
          content: `You are a React architect. Return ONLY a raw JSON array of file paths that need to change.
Example: ["src/App.css", "src/components/Header.jsx"]
No markdown. No explanation. Start with [ and end with ].`,
        },
        {
          role: "user",
          content: `Edit request: "${instructions}"\n\nProject files:\n${fileList}`,
        },
      ],
    });

    let planText = planCompletion.choices[0]?.message?.content || "[]";
    planText = planText
      .replace(/```json/gi, "")
      .replace(/```/g, "")
      .trim();

    const s = planText.indexOf("[");
    const e = planText.lastIndexOf("]");
    if (s !== -1 && e !== -1) {
      pathsToEdit = JSON.parse(planText.slice(s, e + 1)) as string[];
    }

    console.log(`✅ [EDIT 1] kimi identified: ${pathsToEdit.join(", ")}`);
  } catch (err) {
    console.warn("⚠️ [EDIT 1] kimi failed — defaulting to App.jsx + App.css:", err);
    pathsToEdit = existingFiles.some((f) => f.path === "src/App.css")
      ? ["src/App.css", "src/App.jsx"]
      : ["src/App.jsx"];
  }

  if (pathsToEdit.length === 0) {
    console.warn("⚠️ [EDIT 1] Empty file list — defaulting to App.jsx + App.css");
    pathsToEdit = ["src/App.jsx", "src/App.css"];
  }

  // ── Edit Stage 2: deepseek rewrites affected files ──
  const filesToEdit = pathsToEdit
    .map((p) => fileMap.get(p))
    .filter((f): f is AIFileNode => !!f);

  const rewritten = await rewriteFilesWithDeepseek(instructions, filesToEdit);

  // ── Edit Stage 3: phi validates edited files ──
  const validated = await validateFilesWithPhi(rewritten);

  // Merge back into the full file list (unchanged files stay intact)
  validated.forEach((f) => fileMap.set(f.path, f));

  const result = Array.from(fileMap.values());

  console.log("✅ ═══════════════════════════════");
  console.log(`✅ EDIT DONE — ${validated.length} file(s) updated`);
  console.log("✅ ═══════════════════════════════");

  return result;
}

  { id: "ai-builder-run" }, // ✅ ONLY ONE CONFIG OBJECT

    const { prompt, projectId, subaccountId, isNewProject } =
      event.data as EventData;

    // -------------------------
    // USER MESSAGE
    // -------------------------
    await step.run("save-user-message", async () => {
      await db.aIMessage.create({
        data: {
          projectId,
          role: "USER",
          content: prompt,
          status: "DONE",
        },
      });

    // ASSISTANT PLACEHOLDER
    // -------------------------

          data: {

            content: "Building your app...",
            status: "PENDING",
          },
        });
        return msg.id;
      }
    );

    // -------------------------
    // PLAN
    // -------------------------
    const plan = await step.run("plan-project", async () => {
      return planProject(prompt);
    });

    // -------------------------
    // AI GENERATE
    // -------------------------
    const aiFiles: AIFile[] = await step.run(
      "generate-code",
      async () => {
        const input = plan ? JSON.stringify(plan, null, 2) : prompt;
        return generateCodeFromAI(input);
      }
    );

    // -------------------------
    // LOAD TEMPLATE
    // -------------------------
    const baseFiles: AIFile[] = await step.run(
      "load-template",
      async () => {
        return loadBaseTemplate();
      }
    );

    // -------------------------
    const finalFiles: AIFile[] = await step.run(
      "merge-files",
      async () => {
        return mergeFiles(baseFiles, aiFiles);
      }
    );

    // SAVE FILES
    // -------------------------
    await step.run("save-files", async () => {
      if (!finalFiles.length) return;

      await db.aIProjectFile.deleteMany({

      });

          status: "DONE",

        },

    });

    };
  }

  messages: Message[];

  setProject: (id: string) => void;
  setFiles: (files: AIFileNode[]) => void;

  messages: [],

      files,

    set({
      activeFilePath: path,
    }),

    const { files } = get();

    const updated = files.map((file) =>
      file.path === path ? { ...file, content } : file
    );

  },

    set((state) => ({
      messages: [...state.messages, msg],
    })),

  reset: () =>
    set({
      projectId: null,
      files: [],
      activeFilePath: null,
      messages: [],
    }),
}));

model User {
  id        String   @id @default(uuid())
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  name      String
  avatarUrl String  @db.Text
  email     String  @unique
  role      Role    @default(SUBACCOUNT_USER)
  agencyId  String?

  agency       Agency?        @relation(fields: [agencyId], references: [id], onDelete: Cascade)
  permissions  Permissions[]
  tickets      Ticket[]
  notification Notification[]

  @@index([agencyId])
}

model Payment {
  id        String   @id @default(uuid())
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  razorpayOrderId   String  @unique
  razorpayPaymentId String? @unique
  razorpaySignature String?

  amount   Int // in paise
  currency String @default("INR")

  platformFee    Int // 1% fee (paise)
  agencyEarnings Int // amount - fee (paise)

  status String @default("created") // created | paid | failed

  funnelId     String
  subAccountId String
  agencyId     String

  funnel     Funnel     @relation(fields: [funnelId], references: [id], onDelete: Cascade)
  subAccount SubAccount @relation(fields: [subAccountId], references: [id], onDelete: Cascade)
  agency     Agency     @relation(fields: [agencyId], references: [id], onDelete: Cascade)

  @@index([agencyId])
  @@index([subAccountId])
  @@index([funnelId])
}

model AIProject {
  id        String   @id @default(uuid())
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  name         String

  @@map("AIProject")

model AIProjectFile {
  id        String   @id @default(uuid())
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  path      String   @db.VarChar(500)
  content   String   @db.LongText
  projectId String

  project AIProject @relation(fields: [projectId], references: [id], onDelete: Cascade)

  @@map("AIProjectFile")
  @@unique([projectId, path])
  @@index([projectId])
}
