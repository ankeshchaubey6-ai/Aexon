"use client";

import {
  useEffect,
  useRef,
  useState,
  useCallback,
  useTransition,
} from "react";
import { useParams, useRouter } from "next/navigation";
import { Sparkles, Bot } from "lucide-react";

import { useAIBuilderStore } from "../state/ai-builder.store";
import { useWebContainer } from "../state/use-webcontainer";
import ChatInput from "./ChatInput";
import ChatMessage, { type AIMessage } from "./ChatMessage";

// ─────────────────────────────────────────────────────────────
// CONSTANTS
// ─────────────────────────────────────────────────────────────
const POLL_INTERVAL_MS = 2500;

const STARTERS = [
  "SaaS landing page with pricing table and hero",
  "Kanban board with drag and drop",
  "E-commerce store with cart and checkout",
  "Analytics dashboard with charts",
  "Blog with markdown support",
  "Portfolio with dark mode",
];

// ─────────────────────────────────────────────────────────────
// CHAT PANEL
// ─────────────────────────────────────────────────────────────
const ChatPanel = () => {
  const params = useParams();
  const router = useRouter();
  const subaccountId = params?.subaccountId as string;

  const { projectId, setProject, setFiles } = useAIBuilderStore();
  const { remount, mountAndRun, isRunning } = useWebContainer();

  const [messages, setMessages] = useState<AIMessage[]>([]);
  const [isGenerating, setIsGenerating] = useState(false);
  const [, startTransition] = useTransition();

  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null);
  const hasPendingRef = useRef(false);
  const prevFileCountRef = useRef(0);
  const bottomRef = useRef<HTMLDivElement>(null);

  const routeProjectId = params?.projectId as string | undefined;
  const effectiveProjectId =
    projectId ||
    (routeProjectId && routeProjectId !== "new" ? routeProjectId : null);

  // ── Auto-scroll ──────────────────────────────────────────
  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages.length]);

  // ── Load project files from DB ───────────────────────────
  const loadProjectFiles = useCallback(
    async (pid: string) => {
      try {
        const res = await fetch(`/api/ai/projects/${pid}`);
        if (!res.ok) return;
        const data = await res.json();
        const newFiles = data?.project?.files ?? [];
        if (!newFiles.length) return;
        if (newFiles.length === prevFileCountRef.current) return;
        prevFileCountRef.current = newFiles.length;
        setFiles(newFiles);
        if (isRunning) {
          remount(newFiles);
        } else {
          mountAndRun(newFiles);
        }
      } catch (err) {
        console.error("File load error:", err);
      }
    },
    [isRunning, remount, mountAndRun, setFiles]
  );

  // ── Poll messages ────────────────────────────────────────
  const pollMessages = useCallback(async () => {
    if (!effectiveProjectId) return;
    try {
      const res = await fetch(
        `/api/ai/messages?projectId=${effectiveProjectId}`
      );
      if (!res.ok) return;
      const data = await res.json();
      const fetched: AIMessage[] = data.messages ?? [];

      startTransition(() => setMessages(fetched));

      const hasPending = fetched.some((m) => m.status === "PENDING");

      if (hasPendingRef.current && !hasPending) {
        hasPendingRef.current = false;
        setIsGenerating(false);
        await loadProjectFiles(effectiveProjectId);
      }

      hasPendingRef.current = hasPending;
    } catch (err) {
      console.error("Poll error:", err);
    }
  }, [effectiveProjectId, loadProjectFiles]);

  useEffect(() => {
    if (!effectiveProjectId) return;
    pollMessages();
    pollRef.current = setInterval(pollMessages, POLL_INTERVAL_MS);
    return () => { if (pollRef.current) clearInterval(pollRef.current); };
  }, [effectiveProjectId, pollMessages]);

  // ── Send message ─────────────────────────────────────────
  const handleSend = useCallback(
    async (prompt: string) => {
      if (!prompt.trim() || isGenerating) return;

      setIsGenerating(true);
      hasPendingRef.current = true;

      const optUserId = `opt-user-${Date.now()}`;
      const optAsstId = `opt-asst-${Date.now()}`;

      setMessages((prev) => [
        ...prev,
        {
          id: optUserId,
          role: "USER",
          content: prompt,
          status: "DONE",
          createdAt: new Date().toISOString(),
        } as AIMessage,
        {
          id: optAsstId,
          role: "ASSISTANT",
          content: "Planning and generating your project...",
          status: "PENDING",
          createdAt: new Date().toISOString(),
        } as AIMessage,
      ]);

      try {
        const res = await fetch("/api/ai/chat", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            prompt: prompt.trim(),
            projectId: effectiveProjectId || "new",
            subaccountId,
          }),
        });

        if (!res.ok) {
          const err = await res.json().catch(() => ({}));
          throw new Error(err?.error || `HTTP ${res.status}`);
        }

        const data = await res.json();

        if (data.isNewProject && data.projectId) {
          setProject(data.projectId);
          router.replace(
            `/subaccount/${subaccountId}/ai-builder/${data.projectId}`
          );
        }
        // Polling picks up the result — nothing else needed here

      } catch (err) {
        setIsGenerating(false);
        hasPendingRef.current = false;
        setMessages((prev) =>
          prev.map((m) =>
            m.id === optAsstId
              ? {
                  ...m,
                  content: `Failed: ${err instanceof Error ? err.message : "Unknown error"}`,
                  status: "ERROR" as const,
                }
              : m
          )
        );
      }
    },
    [effectiveProjectId, subaccountId, isGenerating, setProject, router]
  );

  const isEmpty = messages.length === 0;

  return (
    <div className="flex h-full flex-col overflow-hidden">
      {/* Header */}
      <div className="flex items-center gap-2 border-b px-4 py-2.5 shrink-0">
        <div className="flex h-6 w-6 items-center justify-center rounded-full bg-primary/10">
          <Bot className="h-3.5 w-3.5 text-primary" />
        </div>
        <span className="text-sm font-medium">AI Builder</span>
        {isGenerating && (
          <span className="ml-auto flex items-center gap-1.5 text-xs text-muted-foreground">
            <span className="h-1.5 w-1.5 rounded-full bg-yellow-500 animate-pulse" />
            Generating...
          </span>
        )}
      </div>

      {/* Messages */}
      <div className="flex-1 overflow-y-auto min-h-0">
        {isEmpty ? (
          <EmptyState onSelect={handleSend} disabled={isGenerating} />
        ) : (
          <div className="py-2">
            {messages.map((msg) => (
              <ChatMessage key={msg.id} message={msg} />
            ))}
            <div ref={bottomRef} className="h-1" />
          </div>
        )}
      </div>

      {/* Input */}
      <div className="shrink-0 border-t p-3">
        <ChatInput
          onSend={handleSend}
          isLoading={isGenerating}
          disabled={!subaccountId}
          placeholder={
            effectiveProjectId ? "Request a change..." : "Describe your app..."
          }
        />
      </div>
    </div>
  );
};

export default ChatPanel;

"use client";

import React, { useState } from "react";
import { Button } from "@/components/ui/button";
import { ScrollArea } from "@/components/ui/scroll-area";
import MessageList from "./MessageList";
import { useParams } from "next/navigation";
import { useAIBuilderStore } from "../state/ai-builder.store";

const ChatSidebar: React.FC = () => {
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);

  const params = useParams();
  const subaccountId = params?.subaccountId as string;

  const { setProject, setFiles, addMessage, projectId, files } =
    useAIBuilderStore();

  const handleSend = async () => {
    if (!input.trim() || loading) return;

    const prompt = input.trim();
    const isEditMode = !!(projectId && files && files.length > 0);

    addMessage({ role: "user", content: prompt });
    addMessage({
      role: "assistant",
      content: isEditMode
        ? "Applying your changes... ⏳"
        : "Generating your project... ⏳",
    });

    setInput("");
    setLoading(true);

    try {
      const endpoint = isEditMode ? "/api/ai/edit" : "/api/ai/generate";

      const body = isEditMode
        ? { subaccountId, projectId, instructions: prompt }
        : { subaccountId, prompt, name: "AI Generated Project" };

      const res = await fetch(endpoint, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
      });

      if (!res.ok) {
        throw new Error(`HTTP ${res.status}`);
      }

      const data = await res.json();

      if (data?.projectId && data?.files?.length) {
        // Update store — LivePreview's useEffect picks up the diff and
        // writes only changed files into the running WebContainer via HMR.
        setProject(data.projectId);
        setFiles(data.files);

        addMessage({
          role: "assistant",
          content: isEditMode
            ? `Done! Changes applied to ${data.files.length} file(s) ✨`
            : "Project generated successfully 🚀",
        });
      } else {
        addMessage({
          role: "assistant",
          content: "AI returned an empty result ❌",
        });
      }
    } catch (error) {
      console.error("AI Error:", error);
      addMessage({
        role: "assistant",
        content: "Something went wrong. Please try again ❌",
      });
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="h-full flex flex-col">
      <div className="p-3 border-b text-sm font-medium">
        AI Chat{" "}
        {projectId && (
          <span className="text-xs text-muted-foreground ml-2">
            (Editing Mode)
          </span>
        )}
      </div>

      <ScrollArea className="flex-1 p-3">
        <MessageList />
      </ScrollArea>

      <div className="p-3 border-t flex gap-2">
        <input
          className="flex-1 border rounded-md px-2 py-1 text-sm bg-background"
          placeholder={
            projectId
              ? "Describe changes you want..."
              : "Describe the app you want to build..."
          }
          value={input}
          onChange={(e) => setInput(e.target.value)}
          disabled={loading}
          onKeyDown={(e) => {
            if (e.key === "Enter" && !e.shiftKey) {
              e.preventDefault();
              handleSend();
            }
          }}
        />

        <Button size="sm" onClick={handleSend} disabled={loading}>
          {loading ? "Processing..." : projectId ? "Edit" : "Generate"}
        </Button>
      </div>
    </div>
  );
};

export default ChatSidebar;

"use client";

import { useEffect, useRef, memo } from "react";
import { Loader2, Monitor, AlertCircle, RefreshCw } from "lucide-react";
import { useAIBuilderStore } from "../state/ai-builder.store";
import { useWebContainer } from "../state/use-webcontainer";

type AIFile = { path: string; content: string };

// ─────────────────────────────────────────────────────────────
// STATUS BADGE
// ─────────────────────────────────────────────────────────────
const StatusBadge = ({ status }: { status: string }) => {
  const isRunning = status === "running";
  const isError = status === "error" || status === "install-failed";
  const isLoading = !isRunning && !isError && status !== "idle";

  return (
    <span className="flex items-center gap-1.5 text-xs text-muted-foreground">
      {isRunning && (
        <span className="h-1.5 w-1.5 rounded-full bg-green-500 animate-pulse" />
      )}
      {isLoading && (
        <Loader2 className="h-3 w-3 animate-spin text-yellow-500" />
      )}
      {isError && (
        <AlertCircle className="h-3 w-3 text-red-500" />
      )}
      {status}
    </span>
  );
};

// ─────────────────────────────────────────────────────────────
// LIVE PREVIEW
// ─────────────────────────────────────────────────────────────
const LivePreview = memo(() => {
  const { files } = useAIBuilderStore();
  const { mountAndRun, writeFiles, remount, url, status } = useWebContainer();

  // Track boot state and previous files for diffing
  const isBootedRef = useRef(false);
  const prevFilesRef = useRef<AIFile[]>([]);
  const prevFilesMapRef = useRef<Map<string, string>>(new Map());

  useEffect(() => {
    if (!files.length) return;

    // ── First load: boot WebContainer with all files ──
    if (!isBootedRef.current) {
      isBootedRef.current = true;
      prevFilesRef.current = files;
      prevFilesMapRef.current = new Map(files.map((f) => [f.path, f.content]));
      mountAndRun(files);
      return;
    }

    // ── Subsequent updates: diff and patch only changed files ──
    const changed: AIFile[] = [];
    const newMap = new Map(files.map((f) => [f.path, f.content]));

    for (const file of files) {
      const prev = prevFilesMapRef.current.get(file.path);
      if (prev !== file.content) {
        changed.push(file);
      }
    }

    // Check for deleted files (path in prev but not in new)
    // WebContainers doesn't support delete easily — we skip this edge case
    // and rely on the file content changing to trigger HMR

    if (changed.length === 0) return;

    // If it's a structural change (package.json, vite.config) — full remount
    const needsRemount = changed.some(
      (f) =>
        f.path === "package.json" ||
        f.path === "vite.config.js" ||
        f.path === "vite.config.ts" ||
        f.path === "index.html"
    );

    if (needsRemount) {
      console.log("🔄 Structural change detected — remounting");
      remount(files);
    } else {
      console.log(`⚡ Patching ${changed.length} file(s):`, changed.map((f) => f.path));
      writeFiles(changed);
    }

    // Update refs
    prevFilesRef.current = files;
    prevFilesMapRef.current = newMap;
  }, [files, mountAndRun, writeFiles, remount]);

  // ── Manual refresh ──
  const handleRefresh = () => {
    if (!files.length) return;
    console.log("🔄 Manual remount triggered");
    isBootedRef.current = false;
    prevFilesRef.current = [];
    prevFilesMapRef.current = new Map();
    remount(files);
  };

  return (
    <div className="h-full flex flex-col bg-background">
      {/* Header */}
      <div className="flex items-center justify-between px-3 py-2 border-b shrink-0">
        <div className="flex items-center gap-2">
          <Monitor className="h-3.5 w-3.5 text-muted-foreground" />
          <span className="text-xs font-medium">Preview</span>
        </div>
        <div className="flex items-center gap-2">
          <StatusBadge status={status} />
          <button
            onClick={handleRefresh}
            disabled={!files.length}
            title="Force remount"
            className="p-1 rounded hover:bg-muted transition-colors disabled:opacity-40"
          >
            <RefreshCw className="h-3 w-3 text-muted-foreground" />
          </button>
        </div>
      </div>

      {/* Preview area */}
      <div className="flex-1 relative overflow-hidden">
        {url ? (
          <iframe
            src={url}
            className="absolute inset-0 w-full h-full border-0 bg-white"
            title="Live preview"
            // allow is required for some WebContainer features
            allow="cross-origin-isolated"
          />
        ) : (
          <PreviewPlaceholder status={status} />
        )}
      </div>
    </div>
  );
});

LivePreview.displayName = "LivePreview";
export default LivePreview;

const STAGE_MESSAGES: Record<string, string> = {
  idle:             "Generate a project to see the preview",
  booting:          "Booting WebContainer...",
  mounting:         "Mounting project files...",
  installing:       "Installing dependencies (this takes ~30s the first time)...",
  "install-failed": "Dependency installation failed",
  starting:         "Starting Vite dev server...",
  remounting:       "Remounting project...",
  error:            "Something went wrong — try refreshing",
};

function PreviewPlaceholder({ status }: { status: string }) {
  const message = STAGE_MESSAGES[status] ?? status;
  const isError = status === "error" || status === "install-failed";

  return (
    <div className="w-full h-full flex flex-col items-center justify-center gap-3 text-center px-6">
      {isError ? (
        <AlertCircle className="h-8 w-8 text-red-400" />
      ) : status === "idle" ? (
        <Monitor className="h-8 w-8 text-muted-foreground/40" />
      ) : (
        <Loader2 className="h-8 w-8 animate-spin text-primary/60" />
      )}
      <p className="text-sm text-muted-foreground max-w-xs">{message}</p>
      {status === "installing" && (
        <p className="text-xs text-muted-foreground/60">
          Dependencies are cached after the first run
        </p>
      )}
    </div>
  );
}

"use client";

import React, { useEffect, useState } from "react";
import { useAIBuilderStore } from "../state/ai-builder.store";

const MonacoEditor: React.FC = () => {
  const {
    files,
    activeFilePath,
    updateFile,
  } = useAIBuilderStore();

  const activeFile = files.find(
    (f) => f.path === activeFilePath
  );

  const [content, setContent] = useState("");

  useEffect(() => {
    if (activeFile) {
      setContent(activeFile.content);
    } else {
      setContent("");
    }
  }, [activeFilePath, activeFile?.content]);

  const handleChange = (val: string) => {
    setContent(val);
    if (activeFilePath) {
      updateFile(activeFilePath, val);
    }
  };

  return (
    <div className="h-full flex flex-col">
      <div className="p-3 border-b text-sm font-medium flex items-center justify-between">
        <span>Code Editor</span>
        <span className="text-xs text-muted-foreground">
          {activeFilePath || "No file selected"}
        </span>
      </div>

      <div className="flex-1 p-3">
        {activeFile ? (
          <textarea
            className="w-full h-full rounded-md border bg-background p-3 text-xs font-mono outline-none resize-none"
            value={content}
            onChange={(e) =>
              handleChange(e.target.value)
            }
          />
        ) : (
          <div className="w-full h-full flex items-center justify-center text-sm text-muted-foreground">
            Generate a project and select a file to start editing.
          </div>
        )}
      </div>
    </div>
  );
};

export default MonacoEditor;

"use client";

import { create } from "zustand";
import { AIFileNode } from "@/lib/ai/types/ai-project.types";

type Message = {
  role: "user" | "assistant";
  content: string;
};

interface AIBuilderState {
  projectId: string | null;
  files: AIFileNode[];
  activeFilePath: string | null;
  messages: Message[];

  setProject: (id: string) => void;
  setFiles: (files: AIFileNode[]) => void;
  setActiveFile: (path: string) => void;
  updateFile: (path: string, content: string) => void;

  addMessage: (msg: Message) => void;
  reset: () => void;
}

export const useAIBuilderStore = create<AIBuilderState>((set, get) => ({
  projectId: null,
  files: [],
  activeFilePath: null,
  messages: [],

  setProject: (id) =>
    set({
      projectId: id,
    }),

  setFiles: (files) =>
    set({
      files,
      activeFilePath: files.length ? files[0].path : null,
    }),

  setActiveFile: (path) =>
    set({
      activeFilePath: path,
    }),

  updateFile: (path, content) => {
    const { files } = get();

    const updated = files.map((file) =>
      file.path === path ? { ...file, content } : file
    );

    set({ files: updated });
  },

  addMessage: (msg) =>
    set((state) => ({
      messages: [...state.messages, msg],
    })),

  reset: () =>
    set({
      projectId: null,
      files: [],
      activeFilePath: null,
      messages: [],
    }),
}));
