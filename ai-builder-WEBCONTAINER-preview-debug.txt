"use client";

// src/components/modules/ai-builder/state/use-webcontainer.ts
//
// FIXES:
//  - remount() now WRITES changed files + restarts dev server instead of
//    running npm install again (was 30-60s, now ~3s)
//  - Added _writeAndRestart() for edit-mode updates
//  - Queue-based busy guard: if a mount is in progress, queue the next one
//    instead of silently dropping it
//  - Idempotent boot: reuses the WebContainer instance across hot reloads
//

import { useState, useCallback, useEffect } from "react";
import { WebContainer } from "@webcontainer/api";
import type { FileSystemTree } from "@webcontainer/api";

type AIFile = { path: string; content: string };

type WCStatus =
  | "idle"
  | "waiting-files"
  | "booting"
  | "mounting"
  | "installing"
  | "install-failed"
  | "starting"
  | "writing"
  | "restarting"
  | "running"
  | "error";

// ─── Global singleton ─────────────────────────────────────────────────────────

let _instance: WebContainer | null = null;
let _bootPromise: Promise<WebContainer> | null = null;
let _globalStatus: WCStatus = "idle";
let _globalUrl: string | null = null;
let _devProcess: Awaited<ReturnType<WebContainer["spawn"]>> | null = null;
let _installed = false;
let _busy = false;

// Queue: if remount arrives while busy, store it and run after current op
let _pendingFiles: AIFile[] | null = null;

const _listeners = new Set<() => void>();
function _notify() { _listeners.forEach((fn) => fn()); }
function _setStatus(s: WCStatus) { _globalStatus = s; _notify(); }
function _setUrl(u: string | null) { _globalUrl = u; _notify(); }

// ─── Reset ────────────────────────────────────────────────────────────────────

export function resetContainer() {
  if (_devProcess) {
    try { _devProcess.kill(); } catch {}
    _devProcess = null;
  }
  _installed = false;
  _busy = false;
  _pendingFiles = null;
  _setUrl(null);
  _setStatus("idle");
}

// ─── Boot ─────────────────────────────────────────────────────────────────────

async function bootOnce(): Promise<WebContainer> {
  if (_instance) return _instance;
  if (_bootPromise) return _bootPromise;
  _bootPromise = WebContainer.boot().then((wc) => { _instance = wc; return wc; });
  return _bootPromise;
}

// ─── Helpers ──────────────────────────────────────────────────────────────────

function buildFileTree(files: AIFile[]): FileSystemTree {
  const tree: FileSystemTree = {};
  for (const file of files) {
    const parts = file.path.replace(/^\//, "").split("/");
    let current: any = tree;
    parts.forEach((part, index) => {
      if (index === parts.length - 1) {
        current[part] = { file: { contents: file.content } };
      } else {
        if (!current[part]) current[part] = { directory: {} };
        current = current[part].directory;
      }
    });
  }
  return tree;
}

function hasRequiredFiles(files: AIFile[]): boolean {
  return (
    files.some((f) => f.path === "package.json") &&
    files.some((f) => f.path === "src/main.jsx") &&
    files.some((f) => f.path === "src/App.jsx")
  );
}

async function waitForFile(path: string) {
  if (!_instance) return;
  for (let i = 0; i < 50; i++) {
    try { await _instance.fs.readFile(path); return; } catch {}
    await new Promise((r) => setTimeout(r, 100));
  }
}

// ─── Kill dev server ──────────────────────────────────────────────────────────

async function killDevServer() {
  if (_devProcess) {
    try { _devProcess.kill(); } catch {}
    _devProcess = null;
    // Give the port a moment to free up
    await new Promise((r) => setTimeout(r, 300));
  }
}

// ─── Write files to existing FS (no remount, no reinstall) ───────────────────

async function _writeFilesToFs(files: AIFile[]): Promise<void> {
  if (!_instance) return;
  for (const file of files) {
    const path = file.path.replace(/^\//, "");
    const parts = path.split("/");
    if (parts.length > 1) {
      await _instance.fs.mkdir(parts.slice(0, -1).join("/"), { recursive: true });
    }
    await _instance.fs.writeFile(path, file.content);
  }
}

// ─── Start dev server and wait for server-ready ───────────────────────────────

async function _startDevServer(wc: WebContainer): Promise<void> {
  _setStatus("starting");

  const dev = await wc.spawn("npm", ["run", "dev"]);
  _devProcess = dev;

  dev.output.pipeTo(
    new WritableStream({ write(chunk) { process.stdout.write?.("[vite] " + chunk); } })
  );

  await new Promise<void>((resolve) => {
    let resolved = false;
    wc.on("server-ready", (_port, serverUrl) => {
      if (resolved) return;
      resolved = true;
      _setUrl(serverUrl);
      _setStatus("running");
      resolve();
    });
  });
}

// ─── FULL MOUNT + INSTALL + START (first generation) ─────────────────────────

async function _doMountAndRun(files: AIFile[]) {
  if (!hasRequiredFiles(files)) {
    _setStatus("waiting-files");
    return;
  }

  _setStatus("booting");
  _setUrl(null);

  await killDevServer();

  const wc = await bootOnce();

  _setStatus("mounting");
  await wc.mount(buildFileTree(files));

  await waitForFile("src/App.jsx");
  await waitForFile("src/main.jsx");

  _setStatus("installing");
  const install = await wc.spawn("npm", ["install"]);
  install.output.pipeTo(
    new WritableStream({ write(chunk) { process.stdout.write?.("[npm] " + chunk); } })
  );

  const exitCode = await install.exit;
  if (exitCode !== 0) {
    _setStatus("install-failed");
    return;
  }
  _installed = true;

  await _startDevServer(wc);
}

// ─── FAST REMOUNT: write changed files + restart dev server (NO npm install) ──
//
// This is the key fix. Previously remount() set _installed=false and ran the
// full pipeline again including npm install (30-60s). For edit-mode updates
// we only need to write the changed files and restart vite — ~3s total.
//

async function _doWriteAndRestart(files: AIFile[]) {
  if (!_instance || !_installed) {
    // If not installed yet, fall back to full mount
    return _doMountAndRun(files);
  }

  if (!hasRequiredFiles(files)) {
    _setStatus("waiting-files");
    return;
  }

  _setStatus("writing");
  await killDevServer();

  // Write all files to the existing FS
  await _writeFilesToFs(files);

  _setStatus("restarting");
  const wc = _instance;
  await _startDevServer(wc);
}

// ─── Busy guard with queue ────────────────────────────────────────────────────

async function _runExclusive(fn: () => Promise<void>) {
  if (_busy) {
    // Queue the latest request — earlier ones are stale
    _pendingFiles = null; // will be set by caller if needed
    return;
  }

  _busy = true;
  try {
    await fn();
  } catch (err) {
    console.error("[WebContainer]", err);
    _setStatus("error");
  } finally {
    _busy = false;
    _notify();
  }
}

// ─── Hook ─────────────────────────────────────────────────────────────────────

export const useWebContainer = () => {
  const [, rerender] = useState(0);

  useEffect(() => {
    const listener = () => rerender((n) => n + 1);
    _listeners.add(listener);
    return () => { _listeners.delete(listener); };
  }, []);

  /**
   * First-time generation: full mount + npm install + vite start.
   * Only call this once per project, not on edits.
   */
  const mountAndRun = useCallback(async (files: AIFile[]) => {
    await _runExclusive(() => _doMountAndRun(files));
  }, []);

  /**
   * Edit-mode update: write changed files + restart vite.
   * Skips npm install — ~3s instead of 30-60s.
   * Falls back to full mount if WebContainer isn't installed yet.
   */
  const remount = useCallback(async (files: AIFile[]) => {
    await _runExclusive(() => _doWriteAndRestart(files));
  }, []);

  /**
   * Write files without restarting the dev server.
   * Useful for hot-reload-friendly updates (CSS, data files).
   */
  const writeFiles = useCallback(async (files: AIFile[]) => {
    if (!_instance) return;
    await _writeFilesToFs(files);
  }, []);

  return {
    status: _globalStatus,
    url: _globalUrl,
    isRunning: _globalStatus === "running",
    isInstalled: _installed,
    mountAndRun,
    remount,
    writeFiles,
  };
};

// src/components/modules/ai-builder/preview/LivePreview.tsx

import { useEffect, useRef, memo } from "react";
import { Loader2, Monitor, AlertCircle, RefreshCw } from "lucide-react";
import { useAIBuilderStore } from "../state/ai-builder.store";
import { useWebContainer } from "../state/use-webcontainer";

type AIFile = { path: string; content: string };

const StatusBadge = ({ status }: { status: string }) => {
  const isRunning = status === "running";
  const isError = status === "error" || status === "install-failed";
  const isLoading = !isRunning && !isError && status !== "idle" && status !== "waiting";

  return (
    <span className="flex items-center gap-1.5 text-xs text-muted-foreground">
      {isRunning && <span className="h-1.5 w-1.5 rounded-full bg-green-500 animate-pulse" />}
      {isLoading && <Loader2 className="h-3 w-3 animate-spin text-yellow-500" />}
      {isError && <AlertCircle className="h-3 w-3 text-red-500" />}
      {status}
    </span>
  );
};

// projectId prop is used purely to detect project switches and reset boot state.
const LivePreview = memo(({ projectId }: { projectId: string | null }) => {
  const { files } = useAIBuilderStore();
  const { mountAndRun, writeFiles, remount, url, status } = useWebContainer();

  const isBootedRef = useRef(false);
  const prevFilesMapRef = useRef<Map<string, string>>(new Map());
  // Track which project we last booted for so we reset on switch.
  const bootedProjectIdRef = useRef<string | null>(null);

  useEffect(() => {
    if (!files.length) return;

    const hasPackageJson = files.some((f) => f.path === "package.json");
    if (!hasPackageJson) {
      console.log("⏳ [LivePreview] Waiting for package.json...");
      return;
    }

    // ── Project switch detected: reset boot state ──────────────────────
    // When AIWorkspace switches projects it calls resetContainer() which
    // clears _globalUrl to null. We mirror that here by resetting our refs
    // whenever the projectId prop changes.
    if (projectId !== bootedProjectIdRef.current) {
      console.log(`🔁 [LivePreview] Project changed → ${projectId}, resetting boot state`);
      isBootedRef.current = false;
      prevFilesMapRef.current = new Map();
      bootedProjectIdRef.current = projectId;
    }

    // ── First boot for this project ────────────────────────────────────
    if (!isBootedRef.current) {
      isBootedRef.current = true;
      prevFilesMapRef.current = new Map(files.map((f) => [f.path, f.content]));
      mountAndRun(files);
      return;
    }

    // ── Diff and patch ─────────────────────────────────────────────────
    const changed: AIFile[] = [];
    const newMap = new Map(files.map((f) => [f.path, f.content]));

    for (const file of files) {
      const prev = prevFilesMapRef.current.get(file.path);
      if (prev !== file.content) changed.push(file);
    }

    // Also catch newly added files that didn't exist before
    for (const file of files) {
      if (!prevFilesMapRef.current.has(file.path)) {
        if (!changed.find((c) => c.path === file.path)) {
          changed.push(file);
        }
      }
    }

    if (changed.length === 0) return;

    const needsRemount = changed.some(
      (f) =>
        f.path === "package.json" ||
        f.path === "vite.config.js" ||
        f.path === "vite.config.ts" ||
        f.path === "index.html"
    );

    if (needsRemount) {
      console.log("🔄 Structural change detected — remounting");
      remount(files);
    } else {
      console.log(`⚡ Patching ${changed.length} file(s):`, changed.map((f) => f.path));
      writeFiles(changed);
    }

    prevFilesMapRef.current = newMap;
  }, [files, projectId, mountAndRun, writeFiles, remount]);

  const handleRefresh = () => {
    if (!files.length) return;
    console.log("🔄 Manual remount triggered");
    isBootedRef.current = false;
    prevFilesMapRef.current = new Map();
    bootedProjectIdRef.current = null;
    remount(files);
  };

  return (
    <div className="h-full flex flex-col bg-background">
      <div className="flex items-center justify-between px-3 py-2 border-b shrink-0">
        <div className="flex items-center gap-2">
          <Monitor className="h-3.5 w-3.5 text-muted-foreground" />
          <span className="text-xs font-medium">Preview</span>
        </div>
        <div className="flex items-center gap-2">
          <StatusBadge status={status} />
          <button
            onClick={handleRefresh}
            disabled={!files.length}
            title="Force remount"
            className="p-1 rounded hover:bg-muted transition-colors disabled:opacity-40"
          >
            <RefreshCw className="h-3 w-3 text-muted-foreground" />
          </button>
        </div>
      </div>

      <div className="flex-1 relative overflow-hidden">
        {url ? (
          <iframe
            src={url}
            className="absolute inset-0 w-full h-full border-0 bg-white"
            title="Live preview"
            allow="cross-origin-isolated"
          />
        ) : (
          <PreviewPlaceholder status={status} />
        )}
      </div>
    </div>
  );
});

// src/lib/ai/preview/sandbox-manager.ts
import path from "path";
import fs from "fs/promises";
import { AIFileNode } from "../types/ai-project.types";

const SANDBOX_ROOT = path.join(process.cwd(), "public", "sandbox");

export const SandboxManager = {
  async ensureProjectDir(projectId: string) {
    const projectDir = path.join(SANDBOX_ROOT, projectId);
    await fs.mkdir(projectDir, { recursive: true });
    return projectDir;
  },

  async writeFiles(projectId: string, files: AIFileNode[]) {
    const projectDir = await this.ensureProjectDir(projectId);

    await Promise.all(
      files.map(async (file) => {
        if (file.type === "folder") return;

        const filePath = path.join(projectDir, file.path);
        const dir = path.dirname(filePath);

        await fs.mkdir(dir, { recursive: true });
        await fs.writeFile(filePath, file.content, "utf-8");
      })
    );

    return `/sandbox/${projectId}/index.html`;
  },

  async clearProject(projectId: string) {
    const projectDir = path.join(SANDBOX_ROOT, projectId);
    await fs.rm(projectDir, { recursive: true, force: true });
  },
};

"use client";

// src/components/modules/ai-builder/state/ai-builder.store.ts
// UPGRADED: undo/redo history stack, session messages separate from archived history

import { create } from "zustand";
import { AIFileNode } from "@/lib/ai/types/ai-project.types";

type Message = {
  role: "user" | "assistant";
  content: string;
};

// A saved snapshot for undo/redo
type Snapshot = {
  files: AIFileNode[];
  label: string; // e.g. "Applied: dark mode"
};

interface AIBuilderState {
  // Project
  projectId: string | null;
  files: AIFileNode[];
  activeFilePath: string | null;

  // Current session messages (cleared when session archived)
  messages: Message[];

  // Undo / Redo stacks
  undoStack: Snapshot[];
  redoStack: Snapshot[];
  canUndo: boolean;
  canRedo: boolean;

  // Actions
  setProject: (id: string) => void;

  /**
   * setFiles — replaces all files and pushes a snapshot onto the undo stack.
   * @param files - new file array
   * @param label - human-readable label for the undo entry (e.g. prompt summary)
   */
  setFiles: (files: AIFileNode[], label?: string) => void;

  setActiveFile: (path: string) => void;

  /**
   * updateFile — patch a single file's content in-place, also undo-able.
   */
  updateFile: (path: string, content: string) => void;

  addMessage: (msg: Message) => void;
  clearMessages: () => void;

  undo: () => void;
  redo: () => void;

  reset: () => void;
}

const MAX_HISTORY = 30;

export const useAIBuilderStore = create<AIBuilderState>((set, get) => ({
  projectId: null,
  files: [],
  activeFilePath: null,
  messages: [],
  undoStack: [],
  redoStack: [],
  canUndo: false,
  canRedo: false,

  setProject: (id) => set({ projectId: id }),

  setFiles: (files, label = "Updated files") => {
    const { files: prev, undoStack } = get();

    // Push current state onto undo stack (cap at MAX_HISTORY)
    const newUndo: Snapshot[] = [
      ...undoStack.slice(-(MAX_HISTORY - 1)),
      { files: prev, label },
    ];

    set({
      files,
      activeFilePath: files.length ? files[0].path : null,
      undoStack: newUndo,
      redoStack: [], // new change clears redo
      canUndo: newUndo.length > 0,
      canRedo: false,
    });
  },

  setActiveFile: (path) => set({ activeFilePath: path }),

  updateFile: (path, content) => {
    const { files, undoStack } = get();
    const newUndo: Snapshot[] = [
      ...undoStack.slice(-(MAX_HISTORY - 1)),
      { files, label: `Edited ${path.split("/").pop()}` },
    ];
    const updated = files.map((f) => (f.path === path ? { ...f, content } : f));
    set({
      files: updated,
      undoStack: newUndo,
      redoStack: [],
      canUndo: true,
      canRedo: false,
    });
  },

  addMessage: (msg) =>
    set((state) => ({ messages: [...state.messages, msg] })),

  clearMessages: () => set({ messages: [] }),

  undo: () => {
    const { undoStack, redoStack, files } = get();
    if (!undoStack.length) return;

    const snapshot = undoStack[undoStack.length - 1];
    const newUndo = undoStack.slice(0, -1);
    const newRedo: Snapshot[] = [...redoStack, { files, label: snapshot.label }];

    set({
      files: snapshot.files,
      activeFilePath: snapshot.files.length ? snapshot.files[0].path : null,
      undoStack: newUndo,
      redoStack: newRedo,
      canUndo: newUndo.length > 0,
      canRedo: true,
    });
  },

  redo: () => {
    const { undoStack, redoStack, files } = get();
    if (!redoStack.length) return;

    const snapshot = redoStack[redoStack.length - 1];
    const newRedo = redoStack.slice(0, -1);
    const newUndo: Snapshot[] = [...undoStack, { files, label: snapshot.label }];

    set({
      files: snapshot.files,
      activeFilePath: snapshot.files.length ? snapshot.files[0].path : null,
      undoStack: newUndo,
      redoStack: newRedo,
      canUndo: true,
      canRedo: newRedo.length > 0,
    });
  },

  reset: () =>
    set({
      projectId: null,
      files: [],
      activeFilePath: null,
      messages: [],
      undoStack: [],
      redoStack: [],
      canUndo: false,
      canRedo: false,
    }),
}));

import { AIFileNode } from "../types/ai-project.types";

const fileStorage = new Map<string, AIFileNode[]>();

export class FileTreeService {
  static save(projectId: string, files: AIFileNode[]) {
    fileStorage.set(projectId, files);
  }

  static load(projectId: string): AIFileNode[] {
    return fileStorage.get(projectId) || [];
  }

  static update(
    projectId: string,
    files: AIFileNode[]
  ) {
    fileStorage.set(projectId, files);
  }

  static exists(projectId: string): boolean {
    return fileStorage.has(projectId);
  }
}

// src/lib/ai/engine/project-planner.ts
//
// Stage 1 wrapper — delegates to the pipeline's kimi-k2-instruct planner.
// Kept as a class so the rest of the codebase can call
// ProjectPlanner.planFromPrompt() without importing pipeline.ts directly.
//

import { planProject } from "@/lib/ai/llm/pipeline";

export interface BasicPlan {
  name: string;
  pages: string[];
}

export class ProjectPlanner {
  // ─────────────────────────────────────────────
  // Full AI plan via kimi-k2-instruct (Stage 1)
  // Returns the full ProjectSpec used by the pipeline,
  // or null if planning fails.
  // ─────────────────────────────────────────────
  static async planFromPrompt(prompt: string) {
    return planProject(prompt);
  }

  // ─────────────────────────────────────────────
  // Quick local heuristic — zero-latency fallback.
  // Used when you need an instant structural hint
  // without waiting for a network call.
  // ─────────────────────────────────────────────
  static plan(prompt: string): BasicPlan {
    const lower = prompt.toLowerCase();

    if (lower.includes("landing")) {
      return {
        name: "ai-landing-app",
        pages: ["home", "pricing", "contact"],
      };
    }

    if (lower.includes("dashboard")) {
      return {
        name: "ai-dashboard-app",
        pages: ["dashboard", "analytics", "settings"],
      };
    }

    if (lower.includes("blog")) {
      return {
        name: "ai-blog-app",
        pages: ["home", "post", "about"],
      };
    }

//  Stage 3 — deepseek-v3.2       → parallel codegen (Promise.all)

import { NextRequest, NextResponse } from "next/server";
import { currentUser } from "@clerk/nextjs/server";
import { Role } from "@prisma/client";

import { getAuthUserDetails } from "@/queries/auth";

import { generateProjectWithPipeline } from "@/lib/ai/llm/pipeline";
import { AIFileNode } from "@/lib/ai/types/ai-project.types";

export async function POST(req: NextRequest) {
  try {
    // ── Auth ────────────────────────────────────────────────────────────
    const clerkUser = await currentUser();
    if (!clerkUser) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    }

    // ── Body ─────────────────────────────────────────────────────────────
    const body = await req.json();
    const { subaccountId, prompt, name } = body as {
      subaccountId?: string;
      prompt?: string;
      name?: string;
    };

    if (!subaccountId || !prompt) {
      return NextResponse.json(
        { error: "Missing required fields: subaccountId, prompt" },
        { status: 400 }
      );
    }

    // ── Permission check ─────────────────────────────────────────────────
    const authUser = await getAuthUserDetails();
    if (!authUser || !authUser.agency) {
      return NextResponse.json({ error: "User not found" }, { status: 401 });
    }

    const role = clerkUser.privateMetadata.role as Role | undefined;
    const isOwnerOrAdmin =
      role === Role.AGENCY_OWNER || role === Role.AGENCY_ADMIN;

    if (!isOwnerOrAdmin) {
      const hasPermission = authUser.permissions.find(
        (p) => p.access && p.subAccountId === subaccountId
      );
      if (!hasPermission) {
        return NextResponse.json({ error: "No access" }, { status: 403 });
      }
    }

    // ── Pipeline — 5 stages ──────────────────────────────────────────────
    const projectId = crypto.randomUUID();
    let files: AIFileNode[] = [];

    try {
      files = await generateProjectWithPipeline(prompt);

      if (!Array.isArray(files) || files.length === 0) {
        throw new Error("Pipeline returned no files");
      }
    } catch (pipelineErr) {
      console.error("⚠️ Pipeline failed — using hard fallback:", pipelineErr);

      // Hard fallback: minimal runnable Vite+React app

          path: "package.json",

          type: "file",
        },

    }

    AIOrchestrator.setProjectFiles(projectId, files);
    console.log(`✅ Project ${projectId} stored with ${files.length} files`);

    // ── Persist to DB / in-memory project store ───────────────────────────

      name: name || prompt.slice(0, 50) || "AI Project",

      success: true,

      files,
      fileCount: files.length,
    });
  } catch (err) {
    console.error("❌ /api/ai/generate error:", err);

      { error: "Failed to generate project" },
      { status: 500 }
    );
  }

  prompt: string;
  projectId: string;

export const aiBuilderFunction = inngest.createFunction(
  { id: "ai-builder-run" },
  { event: "ai/build" }, // ✅ FIXED: was missing — this is why nothing triggered

  async ({ event, step }: any) => {
    const { prompt, projectId, subaccountId } = event.data as EventData;

    // ── Save user message ──────────────────────────────
    await step.run("save-user-message", async () => {
      await db.aIMessage.create({
        data: {
          projectId,
          role: "USER",
          content: prompt,
          status: "DONE",
        },
      });
    });

    // ── Create assistant placeholder ───────────────────
    const assistantMessageId = await step.run(
      "create-assistant-message",
      async () => {
        const msg = await db.aIMessage.create({
          data: {
            projectId,
            role: "ASSISTANT",
            content: "Building your app...",
            status: "PENDING",
          },
        });
        return msg.id;
      }
    );

    // ── Stage 1: Plan ──────────────────────────────────
    const plan = await step.run("plan-project", async () => {
      return planProject(prompt);
    });

    if (!plan) {
      await step.run("mark-failed", async () => {
        await db.aIMessage.update({
          where: { id: assistantMessageId },
          data: { content: "❌ Planning failed. Please try again.", status: "ERROR" },
        });
      });
      return { projectId, files: 0, error: "Planning failed" };
    }

    // ── Stage 2: Generate files in parallel ───────────
    const generatedFiles: AIFile[] = await step.run(
      "generate-code",
      async () => {
        const results = await generateFilesInParallel(plan);
        return results.map((f) => ({ path: f.path, content: f.content }));
      }
    );

    // ── Save files to DB ───────────────────────────────
    await step.run("save-files", async () => {
      if (!generatedFiles.length) return;

      await db.aIProjectFile.deleteMany({ where: { projectId } });

      await db.aIProjectFile.createMany({
        data: generatedFiles.map((f) => ({
          projectId,
          path: f.path,
          content: f.content,
        })),
      });
    });

    // ── Update assistant message ───────────────────────
    await step.run("update-message", async () => {
      await db.aIMessage.update({
        where: { id: assistantMessageId },
        data: {
          content: `🚀 Done! Generated ${generatedFiles.length} files. Preview is ready.`,
          status: "DONE",
          filesGenerated: generatedFiles.length,
        },
      });
    });

    return { projectId, files: generatedFiles.length };
  }
);

              className="w-full flex items-center gap-2 px-3 py-2.5 text-xs hover:bg-muted transition-colors">
              <Download className="w-3.5 h-3.5 text-muted-foreground" />
              Download ZIP
            </button>
            <button onClick={handlePushGithub}
              className="w-full flex items-center gap-2 px-3 py-2.5 text-xs hover:bg-muted transition-colors">
              <Github className="w-3.5 h-3.5 text-muted-foreground" />
              Push to GitHub
            </button>
          </div>
        )}
      </div>
      <button
        onClick={onDeploy}
        disabled={!projectId}
        className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-gradient-to-r from-violet-600 to-fuchsia-600 hover:from-violet-500 hover:to-fuchsia-500 text-xs font-medium text-white transition-all disabled:opacity-40 shadow-sm">
        <Rocket className="w-3 h-3" /> Deploy
      </button>
    </div>
  );
}

// ─── Main Workspace ────────────────────────────────────────────────────────────
const AIWorkspace = () => {
  const params = useParams();
  const subaccountId = params?.subaccountId as string;
  const projectId    = params?.projectId as string;

  const {
    setProject, setFiles, reset,
    files, projectId: storeProjectId,
    undo, redo, canUndo, canRedo,
  } = useAIBuilderStore();

  const { url: previewUrl } = useWebContainer();

  // Track which projectId we last fully loaded to avoid double-fetching.
  const loadedForProjectRef = useRef<string | null>(null);

  const [sidebarOpen, setSidebarOpen]     = useState(true);
  const [sidebarTab, setSidebarTab]       = useState<SidebarTab>("history");
  const [previewTab, setPreviewTab]       = useState<PreviewTab>("preview");
  const [device, setDevice]               = useState<Device>("desktop");
  const [showDeployDrawer, setShowDeployDrawer] = useState(false);
  const [exportMenuOpen, setExportMenuOpen]     = useState(false);

  // ── Reset store + WebContainer whenever the URL projectId changes ────────────
  // This is the fix for Bug 2: stale preview from previous project.
  // resetContainer() kills the dev server and clears _globalUrl immediately,
  // so LivePreview stops rendering the old iframe before new files load.
  useEffect(() => {
    if (!projectId || projectId === "new") return;
    if (loadedForProjectRef.current === projectId) return;

    // 1. Wipe the old project's WebContainer state (kills dev server, clears URL)
    resetContainer();

    // 2. Wipe the Zustand store so ChatPanel's isEditMode resets correctly
    reset();
    loadedForProjectRef.current = null;

    const load = async () => {
      try {
        const res = await fetch(`/api/ai/projects/${projectId}`);
        if (!res.ok) {
          console.warn(`[AIWorkspace] Project ${projectId} not found — new project`);
          setProject(projectId);
          loadedForProjectRef.current = projectId;
          return;
        }

        const data = await res.json();
        const loadedFiles = data?.project?.files ?? [];

        loadedForProjectRef.current = projectId;
        setProject(projectId);

        if (loadedFiles.length > 0) {
          setFiles(loadedFiles, "Loaded from DB");
        }
      } catch (err) {
        console.error("[AIWorkspace] Failed to load project files:", err);
      }
    };

    load();
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [projectId]);

  useEffect(() => {
    if (!exportMenuOpen) return;
    const handler = () => setExportMenuOpen(false);
    document.addEventListener("mousedown", handler);
    return () => document.removeEventListener("mousedown", handler);
  }, [exportMenuOpen]);

  const handleUndo = useCallback(() => undo?.(), [undo]);
  const handleRedo = useCallback(() => redo?.(), [redo]);

  return (
    <div className="w-full h-full flex flex-col bg-background overflow-hidden">
      <WorkspaceToolbar
        projectId={storeProjectId}
        subaccountId={subaccountId}
        previewUrl={previewUrl}
        onDeploy={() => setShowDeployDrawer(true)}
        canUndo={canUndo ?? false}
        canRedo={canRedo ?? false}
        onUndo={handleUndo}
        onRedo={handleRedo}
      />

      <div className="flex flex-1 min-h-0 overflow-hidden">
        {/* Sidebar */}
        <div className={cn(
          "shrink-0 flex flex-col border-r border-border overflow-hidden transition-all duration-200",
          sidebarOpen ? "w-52" : "w-0"
        )}>
          <div className="flex items-center shrink-0 border-b border-border">
            <button
              onClick={() => setSidebarTab("history")}
              className={cn(
                "flex items-center gap-1.5 flex-1 px-3 py-2 text-xs font-medium transition-colors",
                sidebarTab === "history"
                  ? "text-foreground border-b-2 border-primary"
                  : "text-muted-foreground hover:text-foreground"
              )}>
              <MessageSquare className="w-3 h-3" /> History
            </button>
            <button
              onClick={() => setSidebarTab("files")}
              className={cn(
                "flex items-center gap-1.5 flex-1 px-3 py-2 text-xs font-medium transition-colors",
                sidebarTab === "files"
                  ? "text-foreground border-b-2 border-primary"
                  : "text-muted-foreground hover:text-foreground"
              )}>
              <FolderOpen className="w-3 h-3" /> Files
            </button>
          </div>
          <div className="flex-1 overflow-hidden">
            {sidebarTab === "history"
              ? <ProjectSidebar subaccountId={subaccountId} />
              : <FileExplorer />}
          </div>
        </div>

        {/* Sidebar toggle */}
        <button
          onClick={() => setSidebarOpen((v) => !v)}
          title={sidebarOpen ? "Collapse sidebar" : "Expand sidebar"}
          className="shrink-0 self-center -ml-px z-10 w-5 h-10 flex items-center justify-center border border-border rounded-r-lg bg-background hover:bg-muted transition-colors text-muted-foreground hover:text-foreground">
          {sidebarOpen ? <PanelLeftClose className="w-3 h-3" /> : <PanelLeftOpen className="w-3 h-3" />}
        </button>

        {/* Chat */}
        <div className="w-[360px] shrink-0 flex flex-col border-r border-border overflow-hidden">
          <ChatPanel />
        </div>

        {/* Preview / Code */}
        <div className="flex-1 min-w-0 flex flex-col overflow-hidden">
          <div className="shrink-0 flex items-center gap-1 px-3 py-1.5 border-b border-border bg-background">
            <div className="flex items-center rounded-lg border border-border overflow-hidden">
              <button
                onClick={() => setPreviewTab("preview")}
                className={cn(
                  "px-2.5 py-1 text-xs flex items-center gap-1 transition-colors",
                  previewTab === "preview"
                    ? "bg-muted text-foreground"
                    : "text-muted-foreground hover:text-foreground"
                )}>
                <Monitor className="w-3 h-3" /> Preview
              </button>
              <button
                onClick={() => setPreviewTab("code")}
                className={cn(
                  "px-2.5 py-1 text-xs flex items-center gap-1 transition-colors",
                  previewTab === "code"
                    ? "bg-muted text-foreground"
                    : "text-muted-foreground hover:text-foreground"
                )}>
                <Code2 className="w-3 h-3" /> Code
              </button>
            </div>
            {previewTab === "preview" && (
              <div className="flex items-center rounded-lg border border-border overflow-hidden ml-1">
                {(["desktop", "tablet", "mobile"] as Device[]).map((d) => {
                  const Icon = d === "desktop" ? Monitor : d === "tablet" ? Tablet : Smartphone;
                  return (
                    <button
                      key={d}
                      onClick={() => setDevice(d)}
                      title={d.charAt(0).toUpperCase() + d.slice(1)}
                      className={cn(
                        "px-2 py-1 transition-colors",
                        device === d
                          ? "bg-muted text-foreground"
                          : "text-muted-foreground hover:text-foreground"
                      )}>
                      <Icon className="w-3 h-3" />
                    </button>
                  );
                })}
              </div>
            )}
          </div>
          <div className="flex-1 min-h-0 flex items-start justify-center overflow-hidden bg-muted/30">
            {previewTab === "preview" ? (
              <div
                className="h-full transition-all duration-300 overflow-hidden"
                style={{ width: DEVICE_WIDTHS[device] }}>
                {/* Pass storeProjectId so LivePreview can detect project switches */}
                <LivePreview projectId={storeProjectId} />
              </div>
            ) : (
              <div className="w-full h-full">
                <MonacoEditor />
              </div>
            )}
          </div>
        </div>
      </div>

      {showDeployDrawer && storeProjectId && (
        <DeployDrawer
          projectId={storeProjectId}
          subaccountId={subaccountId}
          onClose={() => setShowDeployDrawer(false)}
        />
      )}
    </div>
  );
};

export default AIWorkspace;
